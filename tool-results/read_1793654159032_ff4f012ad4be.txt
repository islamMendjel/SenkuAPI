    81→/** The /-token under the caret, if any. The "/" must sit at the very
    82→ *  start of the input or right after whitespace — typing "/" mid-word
    83→ *  ("a/b") or inside a URL ("https://x") never triggers the popover. */
    84→function activeSlashToken(
    85→  value: string,
    86→  caret: number
    87→): { query: string; start: number } | null {
    88→  const upto = value.slice(0, caret);
    89→  const m = /(^|\s)\/([\w-]{0,20})$/.exec(upto);
    90→  if (!m) return null;
    91→  return { query: m[2].toLowerCase(), start: caret - m[2].length - 1 };
    92→}

    94→function formatChars(n: number): string {
    95→  if (n >= 1000) return `${(n / 1000).toFixed(1)}k chars`;
    96→  return `${n} chars`;
