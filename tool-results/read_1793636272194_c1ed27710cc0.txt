   163→/** Fires exactly when the usage day rolls over (01:00 Africa/Algiers),
   164→ *  so the daily token counter visibly resets to 0 at that moment —
   165→ *  no polling loop, one timer armed from the server's own schedule. */
   166→let rolloverTimer: ReturnType<typeof setTimeout> | null = null;
   167→function scheduleDailyRolloverRefresh(dailyResetsAt?: string) {
   168→  if (rolloverTimer) {
   169→    clearTimeout(rolloverTimer);
   170→    rolloverTimer = null;
   171→  }
   172→  if (typeof dailyResetsAt !== "string") return;
   173→  const at = new Date(dailyResetsAt).getTime();
   174→  if (!Number.isFinite(at)) return;
   175→  const delay = Math.max(at - Date.now() + 1000, 30_000); // never tighter than 30s
   176→  rolloverTimer = setTimeout(() => {
   177→    rolloverTimer = null;
   178→    void useSenku.getState().refreshUsage();
   179→  }, delay);
   180→}

   183→  providers: ProviderView[],
   184→  view: ProviderView

   186→  return providers.map((p) => (p.id === view.id ? view : p));

   217→  async addAttachments(files) {
   218→    const existing = get().pendingAttachments;
   219→    const room = 5 - existing.length;
   220→    if (room <= 0) {
   221→      return; // caller toasts about the cap
   222→    }
   223→    const accepted = files.slice(0, room);
   224→    const skipped = files.length - accepted.length;
   225→    const staged: PendingAttachment[] = accepted.map((f) => ({
   226→      id: `p-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   227→      name: f.name,
   228→      kindGuess: f.type.startsWith("image/")
   229→        ? "image"
   230→        : f.type === "application/pdf" || f.name.toLowerCase().endsWith(".pdf")
   231→          ? "pdf"
   232→          : f.type.startsWith("text/") || f.name.includes(".")
   233→            ? "text"
   234→            : "unknown",
   235→      status: "extracting",
   236→    }));
   237→    set({ pendingAttachments: [...existing, ...staged] });
   238→    if (skipped > 0) return; // cap message handled by the caller
   239→
   240→    try {
   241→      const form = new FormData();
   242→      for (const f of accepted) form.append("files", f);
   243→      const res = await fetch("/api/attachments/extract", {
   244→        method: "POST",
   245→        body: form,
   246→      });
   247→      const data = (await res.json().catch(() => ({}))) as {
   248→        records?: AttachmentRecord[];
   249→        errors?: { name: string; error: string }[];
   250→        error?: string;
   251→      };
   252→      if (!res.ok) throw new Error(data.error || "Extraction failed");
   253→      const records = data.records ?? [];
   254→      const errors = data.errors ?? [];
   255→      set((s) => ({
   256→        pendingAttachments: s.pendingAttachments.map((p) => {
   257→          const rec = records.find((r) => r.name === p.name);
   258→          if (rec) {
   259→            return {
   260→              ...p,
   261→              status: "ready" as const,
   262→              kindGuess: rec.kind,
   263→              record: rec,
   264→            };
   265→          }
   266→          const err = errors.find((e) => e.name === p.name);
   267→          if (err) {
   268→            return { ...p, status: "error" as const, error: err.error };
   269→          }
   270→          // No result at all (name mismatch etc.) — treat as error.
   271→          return {
   272→            ...p,
   273→            status: "error" as const,
   274→            error: "Could not read this file.",
   275→          };
   276→        }),
   277→      }));
   278→    } catch (e) {
   279→      // Whole upload failed — every staged file errors out.
   280→      const msg = (e as Error).message || "Extraction failed";
   281→      set((s) => ({
   282→        pendingAttachments: s.pendingAttachments.map((p) =>
   283→          staged.some((q) => q.id === p.id)
   284→            ? { ...p, status: "error" as const, error: msg }
   285→            : p
   286→        ),
   287→      }));
   288→    }
   289→  },

   534→  async sendMessage(content) {
   535→    const state = get();
   536→    if (state.streaming || !content.trim()) return;
   537→    // A duel still awaiting its pick? The primary lane becomes the
   538→    // canonical answer so the next turn's history has one voice.
   539→    const pending = [...state.messages].reverse().find((m) => m.duel && !m.duelPicked);
   540→    if (pending) {
   541→      await get().pickDuelWinner(pending.id, "a");
   542→    }
   543→    const s2 = get();
   544→    const history: ChatMessageData[] = s2.messages.map((m) => ({
   545→      role: m.role,
   546→      content: m.content,
   547→      ...(m.role === "user" && m.attachments?.length
   548→        ? { attachments: m.attachments }
   549→        : {}),
   550→    }));
   551→    // Files staged in the composer: only READY records are sent (a
   552→    // file that couldn't be read is never sent — the model must not
   553→    // pretend it saw it). Failed/error chips stay for the user to
   554→    // remove or ignore.
   555→    const attachments = s2.pendingAttachments
   556→      .filter((p) => p.status === "ready" && p.record)
   557→      .map((p) => p.record!);
   558→    const stillExtracting = s2.pendingAttachments.some(
   559→      (p) => p.status === "extracting"
   560→    );
   561→    if (stillExtracting) return; // InputBar disables send meanwhile
   562→    set({ pendingAttachments: [] });
   563→    // Resolve the challenger at send time — the model list may have
   564→    // changed since duel was armed.
   565→    let challenger = s2.challenger;
   566→    if (challenger === s2.model || !s2.models.includes(challenger ?? "")) {
   567→      challenger = autoChallenger(s2.models, s2.model);
   568→    }
   569→    const duel =
   570→      s2.duelEnabled && challenger ? { challenger } : undefined;
   571→    await streamTurn(set, get, {
   572→      chatId: s2.temporary ? null : s2.currentChatId,
   573→      history,
   574→      content,
   575→      persistUser: true,
   576→      duel,
   577→      attachments,
   578→    });
   579→  },

   581→  async editMessage(messageId, content) {
   582→    const state = get();
   583→    if (state.streaming || !content.trim()) return;
   584→    const idx = state.messages.findIndex((m) => m.id === messageId);
   585→    if (idx === -1) return;
   586→    if (state.temporary || !state.currentChatId) {
   587→      // Local-only timeline: truncate everything after the edited message.
   588→      const kept = state.messages
   589→        .slice(0, idx + 1)
   590→        .map((m) => (m.id === messageId ? { ...m, content: content.trim() } : m));
   591→      set({ messages: kept });
   592→      const history: ChatMessageData[] = kept.slice(0, -1).map((m) => ({
   593→        role: m.role,
   594→        content: m.content,
   595→        ...(m.role === "user" && m.attachments?.length
   596→          ? { attachments: m.attachments }
   597→          : {}),
   598→      }));
   599→      const edited = kept[kept.length - 1];
   600→      await streamTurn(set, get, {
   601→        chatId: null,
   602→        history,
   603→        content: content.trim(),
   604→        persistUser: false,
   605→        temporary: true,
   606→        attachments: edited?.role === "user" ? edited.attachments ?? [] : [],
   607→      });
   608→      return;
   609→    }
   610→
   611→    const res = await fetch("/api/chat/truncate", {
   612→      method: "POST",
   613→      headers: { "Content-Type": "application/json" },
   614→      body: JSON.stringify({ messageId, content: content.trim() }),
   615→    });
   616→    const data = await res.json();
   617→    if (!Array.isArray(data.messages)) return;
   618→    const kept: UIMessage[] = data.messages.map((m: UIMessage) => ({
   619→      ...m,
   620→      sources: m.sources ?? null,
   621→      attachments: m.attachments ?? null,
   622→    }));
   623→    set({ messages: kept });
   624→    const history: ChatMessageData[] = kept.slice(0, -1).map((m) => ({
   625→      role: m.role,
   626→      content: m.content,
   627→      ...(m.role === "user" && m.attachments?.length
   628→        ? { attachments: m.attachments }
   629→        : {}),
   630→    }));
   631→    const edited = kept[kept.length - 1];
   632→    await streamTurn(set, get, {
   633→      chatId: state.currentChatId,
   634→      history,
   635→      content: content.trim(),
   636→      persistUser: false,
   637→      attachments: edited?.role === "user" ? edited.attachments ?? [] : [],
   638→    });
   639→  },
   640→
   641→  async retryLast(modelOverride) {
   642→    const state = get();
   643→    if (state.streaming) return;
   644→
   645→    // The most recent failed assistant reply.
   646→    let failedIdx = -1;
   647→    for (let i = state.messages.length - 1; i >= 0; i--) {
   648→      const m = state.messages[i];
   649→      if (m.role === "assistant" && m.error) {
   650→        failedIdx = i;
   651→        break;
   652→      }
   653→    }
   654→    if (failedIdx === -1) return;
   655→
   656→    // The user message that failed turn was answering.
   657→    let userIdx = -1;
   658→    for (let i = failedIdx - 1; i >= 0; i--) {
   659→      if (state.messages[i].role === "user") {
   660→        userIdx = i;
   661→        break;
   662→      }
   663→    }
   664→    if (userIdx === -1) return;
   665→    const userMsg = state.messages[userIdx];
   666→
   667→    const history: ChatMessageData[] = state.messages
   668→      .slice(0, userIdx)
   669→      .map((m) => ({
   670→        role: m.role,
   671→        content: m.content,
   672→        ...(m.role === "user" && m.attachments?.length
   673→          ? { attachments: m.attachments }
   674→          : {}),
   675→      }));
   676→
   677→    // Drop the failed bubble from the timeline.
   678→    set({ messages: state.messages.slice(0, failedIdx) });
   679→
   680→    // If the user message never received a server id, the turn died
   681→    // before persistence — persist it on retry. Otherwise it is
   682→    // already in the DB and must not be duplicated.
   683→    const persisted = !userMsg.id.startsWith("u-");
   684→    await streamTurn(set, get, {
   685→      chatId: state.temporary ? null : state.currentChatId,
   686→      history,
   687→      content: userMsg.content,
   688→      persistUser: !persisted,
   689→      appendUserBubble: false, // the original user bubble stays
   690→      temporary: state.temporary,
   691→      model: modelOverride,
   692→      // The retried turn sees the same files again.
   693→      attachments: userMsg.attachments ?? [],
   694→    });
   695→  },
   696→
   697→  async pickDuelWinner(messageId, laneId) {
   698→    const s = get();
   699→    const msg = s.messages.find((m) => m.id === messageId);
   700→    if (!msg?.duel || msg.duelPicked) return;
   701→    const winner = msg.duel[laneId];
   702→    const loser = msg.duel[laneId === "a" ? "b" : "a"];
   703→    if (!winner.content.trim()) return; // nothing to keep
   704→    const providerName = s.limiter?.providerName ?? null;
   705→
   706→    // Local commit first — the pick must feel instant.
   707→    set({
   708→      messages: s.messages.map((m) =>
   709→        m.id === messageId
   710→          ? {
   711→              ...m,
   712→              content: winner.content,
   713→              model: winner.model,
   714→              tokens: winner.tokens,
   715→              cached: winner.cached,
   716→              servedBy: winner.servedBy,
   717→              keyLabel: winner.keyLabel,
   718→              provider: providerName,
   719→              duel: null,
   720→              duelPicked: laneId,
   721→              runnerUp: loser.content.trim()
   722→                ? { content: loser.content, model: loser.model }
   723→                : null,
   724→              streaming: false,
   725→              phase: null,
   726→            }
   727→          : m
   728→      ),
   729→    });
   730→
   731→    // Persist for saved chats (temporary duels stay local-only).
   732→    if (!s.temporary && s.currentChatId) {
   733→      try {
   734→        const res = await fetch("/api/chat/duel/pick", {
   735→          method: "POST",
   736→          headers: { "Content-Type": "application/json" },
   737→          body: JSON.stringify({
   738→            chatId: s.currentChatId,
   739→            winner: {
   740→              content: winner.content,
   741→              model: winner.model,
   742→              tokens: winner.tokens,
   743→              cached: winner.cached,
   744→              servedBy: winner.servedBy,
   745→              keyLabel: winner.keyLabel,
   746→            },
   747→            runnerUp: loser.content.trim()
   748→              ? { content: loser.content, model: loser.model }
   749→              : null,
   750→            effort: s.effort,
   751→          }),
   752→        });
   753→        const data = (await res.json()) as { id?: string; error?: string };
   754→        if (typeof data.id === "string" && data.id) {
   755→          // Reconcile with the server id so later edits/truncates hit
   756→          // the real message row.
   757→          set((st) => ({
   758→            messages: st.messages.map((m) =>
   759→              m.id === messageId ? { ...m, id: data.id as string } : m
   760→            ),
   761→          }));
   762→        }
   763→      } catch {
   764→        // keep the local pick — the answer stays in the timeline
   765→      }
   766→    }
   767→  },
   768→
   769→  stopStreaming: () => {
   770→    streamController?.abort();
   771→    streamController = null;
   772→  },
   773→}));
   774→
   775→// ── SSE turn pipeline ─────────────────────────────────────────
   776→type SetState = (
   777→  partial: Partial<SenkuState> | ((s: SenkuState) => Partial<SenkuState>)
   778→) => void;
   779→type GetState = () => SenkuState;
   780→
   781→interface TurnOptions {
   782→  chatId: string | null;
   783→  history: ChatMessageData[];
   784→  content: string;
   785→  persistUser: boolean;
   786→  temporary?: boolean;
   787→  /** Overrides the store's active model (error-card "switch model"). */
   788→  model?: string;
   789→  /** Force whether a new local user bubble is appended. Defaults to
   790→      the persistUser flag — retries keep the original bubble but may
   791→      still need server-side persistence, so the two are decoupled. */
   792→  appendUserBubble?: boolean;
   793→  /** Duel: race the active model against `duel.challenger`. */
   794→  duel?: { challenger: string };
   795→  /** Files the gateway already read for this turn. */
   796→  attachments?: AttachmentRecord[];
   797→}
   798→
   799→async function streamTurn(
   800→  set: SetState,
   801→  get: GetState,
   802→  opts: TurnOptions
   803→): Promise<void> {
   804→  const state = get();
   805→  const temporary = opts.temporary ?? state.temporary;
   806→  const model = opts.model ?? state.model;
   807→  const effort = state.effort;
   808→  const userId = `u-${Date.now()}`;
   809→  const assistantId = `a-${Date.now()}`;
   810→  const search = state.searchEnabled;
   811→  const duel = opts.duel;
   812→  const attachments = opts.attachments ?? [];
   813→  const makeLane = (id: "a" | "b", laneModel: string): DuelLane => ({
   814→    id,
   815→    model: laneModel,
   816→    content: "",
   817→    streaming: true,
   818→    error: null,
   819→    phase: search ? ("searching" as const) : ("thinking" as const),
   820→    tokens: 0,
   821→    cached: false,
   822→    servedBy: null,
   823→    keyLabel: null,
   824→  });
   825→  // Regenerate-after-edit turns keep the (already updated) user message
   826→  // in the timeline — only fresh sends append a new local user bubble.
   827→  // Retries decouple the two: the bubble stays, persistence is decided
   828→  // by whether the failed turn ever reached the server.
   829→  const appendUserBubble = opts.appendUserBubble ?? (opts.persistUser !== false);
   830→  // Set when this turn creates a brand-new persisted chat — the
   831→  // raw derived title then gets refined by a cheap model pass.
   832→  let createdChatId: string | null = null;
   833→
   834→  set({
   835→    streaming: true,
   836→    messages: [
   837→      ...state.messages,
   838→      ...(appendUserBubble
   839→        ? [
   840→            {
   841→              id: userId,
   842→              role: "user" as const,
   843→              content: opts.content,
   844→              sources: search ? [] : null,
   845→              attachments: attachments.length ? attachments : null,
   846→            },
   847→          ]
   848→        : []),
   849→      {
   850→        id: assistantId,
   851→        role: "assistant" as const,
   852→        content: "",
   853→        streaming: true,
   854→        model,
   855→        provider: state.limiter?.providerName ?? null,
   856→        phase: search ? "searching" : "thinking",
   857→        effort,
   858→        ...(duel
   859→          ? {
   860→              duel: {
   861→                a: makeLane("a", model),
   862→                b: makeLane("b", duel.challenger),
   863→              },
   864→              duelPicked: null,
   865→            }
   866→          : {}),
   867→      },
   868→    ],
   869→  });
   870→
   871→  const updateAssistant = (patch: Partial<UIMessage>) => {
   872→    set((s) => ({
   873→      messages: s.messages.map((m) =>
   874→        m.id === assistantId ? { ...m, ...patch } : m
   875→      ),
   876→    }));
   877→  };
   878→  // Duel-only: patch one racing lane of the assistant message.
   879→  const updateLane = (lane: "a" | "b", patch: Partial<DuelLane>) => {
   880→    set((s) => ({
   881→      messages: s.messages.map((m) =>
   882→        m.id === assistantId && m.duel
   883→          ? { ...m, duel: { ...m.duel, [lane]: { ...m.duel[lane], ...patch } } }
   884→          : m
   885→      ),
   886→    }));
   887→  };
   888→  const updateUser = (patch: Partial<UIMessage>) => {
   889→    set((s) => {
   890→      const idx = appendUserBubble
   891→        ? s.messages.findIndex((m) => m.id === currentUserMsgId)
   892→        : (() => {
   893→            for (let i = s.messages.length - 1; i >= 0; i--) {
   894→              if (s.messages[i].role === "user") return i;
   895→            }
   896→            return -1;
   897→          })();
   898→      if (idx === -1) return {};
   899→      return {
   900→        messages: s.messages.map((m, i) => (i === idx ? { ...m, ...patch } : m)),
   901→      };
   902→    });
   903→  };
   904→
   905→  const controller = new AbortController();
   906→  streamController = controller;
   907→  // Reconciled with the server-assigned id as soon as "meta" arrives so
   908→  // later edits hit /api/chat/truncate with the real message id.
   909→  let currentUserMsgId = userId;
   910→
   911→  try {
   912→    const res = await fetch("/api/chat/send", {
   913→      method: "POST",
   914→      headers: { "Content-Type": "application/json" },
   915→      signal: controller.signal,
   916→      body: JSON.stringify({
   917→        chatId: opts.chatId,
   918→        history: opts.history,
   919→        content: opts.content,
   920→        model,
   921→        search,
   922→        temporary,
   923→        persistUser: opts.persistUser,
   924→        effort,
   925→        ...(attachments.length ? { attachments } : {}),
   926→        ...(duel ? { duel: true, challengerModel: duel.challenger } : {}),
   927→      }),
   928→    });
   929→
   930→    if (!res.ok || !res.body) {
   931→      const errText = await res.text().catch(() => "");
   932→      throw new Error(errText || `Request failed (${res.status})`);
   933→    }
   934→
   935→    const reader = res.body.getReader();
   936→    const decoder = new TextDecoder();
   937→    let buffer = "";
   938→    let assistantContent = "";
   939→    // Duel lanes accumulate independently; untagged deltas (normal
   940→    // turns) fall through to assistantContent only.
   941→    const laneContents: Record<"a" | "b", string> = { a: "", b: "" };
   942→
   943→    while (true) {
   944→      const { done, value } = await reader.read();
   945→      if (done) break;
   946→      buffer += decoder.decode(value, { stream: true });
   947→      const parts = buffer.split("\n\n");
   948→      buffer = parts.pop() ?? "";
   949→      for (const part of parts) {
   950→        const line = part.trim();
   951→        if (!line.startsWith("data:")) continue;
   952→        let ev: Record<string, unknown>;
   953→        try {
   954→          ev = JSON.parse(line.slice(5).trim());
   955→        } catch {
   956→          continue;
   957→        }
   958→        if (ev.type === "meta") {
   959→          if (!temporary && typeof ev.chatId === "string" && ev.chatId) {
   960→            set({ currentChatId: ev.chatId });
   961→            if (opts.chatId == null) createdChatId = ev.chatId;
   962→          }
   963→          // The effort dial may route the turn to a different pooled
   964→          // engine (fast → flash tier, deep → frontier tier) — the
   965→          // server's model is the truth the provenance must show.
   966→          if (typeof ev.model === "string" && ev.model) {
   967→            updateAssistant({ model: ev.model });
   968→          }
   969→          if (
   970→            typeof ev.userMessageId === "string" &&
   971→            ev.userMessageId &&
   972→            appendUserBubble
   973→          ) {
   974→            const serverId = ev.userMessageId;
   975→            // patch FIRST (message still carries the local id), then move
   976→            // the cursor so later patches (sources, …) target the new id
   977→            updateUser({ id: serverId });
   978→            currentUserMsgId = serverId;
   979→          }
   980→        } else if (ev.type === "duel_start") {
   981→          // Reconcile lane models with the server's normalization.
   982→          if (duel) {
   983→            if (typeof ev.a === "string" && ev.a) updateLane("a", { model: ev.a });
   984→            if (typeof ev.b === "string" && ev.b) updateLane("b", { model: ev.b });
   985→          }
   986→        } else if (ev.type === "search_status") {
   987→          const status = String(ev.status ?? "");
   988→          const detail =
   989→            typeof ev.detail === "string" && ev.detail ? ev.detail : null;
   990→          if (status === "searching") {
   991→            updateAssistant({ phase: "searching", searchDetail: detail });
   992→            if (duel) {
   993→              updateLane("a", { phase: "searching" });
   994→              updateLane("b", { phase: "searching" });
   995→            }
   996→          } else if (status === "ok") {
   997→            updateAssistant({ phase: "thinking", searchFailed: false, searchDetail: null });
   998→            if (duel) {
   999→              updateLane("a", { phase: "thinking" });
  1000→              updateLane("b", { phase: "thinking" });
  1001→            }
  1002→          } else if (status === "failed") {
  1003→            updateAssistant({ phase: "thinking", searchFailed: true, searchDetail: null });
  1004→            if (duel) {
  1005→              updateLane("a", { phase: "thinking" });
  1006→              updateLane("b", { phase: "thinking" });
  1007→            }
  1008→          }
  1009→        } else if (ev.type === "sources") {
  1010→          const sources = (ev.sources ?? []) as SearchSource[];
  1011→          if (sources.length) updateUser({ sources });
  1012→        } else if (ev.type === "restart") {
  1013→          // A provider died mid-stream; the cloud re-generates that
  1014→          // lane from scratch. Discard only the affected lane's
  1015→          // partial text (untagged = normal single-lane turn).
  1016→          const lane = ev.lane === "b" ? "b" : "a";
  1017→          if (duel) {
  1018→            laneContents[lane] = "";
  1019→            updateLane(lane, { content: "", phase: "thinking" });
  1020→          } else {
  1021→            assistantContent = "";
  1022→            updateAssistant({ content: "", phase: "thinking" });
  1023→          }
  1024→        } else if (ev.type === "delta") {
  1025→          const lane = ev.lane === "b" ? "b" : "a";
  1026→          if (duel) {
  1027→            laneContents[lane] += String(ev.text ?? "");
  1028→            updateLane(lane, { content: laneContents[lane], phase: null });
  1029→          } else {
  1030→            assistantContent += String(ev.text ?? "");
  1031→            updateAssistant({ content: assistantContent, phase: null });
  1032→          }
  1033→        } else if (ev.type === "lane_done") {
  1034→          // Final per-lane stats from the gateway — server truth wins.
  1035→          const lane = ev.lane === "b" ? "b" : "a";
  1036→          if (duel) {
  1037→            updateLane(lane, {
  1038→              content: (ev.content as string) ?? laneContents[lane],
  1039→              streaming: false,
  1040→              phase: null,
  1041→              tokens: (ev.tokens as number) ?? 0,
  1042→              cached: ev.cached === true,
  1043→              servedBy: (ev.servedBy as string) ?? null,
  1044→              keyLabel: (ev.keyLabel as string) ?? null,
  1045→              error: (ev.error as string) ?? null,
  1046→            });
  1047→          }
  1048→        } else if (ev.type === "done") {
  1049→          if (duel && ev.duel && typeof ev.duel === "object") {
  1050→            const lanes = ev.duel as {
  1051→              a: { content: string; error: string | null; tokens: number; cached: boolean; servedBy: string | null; keyLabel: string | null; model: string };
  1052→              b: { content: string; error: string | null; tokens: number; cached: boolean; servedBy: string | null; keyLabel: string | null; model: string };
  1053→            };
  1054→            updateAssistant({
  1055→              streaming: false,
  1056→              phase: null,
  1057→              searchFailed: ev.searchFailed === true,
  1058→            });
  1059→            updateLane("a", {
  1060→              content: lanes.a.content,
  1061→              model: lanes.a.model,
  1062→              streaming: false,
  1063→              phase: null,
  1064→              tokens: lanes.a.tokens,
  1065→              cached: lanes.a.cached,
  1066→              servedBy: lanes.a.servedBy,
  1067→              keyLabel: lanes.a.keyLabel,
  1068→              error: lanes.a.error,
  1069→            });
  1070→            updateLane("b", {
  1071→              content: lanes.b.content,
  1072→              model: lanes.b.model,
  1073→              streaming: false,
  1074→              phase: null,
  1075→              tokens: lanes.b.tokens,
  1076→              cached: lanes.b.cached,
  1077→              servedBy: lanes.b.servedBy,
  1078→              keyLabel: lanes.b.keyLabel,
  1079→              error: lanes.b.error,
  1080→            });
  1081→            // One lane died → the duel silently degrades: keep the
  1082→            // surviving lane as the answer (no pick needed).
  1083→            if (lanes.a.error && !lanes.b.error && lanes.b.content.trim()) {
  1084→              void get().pickDuelWinner(assistantId, "b");
  1085→            } else if (lanes.b.error && !lanes.a.error && lanes.a.content.trim()) {
  1086→              void get().pickDuelWinner(assistantId, "a");
  1087→            }
  1088→            // Both lanes healthy → the message waits for the user's pick.
  1089→          } else {
  1090→            updateAssistant({
  1091→              id: (ev.assistantMessageId as string) ?? assistantId,
  1092→              content: (ev.content as string) ?? assistantContent,
  1093→              streaming: false,
  1094→              cached: Boolean(ev.cached),
  1095→              tokens: (ev.tokens as number) ?? 0,
  1096→              servedBy: (ev.servedBy as string) ?? null,
  1097→              keyLabel: (ev.keyLabel as string) ?? null,
  1098→              provider: (ev.providerName as string) ?? null,
  1099→              model: (ev.model as string) ?? model,
  1100→              effort:
  1101→                ev.effort === "fast" || ev.effort === "deep" || ev.effort === "medium"
  1102→                  ? ev.effort
  1103→                  : effort,
  1104→              phase: null,
  1105→              searchDetail: null,
  1106→              searchFailed: ev.searchFailed === true,
  1107→            });
  1108→          }
  1109→        } else if (ev.type === "error") {
  1110→          updateAssistant({
  1111→            streaming: false,
  1112→            error: String(ev.message ?? "Generation failed"),
  1113→            content: "",
  1114→            phase: null,
  1115→          });
  1116→        }
  1117→      }
  1118→    }
  1119→  } catch (e) {
  1120→    const err = e as Error;
  1121→    if (err.name === "AbortError") {
  1122→      // User pressed stop: a stopped duel keeps both partial lanes —
  1123→      // the user can still pick either one as the answer.
  1124→      if (duel) {
  1125→        updateAssistant({ streaming: false, phase: null });
  1126→        updateLane("a", { streaming: false, phase: null });
  1127→        updateLane("b", { streaming: false, phase: null });
  1128→      } else {
  1129→        updateAssistant({ streaming: false, phase: null });
  1130→      }
  1131→    } else {
  1132→      updateAssistant({
  1133→        streaming: false,
  1134→        error: err.message,
  1135→        content: "",
  1136→        phase: null,
  1137→      });
  1138→    }
  1139→  } finally {
  1140→    streamController = null;
  1141→    set({ streaming: false });
  1142→    if (!temporary) {
  1143→      // Await the refresh so the raw derived title is in the list
  1144→      // BEFORE the refined one patches it (no clobber race).
  1145→      await get().refreshChats();
  1146→      if (createdChatId) void refineTitle(set, createdChatId);
  1147→    }
  1148→    void get().refreshUsage();
  1149→  }
  1150→}

  1152→/**
  1153→ * Replace a brand-new chat's raw derived title ("who r u?") with a
  1154→ * proper one from a cheap cloud model pass: "Who Are You?". Fire and
  1155→ * forget — failure keeps the derived title.
  1156→ */
  1157→async function refineTitle(set: SetState, chatId: string): Promise<void> {
  1158→  try {
  1159→    const res = await fetch("/api/chat/title", {
  1160→      method: "POST",
  1161→      headers: { "Content-Type": "application/json" },
  1162→      body: JSON.stringify({ chatId }),
  1163→    });
  1164→    if (!res.ok) return;
  1165→    const data = (await res.json()) as { title?: string };
  1166→    const title = typeof data.title === "string" ? data.title.trim() : "";
  1167→    if (!title) return;
  1168→    set((s) => ({
  1169→      chats: s.chats.map((c) => (c.id === chatId ? { ...c, title } : c)),
  1170→    }));
  1171→  } catch {
  1172→    // keep the derived title
  1173→  }
  1174→}
