    48→function CopyButton({ text }: { text: string }) {
    49→  const [copied, setCopied] = useState(false);

    51→    <button
    52→      onClick={async () => {
    53→        try {
    54→          await navigator.clipboard.writeText(text);

    74→function SourcesBlock({ sources }: { sources: NonNullable<UIMessage["sources"]> }) {
    75→  // >3 results can only come from the deep-research pipeline

    79→    <div className="mt-3 rounded-xl border border-zinc-800 bg-[#1d1d20] p-3">

    82→        {deep
    83→          ? `Deep research · ${sources.length} sources`
    84→          : `Searched the web · ${sources.length} result${

    89→        className={`space-y-2.5${deep ? " senku-scroll max-h-96 overflow-y-auto pr-1" : ""}`}

   103→                rel="noopener noreferrer"

   238→/** One quiet provenance line: "GLM 5.3 · APInex pool · 1.9k tokens · cache: hit". */
   239→function provenanceLine(m: UIMessage): string {
   240→  const parts: string[] = [];
   241→  // The effort dial leaves its mark — fast/deep turns say so.
   242→  if (m.effort === "fast") parts.push("Fast mode");
   243→  if (m.effort === "deep") parts.push("Deep research");
   244→  if (m.model) parts.push(modelMeta(m.model).name);
   245→  if (m.provider) {
   246→    parts.push(
   247→      m.servedBy === "zai-cloud"
   248→        ? `${m.provider} · cloud failover`
   249→        : `${m.provider} pool`
   250→    );
   251→  }
   252→  if (typeof m.tokens === "number" && m.tokens > 0) {
   253→    parts.push(`${formatCompact(m.tokens)} tokens`);
   254→  }
   255→  parts.push(m.cached ? "cache: hit" : "cache: miss");
   256→  return parts.join(" · ");
   257→}

   259→/** Body without the trailing label line (display only). */
   260→function withoutConfidenceLine(content: string): string {
   261→  if (!confidenceOf(content)) return content;
   262→  const lines = content.trimEnd().split("\n");
   263→  return lines.slice(0, -1).join("\n").trimEnd();
   264→}

   380→/** A duel turn awaiting its pick: two models racing side by side
   381→    (stacked on mobile), each with its own Keep button. */
   382→function DuelMessage({ message }: { message: UIMessage }) {
   383→  const { pickDuelWinner } = useSenku();
   384→  const { toast } = useToast();
   385→  const duel = message.duel;
   386→  if (!duel) return null;
   387→  // Picks unlock once the turn finishes (or is stopped) — picking
   388→  // mid-stream would race the lane_done/done reconciliation.
   389→  const canPick = !message.streaming;
   390→
   391→  const keep = (lane: "a" | "b") => {
   392→    const meta = modelMeta(duel[lane].model);
   393→    void pickDuelWinner(message.id, lane).then(() => {
   394→      toast({
   395→        description: `Kept ${meta.name} — the runner-up stays under the answer.`,
   396→      });
   397→    });
   398→  };
   399→
   400→  return (
   401→    <div>
   402→      <div className="mb-2 flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-amber-300/70">
   403→        <Swords className="h-3 w-3" aria-hidden />
   404→        Duel — pick the keeper
   405→      </div>
   406→      <div className="grid grid-cols-1 items-start gap-3 md:grid-cols-2">
   407→        <DuelLaneCard lane={duel.a} canPick={canPick} onKeep={() => keep("a")} />
   408→        <DuelLaneCard lane={duel.b} canPick={canPick} onKeep={() => keep("b")} />
   409→      </div>
   410→      <p className="mt-2 text-[11px] text-zinc-600">
   411→        Both answers are billed separately ·{" "}
   412→        {canPick
   413→          ? "pick one to keep it in the chat"
   414→          : "generating both answers…"}
   415→      </p>
   416→    </div>
   417→  );
   418→}

   420→function AssistantMessage({
   421→  message,
   422→  speech,
   423→  onToggleSpeech,
   424→}: {
   425→  message: UIMessage;
   426→  speech: SpeechState;
   427→  onToggleSpeech: (m: UIMessage) => void;
   428→}) {
   429→  const { models, model, setModel, retryLast, streaming } = useSenku();
   430→  const isThinking = message.streaming && !message.content && !message.error;
   431→  const isSpeakingThis = speech?.id === message.id;
   432→  const speechLoading = isSpeakingThis && speech.phase === "loading";
   433→  const speechPlaying = isSpeakingThis && speech.phase === "playing";
   434→
   435→  // Error recovery: the next model in the list (wraps around).
   436→  const currentIdx = models.indexOf(model);
   437→  const switchTarget =
   438→    models.length > 1
   439→      ? models[((currentIdx === -1 ? 0 : currentIdx) + 1) % models.length]
   440→      : null;
   441→
   442→  // Grounded-answer contract: the trailing confidence label becomes
   443→  // a badge once streaming completes (while streaming, the final
   444→  // line may still be half-written — render it raw).
   445→  const confidence = message.error ? null : confidenceOf(message.content);
   446→  const displayContent =
   447→    !message.streaming && confidence
   448→      ? withoutConfidenceLine(message.content)
   449→      : message.content;
   450→
   451→  return (
   452→    <div className="group senku-fade-up">
   453→      <div className="flex gap-3">
   454→        {/* While thinking, our own WHITE sparkle logo pulses with the
   455→            exact scale curve measured from the Claude Code reference
   456→            GIF — same motion, our colors — beside a shimmering label. */}
   457→        <Image
   458→          src="/senku-sparkle.png"
   459→          alt="SenkuAPI"
   460→          width={28}
   461→          height={28}
   462→          className={`mt-0.5 h-7 w-7 shrink-0${
   463→            isThinking ? " senku-thinking-sparkle" : ""
   464→          }`}
   465→        />
   466→        <div className="min-w-0 flex-1">
   467→          {message.error ? (
   468→            <div
   469→              className="rounded-xl border border-red-500/30 bg-red-500/10 px-3.5 py-2.5 text-[13px] text-red-300"
   470→              role="alert"
   471→            >
   472→              <p className="leading-relaxed">{message.error}</p>
   473→              <div className="mt-2.5 flex flex-wrap items-center gap-2">
   474→                <button
   475→                  onClick={() => void retryLast()}
   476→                  disabled={streaming}
   477→                  className="flex h-7 items-center gap-1.5 rounded-full border border-zinc-500/60 bg-white/10 px-3 text-[12px] font-medium text-zinc-100 transition-colors hover:bg-white/20 disabled:opacity-40"
   478→                >
   479→                  <RefreshCcw className="h-3 w-3" />
   480→                  Retry
   481→                </button>
   482→                {switchTarget && (
   483→                  <button
   484→                    onClick={() => {
   485→                      setModel(switchTarget);
   486→                      void retryLast(switchTarget);
   487→                    }}
   488→                    disabled={streaming}
   489→                    className="flex h-7 items-center gap-1.5 rounded-full border border-zinc-700 px-3 text-[12px] text-zinc-400 transition-colors hover:bg-white/5 hover:text-zinc-200 disabled:opacity-40"
   490→                    title={`Retry with ${modelMeta(switchTarget).name}`}
   491→                  >
   492→                    <ArrowLeftRight className="h-3 w-3" />
   493→                    Switch model
   494→                  </button>
   495→                )}
   496→              </div>
   497→            </div>
   498→          ) : message.duel ? (
   499→            /* Awaiting-pick duel: two racing lanes instead of one
   500→               answer — per-lane thinking, streaming, badges, Keep. */
   501→            <DuelMessage message={message} />
   502→          ) : isThinking ? (
   503→            <div className="flex h-7 items-center" aria-live="polite">
   504→              <span className="senku-thinking-label text-[14px] font-medium">
   505→                {message.phase === "searching"
   506→                  ? message.searchDetail ?? "Searching the web"
   507→                  : message.effort === "deep"
   508→                    ? "Thinking deeply"
   509→                    : "Thinking"}
   510→              </span>
   511→            </div>
   512→          ) : (
   513→            <StreamedMarkdown
   514→              content={displayContent}
   515→              streaming={Boolean(message.streaming)}
   516→            />
   517→          )}
   518→
   519→          {/* grounded-answer confidence badge (search turns) */}
   520→          {!message.streaming && !message.duel && confidence && (
   521→            <ConfidenceBadge label={confidence} />
   522→          )}
   523→
   524→          {/* search-failure notice (all engines down) */}
   525→          {!message.streaming && message.searchFailed && !message.error && (
   526→            <div className="mt-2 flex items-center gap-1.5 text-[11.5px] text-amber-400/80">
   527→              <TriangleAlert className="h-3.5 w-3.5 shrink-0" />
   528→              Web search unavailable — answered from model knowledge.
   529→            </div>
   530→          )}
   531→
   532→          {/* quiet provenance line + hover actions (desktop reveals on
   533→              hover; touch devices keep it faintly visible) */}
   534→          {!message.streaming && !message.error && !message.duel && (
   535→            <div className="mt-1.5 flex items-center gap-1 senku-meta-row">
   536→              <CopyButton text={message.content} />
   537→              {message.content.trim() && (

   576→                {provenanceLine(message)}

   578→            </div>
   579→          )}
   580→
   581→          {/* duel runner-up: the losing lane stays viewable under the
   582→              kept answer (collapsed, muted — a reference, not the reply) */}

   585→              <summary
   586→                className="flex w-fit cursor-pointer select-none items-center gap-1.5 text-[11.5px] text-zinc-500 transition-colors hover:text-zinc-300 [&::-webkit-details-marker]:hidden"

   588→              >
   589→                <ChevronDown
   590→                  className="h-3 w-3 transition-transform"

   592→                />
   593→                Runner-up · {modelMeta(message.runnerUp.model).name}
   594→              </summary>
   595→              <div className="senku-md mt-2 rounded-xl border border-zinc-800/70 bg-[#1d1d20]/60 px-3.5 py-2.5">
   596→                <Markdown
   597→                  content={
   598→                    confidenceOf(message.runnerUp.content)
   599→                      ? withoutConfidenceLine(message.runnerUp.content)
   600→                      : message.runnerUp.content
   601→                  }
   602→                />
   603→              </div>
   604→            </details>
   605→          )}
   606→        </div>
   607→      </div>
   608→    </div>
   609→  );

   626→export function ChatTimeline() {
   627→  const { messages } = useSenku();
   628→  const { toast } = useToast();
   629→  const bottomRef = useRef<HTMLDivElement>(null);
   630→  const scrollRef = useRef<HTMLDivElement>(null);
   631→  // Auto-scroll stickiness: following the newest word pauses the
   632→  // moment the user scrolls up to read and resumes when they return
   633→  // to the bottom (every StreamedMarkdown consults the same state).
   634→  const stickRef = useStickyScroll(scrollRef);
   635→
   636→  // ── Read-aloud (TTS) playback state ──
   637→  // One audio element for the whole timeline; only one message can
   638→  // speak at a time. A token guards against stale fetches after a stop.
   639→  const [speech, setSpeech] = useState<SpeechState>(null);
   640→  const audioRef = useRef<HTMLAudioElement | null>(null);
   641→  const audioUrlRef = useRef<string | null>(null);
   642→  const speechTokenRef = useRef(0);
   643→
   644→  const stopSpeech = () => {
   645→    speechTokenRef.current += 1;
   646→    if (audioRef.current) {
   647→      audioRef.current.pause();
   648→      audioRef.current.removeAttribute("src");
   649→    }
   650→    if (audioUrlRef.current) {
   651→      URL.revokeObjectURL(audioUrlRef.current);
   652→      audioUrlRef.current = null;
   653→    }
   654→    setSpeech(null);
   655→  };
   656→
   657→  const toggleSpeech = async (m: UIMessage) => {
   658→    if (speech?.id === m.id) {
   659→      stopSpeech();
   660→      return;
   661→    }
   662→    stopSpeech();
   663→    const token = ++speechTokenRef.current;
   664→    setSpeech({ id: m.id, phase: "loading" });
   665→    try {
   666→      const res = await fetch("/api/voice/speak", {
   667→        method: "POST",
   668→        headers: { "Content-Type": "application/json" },
   669→        body: JSON.stringify({ text: m.content }),
   670→      });
   671→      if (!res.ok) {
   672→        const data = await res.json().catch(() => ({}));
   673→        throw new Error(data.error || "Read-aloud is unavailable right now.");
   674→      }
   675→      if (token !== speechTokenRef.current) return; // user moved on
   676→      const blob = await res.blob();
   677→      const url = URL.createObjectURL(blob);
   678→      audioUrlRef.current = url;
   679→      const audio = (audioRef.current ??= new Audio());
   680→      audio.src = url;
   681→      audio.onended = () => {
   682→        if (token === speechTokenRef.current) stopSpeech();
   683→      };
   684→      audio.onerror = () => {
   685→        if (token === speechTokenRef.current) stopSpeech();
   686→      };
   687→      await audio.play();
   688→      if (token !== speechTokenRef.current) return;
   689→      setSpeech({ id: m.id, phase: "playing" });
   690→    } catch (e) {
   691→      if (token === speechTokenRef.current) {
   692→        setSpeech(null);
   693→        if ((e as Error).name !== "NotAllowedError") {
   694→          toast({
   695→            description: (e as Error).message || "Could not play the audio.",
   696→            variant: "destructive",
   697→          });
   698→        }
   699→      }
   700→    }
   701→  };
   702→
   703→  // stop playback when the timeline unmounts or the chat changes
   704→  useEffect(() => {
   705→    return () => {
   706→      speechTokenRef.current += 1;
   707→      audioRef.current?.pause();
   708→      if (audioUrlRef.current) URL.revokeObjectURL(audioUrlRef.current);
   709→    };
   710→  }, []);
   711→
   712→  // Scroll for NEW messages only — content growth is followed
   713→  // word-by-word by the typewriter gate — and never yank the view
   714→  // back down if the user has scrolled up.
   715→  const count = messages.length;
   716→  const lastId = messages[count - 1]?.id ?? "";
   717→  useEffect(() => {
   718→    if (!stickRef.current) return;
   719→    bottomRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
   720→  }, [count, lastId, stickRef]);
   721→
   722→  return (
   723→    <div ref={scrollRef} className="senku-scroll flex-1 overflow-y-auto">
   724→      <div className="mx-auto w-full max-w-3xl space-y-6 px-4 pb-6 pt-4">
   725→        {messages.map((m, i) =>
   726→          m.role === "user" ? (
   727→            <UserMessage key={m.id} message={m} />
   728→          ) : (
   729→            /* Slot-stable key: the server reconciles the assistant
   730→               message id at "done" — keying by id would remount the
   731→               message right then and dump the typewriter's remaining
   732→               reveal. The slot never shifts mid-turn (messages are
   733→               append-only while streaming). */
   734→            <AssistantMessage
   735→              key={`slot-${i}`}
   736→              message={m}
   737→              speech={speech}
   738→              onToggleSpeech={(msg) => void toggleSpeech(msg)}
   739→            />
   740→          )
   741→        )}
   742→        <div ref={bottomRef} />
   743→      </div>
   744→    </div>
   745→  );
   746→}
