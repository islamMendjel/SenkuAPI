    44→function CopyButton({ text }: { text: string }) {

    48→      onClick={async () => {

    70→function SourcesBlock({ sources }: { sources: NonNullable<UIMessage["sources"]> }) {
    71→  // >3 results can only come from the deep-research pipeline
    72→  // (medium search is capped at 3) — say so, and scroll.
    73→  const deep = sources.length > 3;
    74→  return (
    75→    <div className="mt-3 rounded-xl border border-zinc-800 bg-[#1d1d20] p-3">
    76→      <div className="mb-2 flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-zinc-500">
    77→        <RefreshCcw className="h-3 w-3" />
    78→        {deep
    79→          ? `Deep research · ${sources.length} sources`
    80→          : `Searched the web · ${sources.length} result${
    81→              sources.length === 1 ? "" : "s"
    82→            }`}
    83→      </div>
    84→      <div
    85→        className={`space-y-2.5${deep ? " senku-scroll max-h-96 overflow-y-auto pr-1" : ""}`}
    86→      >
    87→        {sources.map((s, i) => {
    88→          let host = s.url;
    89→          try {
    90→            host = new URL(s.url).hostname.replace(/^www\./, "");
    91→          } catch {
    92→            /* keep raw */
    93→          }
    94→          return (
    95→            <div key={i} className="min-w-0">
    96→              <a
    97→                href={s.url}
    98→                target="_blank"
    99→                rel="noopener noreferrer"
   100→                className="block truncate text-[13px] font-medium text-zinc-200 hover:text-violet-300 hover:underline"
   101→              >
   102→                {s.title || host}
   103→              </a>
   104→              <p className="text-[11px] text-zinc-500">
   105→                {host}
   106→                {s.dateHint ? ` · dated: ${s.dateHint}` : ""}
   107→              </p>
   108→              {s.snippet && (
   109→                <p className="mt-0.5 line-clamp-2 text-[12px] leading-snug text-zinc-400">
   110→                  {s.snippet}
   111→                </p>
   112→              )}
   113→            </div>
   114→          );
   115→        })}
   116→      </div>
   117→    </div>
   118→  );
   119→}

   234→/** One quiet provenance line: "GLM 5.3 · APInex pool · 1.9k tokens · cache: hit". */
   235→function provenanceLine(m: UIMessage): string {
   236→  const parts: string[] = [];
   237→  // The effort dial leaves its mark — fast/deep turns say so.
   238→  if (m.effort === "fast") parts.push("Fast mode");
   239→  if (m.effort === "deep") parts.push("Deep research");
   240→  if (m.model) parts.push(modelMeta(m.model).name);
   241→  if (m.provider) {
   242→    parts.push(
   243→      m.servedBy === "zai-cloud"
   244→        ? `${m.provider} · cloud failover`
   245→        : `${m.provider} pool`
   246→    );
   247→  }
   248→  if (typeof m.tokens === "number" && m.tokens > 0) {
   249→    parts.push(`${formatCompact(m.tokens)} tokens`);
   250→  }
   251→  parts.push(m.cached ? "cache: hit" : "cache: miss");
   252→  return parts.join(" · ");
   253→}

   255→/** Body without the trailing label line (display only). */
   256→function withoutConfidenceLine(content: string): string {
   257→  if (!confidenceOf(content)) return content;
   258→  const lines = content.trimEnd().split("\n");
   259→  return lines.slice(0, -1).join("\n").trimEnd();
   260→}

   334→            onClick={onKeep}

   379→/** A duel turn awaiting its pick: two models racing side by side
   380→    (stacked on mobile), each with its own Keep button. */
   381→function DuelMessage({ message }: { message: UIMessage }) {
   382→  const { pickDuelWinner } = useSenku();
   383→  const { toast } = useToast();
   384→  const duel = message.duel;
   385→  if (!duel) return null;
   386→  // Picks unlock once the turn finishes (or is stopped) — picking
   387→  // mid-stream would race the lane_done/done reconciliation.
   388→  const canPick = !message.streaming;
   389→
   390→  const keep = (lane: "a" | "b") => {
   391→    const meta = modelMeta(duel[lane].model);
   392→    void pickDuelWinner(message.id, lane).then(() => {
   393→      toast({
   394→        description: `Kept ${meta.name} — the runner-up stays under the answer.`,
   395→      });
   396→    });
   397→  };
   398→
   399→  return (
   400→    <div>
   401→      <div className="mb-2 flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-amber-300/70">
   402→        <Swords className="h-3 w-3" aria-hidden />
   403→        Duel — pick the keeper
   404→      </div>
   405→      <div className="grid grid-cols-1 items-start gap-3 md:grid-cols-2">
   406→        <DuelLaneCard lane={duel.a} canPick={canPick} onKeep={() => keep("a")} />
   407→        <DuelLaneCard lane={duel.b} canPick={canPick} onKeep={() => keep("b")} />
   408→      </div>
   409→      <p className="mt-2 text-[11px] text-zinc-600">
   410→        Both answers are billed separately ·{" "}
   411→        {canPick
   412→          ? "pick one to keep it in the chat"
   413→          : "generating both answers…"}
   414→      </p>
   415→    </div>
   416→  );
   417→}

   419→function AssistantMessage({
   420→  message,
   421→  speech,
   422→  onToggleSpeech,
   423→}: {
   424→  message: UIMessage;
   425→  speech: SpeechState;
   426→  onToggleSpeech: (m: UIMessage) => void;
   427→}) {
   428→  const { models, model, setModel, retryLast, streaming } = useSenku();
   429→  const isThinking = message.streaming && !message.content && !message.error;
   430→  const isSpeakingThis = speech?.id === message.id;
   431→  const speechLoading = isSpeakingThis && speech.phase === "loading";
   432→  const speechPlaying = isSpeakingThis && speech.phase === "playing";
   433→
   434→  // Error recovery: the next model in the list (wraps around).
   435→  const currentIdx = models.indexOf(model);
   436→  const switchTarget =
   437→    models.length > 1
   438→      ? models[((currentIdx === -1 ? 0 : currentIdx) + 1) % models.length]
   439→      : null;
   440→
   441→  // Grounded-answer contract: the trailing confidence label becomes
   442→  // a badge once streaming completes (while streaming, the final
   443→  // line may still be half-written — render it raw).
   444→  const confidence = message.error ? null : confidenceOf(message.content);
   445→  const displayContent =
   446→    !message.streaming && confidence
   447→      ? withoutConfidenceLine(message.content)
   448→      : message.content;
   449→
   450→  return (
   451→    <div className="group senku-fade-up">
   452→      <div className="flex gap-3">
   453→        {/* While thinking, our own WHITE sparkle logo pulses with the
   454→            exact scale curve measured from the Claude Code reference
   455→            GIF — same motion, our colors — beside a shimmering label. */}
   456→        <Image
   457→          src="/senku-sparkle.png"
   458→          alt="SenkuAPI"
   459→          width={28}
   460→          height={28}
   461→          className={`mt-0.5 h-7 w-7 shrink-0${
   462→            isThinking ? " senku-thinking-sparkle" : ""
   463→          }`}
   464→        />
   465→        <div className="min-w-0 flex-1">
   466→          {message.error ? (
   467→            <div
   468→              className="rounded-xl border border-red-500/30 bg-red-500/10 px-3.5 py-2.5 text-[13px] text-red-300"
   469→              role="alert"
   470→            >
   471→              <p className="leading-relaxed">{message.error}</p>
   472→              <div className="mt-2.5 flex flex-wrap items-center gap-2">
   473→                <button
   474→                  onClick={() => void retryLast()}
   475→                  disabled={streaming}
   476→                  className="flex h-7 items-center gap-1.5 rounded-full border border-zinc-500/60 bg-white/10 px-3 text-[12px] font-medium text-zinc-100 transition-colors hover:bg-white/20 disabled:opacity-40"
   477→                >
   478→                  <RefreshCcw className="h-3 w-3" />
   479→                  Retry
   480→                </button>
   481→                {switchTarget && (
   482→                  <button
   483→                    onClick={() => {
   484→                      setModel(switchTarget);
   485→                      void retryLast(switchTarget);
   486→                    }}
   487→                    disabled={streaming}
   488→                    className="flex h-7 items-center gap-1.5 rounded-full border border-zinc-700 px-3 text-[12px] text-zinc-400 transition-colors hover:bg-white/5 hover:text-zinc-200 disabled:opacity-40"
   489→                    title={`Retry with ${modelMeta(switchTarget).name}`}
   490→                  >
   491→                    <ArrowLeftRight className="h-3 w-3" />
   492→                    Switch model
   493→                  </button>
   494→                )}
   495→              </div>
   496→            </div>
   497→          ) : message.duel ? (
   498→            /* Awaiting-pick duel: two racing lanes instead of one
   499→               answer — per-lane thinking, streaming, badges, Keep. */
   500→            <DuelMessage message={message} />
   501→          ) : isThinking ? (
   502→            <div className="flex h-7 items-center" aria-live="polite">
   503→              <span className="senku-thinking-label text-[14px] font-medium">
   504→                {message.phase === "searching"
   505→                  ? message.searchDetail ?? "Searching the web"
   506→                  : message.effort === "deep"
   507→                    ? "Thinking deeply"
   508→                    : "Thinking"}
   509→              </span>
   510→            </div>
   511→          ) : (
   512→            <div className="senku-md">
   513→              <Markdown content={displayContent} />
   514→              {message.streaming && (
   515→                <span className="senku-caret text-zinc-300">▍</span>
   516→              )}
   517→            </div>
   518→          )}
   519→
   520→          {/* grounded-answer confidence badge (search turns) */}
   521→          {!message.streaming && !message.duel && confidence && (
   522→            <ConfidenceBadge label={confidence} />
   523→          )}
   524→
   525→          {/* search-failure notice (all engines down) */}
   526→          {!message.streaming && message.searchFailed && !message.error && (
   527→            <div className="mt-2 flex items-center gap-1.5 text-[11.5px] text-amber-400/80">
   528→              <TriangleAlert className="h-3.5 w-3.5 shrink-0" />
   529→              Web search unavailable — answered from model knowledge.
   530→            </div>
   531→          )}
   532→
   533→          {/* quiet provenance line + hover actions (desktop reveals on
   534→              hover; touch devices keep it faintly visible) */}
   535→          {!message.streaming && !message.error && !message.duel && (

   580→          )}

   582→          {/* duel runner-up: the losing lane stays viewable under the

   597→                <Markdown

   603→                />
   604→              </div>
   605→            </details>
   606→          )}
   607→        </div>
   608→      </div>
   609→    </div>
   610→  );

   627→export function ChatTimeline() {
   628→  const { messages } = useSenku();
   629→  const { toast } = useToast();
   630→  const bottomRef = useRef<HTMLDivElement>(null);
   631→
   632→  // ── Read-aloud (TTS) playback state ──
   633→  // One audio element for the whole timeline; only one message can
   634→  // speak at a time. A token guards against stale fetches after a stop.
   635→  const [speech, setSpeech] = useState<SpeechState>(null);
   636→  const audioRef = useRef<HTMLAudioElement | null>(null);
   637→  const audioUrlRef = useRef<string | null>(null);
   638→  const speechTokenRef = useRef(0);
   639→
   640→  const stopSpeech = () => {
   641→    speechTokenRef.current += 1;
   642→    if (audioRef.current) {
   643→      audioRef.current.pause();
   644→      audioRef.current.removeAttribute("src");
   645→    }
   646→    if (audioUrlRef.current) {
   647→      URL.revokeObjectURL(audioUrlRef.current);
   648→      audioUrlRef.current = null;
   649→    }
   650→    setSpeech(null);
   651→  };
   652→
   653→  const toggleSpeech = async (m: UIMessage) => {
   654→    if (speech?.id === m.id) {
   655→      stopSpeech();
   656→      return;
   657→    }
   658→    stopSpeech();
   659→    const token = ++speechTokenRef.current;
   660→    setSpeech({ id: m.id, phase: "loading" });
   661→    try {
   662→      const res = await fetch("/api/voice/speak", {
   663→        method: "POST",
   664→        headers: { "Content-Type": "application/json" },
   665→        body: JSON.stringify({ text: m.content }),
   666→      });
   667→      if (!res.ok) {
   668→        const data = await res.json().catch(() => ({}));
   669→        throw new Error(data.error || "Read-aloud is unavailable right now.");
   670→      }
   671→      if (token !== speechTokenRef.current) return; // user moved on
   672→      const blob = await res.blob();
   673→      const url = URL.createObjectURL(blob);
   674→      audioUrlRef.current = url;
   675→      const audio = (audioRef.current ??= new Audio());
   676→      audio.src = url;
   677→      audio.onended = () => {
   678→        if (token === speechTokenRef.current) stopSpeech();
   679→      };
   680→      audio.onerror = () => {
   681→        if (token === speechTokenRef.current) stopSpeech();
   682→      };
   683→      await audio.play();
   684→      if (token !== speechTokenRef.current) return;
   685→      setSpeech({ id: m.id, phase: "playing" });
   686→    } catch (e) {
   687→      if (token === speechTokenRef.current) {
   688→        setSpeech(null);
   689→        if ((e as Error).name !== "NotAllowedError") {
   690→          toast({
   691→            description: (e as Error).message || "Could not play the audio.",
   692→            variant: "destructive",
   693→          });
   694→        }
   695→      }
   696→    }
   697→  };
   698→
   699→  // stop playback when the timeline unmounts or the chat changes
   700→  useEffect(() => {
   701→    return () => {
   702→      speechTokenRef.current += 1;
   703→      audioRef.current?.pause();
   704→      if (audioUrlRef.current) URL.revokeObjectURL(audioUrlRef.current);
   705→    };
   706→  }, []);
   707→
   708→  useEffect(() => {
   709→    bottomRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
   710→  }, [messages]);
   711→
   712→  return (
   713→    <div className="senku-scroll flex-1 overflow-y-auto">
   714→      <div className="mx-auto w-full max-w-3xl space-y-6 px-4 pb-6 pt-4">
   715→        {messages.map((m) =>
   716→          m.role === "user" ? (
   717→            <UserMessage key={m.id} message={m} />
   718→          ) : (
   719→            <AssistantMessage
   720→              key={m.id}
   721→              message={m}
   722→              speech={speech}
   723→              onToggleSpeech={(msg) => void toggleSpeech(msg)}
   724→            />
   725→          )
   726→        )}
   727→        <div ref={bottomRef} />
   728→      </div>
   729→    </div>
   730→  );
   731→}
