    35→    35→function cooldownLeft(coolingUntil: string | null | undefined, now: number) {
    36→    36→  if (!coolingUntil) return null;
    37→    37→  const ms = new Date(coolingUntil).getTime() - now;
    38→    38→  return ms > 0 ? ms : null;
    39→    39→}

    41→    41→function fmtCountdown(ms: number) {
    42→    42→  const s = Math.ceil(ms / 1000);
    43→    43→  if (s >= 3600) return `${Math.floor(s / 3600)}h${Math.floor((s % 3600) / 60)}m`;
    44→    44→  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, "0")}`;
    45→    45→}

    47→    47→// Run 41 — THE LIVING POOL: one honest phrase per key, built from
    48→    48→// the prober's own records. Spoken by tooltips and screen readers
    49→    49→// alike — the state, the evidence (latency / error), and the
    50→    50→// per-boot serve ledger.
    51→    51→function livePhrase(k: KeyView): string {
    52→    52→  const live = k.live;
    53→    53→  if (!live) return "";
    54→    54→  const s =
    55→    55→    live.state === "serving"
    56→    56→      ? `probe-verified live${live.lastLatencyMs != null ? ` · ${live.lastLatencyMs}ms probe` : ""}`
    57→    57→      : live.state === "checkin"
    58→    58→        ? "needs its daily dashboard check-in (402) — it rejoins the pool automatically once checked in"
    59→    59→        : live.state === "cooling"
    60→    60→          ? live.lastProbeStatus === 401 || live.lastProbeStatus === 403
    61→    61→            ? "credential rejected (suspended or rotated) — replace the key; auto re-probed every 10 min"
    62→    62→            : "breaker cooldown"
    63→    63→          : live.state === "down"
    64→    64→            ? `not responding${live.lastError ? ` (${live.lastError})` : ""}`
    65→    65→            : "awaiting first probe";
    66→    66→  const serves =
    67→    67→    live.serves > 0
    68→    68→      ? ` · ${live.serves} serve${live.serves === 1 ? "" : "s"} this boot`
    69→    69→      : "";
    70→    70→  return ` · ${s}${serves}`;
    71→    71→}

   501→   501→function KeyRow({
   502→   502→  k,
   503→   503→  now,
   504→   504→  onDelete,
   505→   505→  onToggle,
   506→   506→  onProbe,
   507→   507→}: {
   508→   508→  k: KeyView;
   509→   509→  now: number;
   510→   510→  onDelete: (id: string) => Promise<void>;
   511→   511→  onToggle: (id: string, isActive: boolean) => Promise<void>;
   512→   512→  onProbe: (id: string) => Promise<ProbeVerdict>;
   513→   513→}) {
   514→   514→  const { toast } = useToast();
   515→   515→  const [probing, setProbing] = useState(false);
   516→   516→  const cooling = cooldownLeft(k.coolingUntil, now);
   517→   517→  const blocked = k.blockedTier ?? null;
   518→   518→  const live = k.live ?? null;
   519→   519→  // Run 41 — the row speaks the key's live truth, with a tone per
   520→   520→  // verdict: amber = an owner ACTION (daily check-in), rose =
   521→   521→  // cooling / not responding, emerald = probe-verified serving.
   522→   522→  let state: string | null = null;
   523→   523→  let stateCls = "";
   524→   524→  if (!k.isActive) {
   525→   525→    state = "off";
   526→   526→  } else if (live?.state === "checkin") {
   527→   527→    state = "needs daily check-in — self-heals once checked in";
   528→   528→    stateCls = " text-amber-300/90";
   529→   529→  } else if (cooling !== null) {
   530→   530→    state =
   531→   531→      live?.lastProbeStatus === 401 || live?.lastProbeStatus === 403
   532→   532→        ? "credential rejected — replace the key"
   533→   533→        : `cooling ${fmtCountdown(cooling)}`;
   534→   534→    stateCls = " text-rose-300/80";
   535→   535→  } else if (live?.state === "down") {
   536→   536→    state = "not responding";
   537→   537→    stateCls = " text-rose-300/80";
   538→   538→  } else if (blocked) {
   539→   539→    state = TIER_TAG[blocked];
   540→   540→    stateCls = " text-amber-200/70";
   541→   541→  } else if (live?.state === "serving") {
   542→   542→    state = `serving${
   543→   543→      live.lastLatencyMs != null ? ` · ${live.lastLatencyMs}ms probe` : ""
   544→   544→    }${
   545→   545→      live.serves > 0
   546→   546→        ? ` · ${live.serves} serve${live.serves === 1 ? "" : "s"} this boot`
   547→   547→        : ""
   548→   548→    }`;
   549→   549→    stateCls = " text-emerald-300/80";
   550→   550→  } else if (live?.state === "unproven") {
   551→   551→    state = "probing…";
   552→   552→    stateCls = " text-zinc-500";
   553→   553→  }
   554→   554→
   555→   555→  // Run 42 — the probe speaks its verdict out loud: live (with the
   556→   556→  // latency it just measured), still check-in-blocked, credential
   557→   557→  // rejected, or whatever the upstream actually said. No silent
   558→   558→  // failures — every click lands somewhere the owner can act on.
   559→   559→  const handleProbe = async () => {
   560→   560→    setProbing(true);
   561→   561→    try {
   562→   562→      const r = await onProbe(k.id);
   563→   563→      if (r.ok) {
   564→   564→        toast({
   565→   565→          description: `${k.label} verified live · ${r.latencyMs}ms probe`,
   566→   566→        });
   567→   567→      } else if (r.status === 402) {
   568→   568→        toast({
   569→   569→          description:
   570→   570→            "Still demands its daily check-in (402). Complete it on the dashboard, then probe again.",
   571→   571→          variant: "destructive",
   572→   572→        });
   573→   573→      } else if (r.status === 401 || r.status === 403) {
   574→   574→        toast({
   575→   575→          description: "Credential rejected (401/403) — replace the key.",
   576→   576→          variant: "destructive",
   577→   577→        });
   578→   578→      } else {
   579→   579→        toast({
   580→   580→          description: `Probe failed · ${r.error || `status ${r.status}`}`,
   581→   581→          variant: "destructive",
   582→   582→        });
   583→   583→      }
   584→   584→    } catch (e) {
   585→   585→      toast({ description: (e as Error).message, variant: "destructive" });
   586→   586→    } finally {
   587→   587→      setProbing(false);
   588→   588→    }
   589→   589→  };
   590→   590→  return (
   591→   591→    <div className="flex items-center gap-2.5 rounded-lg border border-zinc-800/80 bg-[#18181b] px-2.5 py-2">
   592→   592→      <div className="min-w-0 flex-1">
   593→   593→        <p className="truncate font-mono text-[12px] text-zinc-300">
   594→   594→          {k.masked}
   595→   595→        </p>
   596→   596→        <p className="text-[10.5px] text-zinc-600">
   597→   597→          {k.label} · {formatFull(k.tokensUsedThisMonth)} tokens this month
   598→   598→          {k.lastUsedAt
   599→   599→            ? ` · used ${new Date(k.lastUsedAt).toLocaleDateString()}`
   600→   600→            : " · never used"}
   601→   601→          {state && (
   602→   602→            <span className={stateCls}> · {state}</span>
   603→   603→          )}
   604→   604→        </p>
   605→   605→      </div>
   606→   606→      {/* Run 42 — self-service healing: one liveness probe NOW.
   607→   607→          After a dashboard check-in or a key swap, the owner clicks
   608→   608→          and watches the seat flip emerald — no waiting for the
   609→   609→          prober's tick or the breaker's cooldown. Amber-ring when
   610→   610→          the key awaits its check-in: that's the "I checked in —
   611→   611→          re-verify now" button. */}
   612→   612→      <button
   613→   613→        onClick={() => void handleProbe()}
   614→   614→        disabled={probing || !k.isActive}
   615→   615→        aria-label={`Probe ${k.label} now`}
   616→   616→        title="Probe now — verify this key is live instantly"
   617→   617→        className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-md transition-colors disabled:opacity-40 ${
   618→   618→          k.live?.state === "checkin"
   619→   619→            ? "border border-amber-400/40 text-amber-300 hover:bg-amber-400/10"
   620→   620→            : "text-zinc-500 hover:bg-white/5 hover:text-zinc-300"
   621→   621→        }`}
   622→   622→      >
   623→   623→        {probing ? (
   624→   624→          <Loader2 className="h-3.5 w-3.5 animate-spin" />
   625→   625→        ) : (
   626→   626→          <Zap className="h-3.5 w-3.5" />
   627→   627→        )}
   628→   628→      </button>
   629→   629→      <Switch
   630→   630→        checked={k.isActive}
   631→   631→        onCheckedChange={(v) => void onToggle(k.id, v)}
   632→   632→        aria-label={`Toggle key ${k.label}`}
   633→   633→      />
   634→   634→      <button
   635→   635→        onClick={() => void onDelete(k.id)}
   636→   636→        aria-label={`Delete key ${k.label}`}
   637→   637→        className="flex h-7 w-7 items-center justify-center rounded-md text-zinc-500 transition-colors hover:bg-red-400/10 hover:text-red-400"
   638→   638→      >
   639→   639→        <Trash2 className="h-3.5 w-3.5" />
   640→   640→      </button>
   641→   641→    </div>
   642→   642→  );
   643→   643→}

   647→   647→export function ProvidersDialog({
   648→   648→  open,
   649→   649→  onOpenChange,
   650→   650→}: {
   651→   651→  open: boolean;
   652→   652→  onOpenChange: (v: boolean) => void;
   653→   653→}) {
   654→   654→  const { providers, switchProvider, refreshUsage } = useHermes();
   655→   655→  // Live ring: while the dialog is open, poll the store's own
   656→   656→  // /api/usage refresh so the orrery advances in real time as the
   657→   657→  // gateway serves turns. User mutations (switch/add/toggle/delete)
   658→   658→  // already write through the same store — one source of truth,
   659→   659→  // zero staleness windows.
   660→   660→  const [now, setNow] = useState(() => Date.now());
   661→   661→
   662→   662→  useEffect(() => {
   663→   663→    if (!open) return;
   664→   664→    let alive = true;
   665→   665→    const tick = () => setNow(Date.now());
   666→   666→    const poll = () => {
   667→   667→      if (alive) void refreshUsage();
   668→   668→    };
   669→   669→    poll();
   670→   670→    const poller = setInterval(poll, 4000);
   671→   671→    const ticker = setInterval(tick, 1000);
   672→   672→    return () => {
   673→   673→      alive = false;
   674→   674→      clearInterval(poller);
   675→   675→      clearInterval(ticker);
   676→   676→    };
   677→   677→  }, [open, refreshUsage]);
   678→   678→
   679→   679→  return (
   680→   680→    <Dialog open={open} onOpenChange={onOpenChange}>
   681→   681→      <DialogContent className="max-h-[85vh] overflow-y-auto rounded-2xl border-zinc-800 bg-[#212121] p-5 text-zinc-200 sm:max-w-lg">
   682→   682→        <DialogHeader>
   683→   683→          <DialogTitle className="text-[15px] font-semibold text-white">
   684→   684→            Manage Providers
   685→   685→          </DialogTitle>
   686→   686→          <DialogDescription className="text-[12.5px] leading-snug text-zinc-500">
   687→   687→            Exactly one provider is active at a time. Switching activates the
   688→   688→            selected provider and deactivates all others — rotation then runs
   689→   689→            strictly on its keys, and spend-balanced drift repair keeps every
   690→   690→            monthly pool even.
   691→   691→          </DialogDescription>
   692→   692→        </DialogHeader>
   693→   693→        <div className="min-w-0 space-y-3 pt-1">
   694→   694→          {/* Run 35 — the ACTIVE provider leads: its live ring is
   695→   695→              the first thing the eye lands on when the dialog opens. */}
   696→   696→          {[...providers]
   697→   697→            .sort((a, b) => Number(b.isActive) - Number(a.isActive))
   698→   698→            .map((p) => (
   699→   699→              <ProviderCard
   700→   700→                key={p.id}
   701→   701→                provider={p}
   702→   702→                onSwitch={switchProvider}
   703→   703→                now={now}
   704→   704→              />
   705→   705→            ))}
   706→   706→        </div>
   707→   707→      </DialogContent>
   708→   708→    </Dialog>
   709→   709→  );
   710→   710→}
