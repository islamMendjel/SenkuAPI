     2→// One chat turn: seed → active provider → web search (optional) →
     3→// exact-match cache → health-aware key dispatch (pool retry) →
     4→// cloud completion → usage accounting → cache fill → persistence.

    35→export interface TurnEvent {
    36→  type:
    37→    | "meta"
    38→    | "search_status"
    39→    | "sources"
    40→    | "delta"
    41→    | "restart"
    42→    | "duel_start"
    43→    | "lane_done"
    44→    | "done"
    45→    | "error";
    46→  [k: string]: unknown;
    47→}
    48→
    49→export interface TurnParams {
    50→  chatId?: string | null;
    51→  history: ChatMessageData[]; // prior turns (already persisted)
    52→  content: string; // new user message
    53→  model: string;
    54→  search: boolean;
    55→  temporary: boolean; // temporary chats are never persisted
    56→  persistUser?: boolean; // false when regenerating after an edit (message already updated)
    57→  /** Effort dial. "medium" (default) = byte-identical legacy
    58→   *  behavior; "fast" routes to the fastest pooled engine with a
    59→   *  brevity directive; "deep" routes to the strongest pooled
    60→   *  engine with thinking + multi-round audited retrieval. */
    61→  effort?: EffortLevel;
    62→  /** Files the gateway ALREADY read (extraction happened at pick
    63→   *  time). Their text is appended to the request context — never
    64→   *  to the stored message content — and persisted on the user
    65→   *  message row so follow-up turns and reloads keep the context. */
    66→  attachments?: AttachmentRecord[];
    67→  /** Fires when the CLIENT disconnects (Stop button, tab close).
    68→   *  The upstream generation is cancelled at the source, the
    69→   *  partial reply is persisted, and the truncated answer NEVER
    70→   *  enters the response cache. */
    71→  signal?: AbortSignal;
    72→  emit: (ev: TurnEvent) => void;
    73→}
    74→
    75→export interface TurnResult {
    76→  chatId: string | null;
    77→  userMessageId: string | null;
    78→  assistantMessageId: string | null;
    79→  title: string | null;
    80→  sources: SearchSource[];
    81→  content: string;
    82→  cached: boolean;
    83→  tokens: number; // effective (multiplied) tokens billed — 0 when cached
    84→  rawTokens: number;
    85→  servedBy: "provider" | "zai-cloud" | "cache";
    86→  keyLabel: string | null;
    87→  providerName: string;
    88→  model: string;
    89→  /** which effort level served this turn */
    90→  effort: EffortLevel;
    91→  /** true when search was requested but every engine failed */
    92→  searchFailed: boolean;
    93→  /** true when the client disconnected mid-generation — content is
    94→   *  the partial text that actually streamed */
    95→  aborted?: boolean;
    96→}
    97→
    98→function deriveTitle(content: string): string {
    99→  const clean = content
   100→    .replace(/\s+/g, " ")
   101→    .replace(/[^\p{L}\p{N} ?!.,'’-]/gu, "")
   102→    .trim();
   103→  const words = clean.split(" ").filter(Boolean).slice(0, 6).join(" ");
   104→  return (words || "New chat").slice(0, 60);
   105→}
   106→
   107→function monthKey(d = new Date()): string {
   108→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   109→}
   110→
   111→/** History rows keep their attachment extracts — a follow-up
   112→ *  question ("what was the revenue figure?") must still see the
   113→ *  file from the earlier turn, exactly like the incumbents. */
   114→function expandHistory(history: ChatMessageData[]): ChatMessageData[] {
   115→  return history.map((m) =>
   116→    m.role === "user" && m.attachments?.length
   117→      ? { ...m, content: m.content + attachmentContextBlock(m.attachments) }
   118→      : m
   119→  );
   120→}
   121→
   122→export async function runChatTurn(params: TurnParams): Promise<TurnResult> {
   123→  const { emit } = params;
   124→  const provider = await getActiveProvider();
   125→  const providerModels = JSON.parse(provider.models || "[]") as string[];
   126→  const requestedModel =
   127→    params.model || providerModels[0] || "free/glm-5.3";
   128→  const effort: EffortLevel = params.effort ?? "medium";
   129→  // The effort dial routes across the POOL: fast → the fastest
   130→  // engine the provider offers, deep → the strongest. Medium keeps
   131→  // the user's selection untouched (legacy behavior).
   132→  const model = effortModel(providerModels, requestedModel, effort);
   133→  // Untrusted client payload — rebuilt field-by-field server-side.
   134→  const attachments = sanitizeAttachments(params.attachments);
   135→
   136→  // ── 1. Web search ──
   137→  // Medium/fast: strict discipline (≤3 snippets, ≤300 chars), one
   138→  // pass. Deep: multi-round iterative retrieval — plan queries,
   139→  // search the question verbatim, then the planned angles in
   140→  // parallel, merged URL-deduped (≤8 sources), with live stage
   141→  // labels streamed to the client.
   142→  let sources: SearchSource[] = [];
   143→  let searchEngine: string | undefined;
   144→  if (params.search) {
   145→    if (effort === "deep") {
   146→      emit({ type: "search_status", status: "searching", depth: "deep" });
   147→      const outcome = await deepSearchDetailed(params.content, (detail) =>
   148→        emit({ type: "search_status", status: "searching", detail, depth: "deep" })
   149→      );
   150→      sources = outcome.results;
   151→      searchEngine = "deep";
   152→    } else {
   153→      emit({ type: "search_status", status: "searching" });
   154→      const outcome = await webSearchDetailed(params.content);
   155→      sources = outcome.results;
   156→      searchEngine = outcome.source;
   157→    }
   158→    emit({ type: "sources", sources });
   159→    emit({
   160→      type: "search_status",
   161→      status: sources.length ? "ok" : "failed",
   162→      engine: searchEngine,
   163→      depth: effort === "deep" ? "deep" : undefined,
   164→    });
   165→  }
   166→
   167→  // Effort directives append AFTER the base contract. Medium adds
   168→  // NOTHING — the prompt stays byte-identical to pre-dial turns so
   169→  // the 24h response cache still hits. Fast/deep additions also
   170→  // namespace the cache key per effort level for free.
   171→  const effortAddon =
   172→    effort === "fast"
   173→      ? FAST_SYSTEM_ADDON
   174→      : effort === "deep"
   175→        ? sources.length
   176→          ? DEEP_RESEARCH_ADDON
   177→          : DEEP_THINKING_ADDON
   178→        : "";
   179→  const baseSystemPrompt =
   180→    (params.search && sources.length ? SEARCH_SYSTEM_DIRECTIVE : BASE_SYSTEM_PROMPT) +
   181→    (effortAddon ? `\n\n${effortAddon}` : "");
   182→  // The model always knows the real "now" (user's timezone).
   183→  const systemPrompt = `${baseSystemPrompt}\n\n${liveContextBlock()}`;
   184→
   185→  const finalUserContent =
   186→    params.content +
   187→    attachmentContextBlock(attachments) +
   188→    (sources.length ? searchContextBlock(sources) : "");
   189→
   190→  // ── 2. Exact-match cache (prompt + model + hour-stable system prompt) ──
   191→  // The clock block is hour-stable here so repeat questions within the
   192→  // same hour still hit the cache, while the model sees second-level time.
   193→  const cacheKey = computeCacheKey(
   194→    finalUserContent,
   195→    model,
   196→    `${baseSystemPrompt}\n\n${liveContextBlockHourStable()}`
   197→  );
   198→  const cached = await getCached(cacheKey);
   199→
   200→  // ── 3. Persistence: chat row + user message ──
   201→  let chatId = params.chatId ?? null;
   202→  let userMessageId: string | null = null;
   203→  let title: string | null = null;
   204→
   205→  if (!params.temporary && params.persistUser !== false) {
   206→    if (!chatId) {
   207→      title = deriveTitle(params.content);
   208→      const chat = await db.chat.create({
   209→        data: {
   210→          title,
   211→          model,
   212→          provider: provider.name,
   213→          isTemporary: false,
   214→        },
   215→      });
   216→      chatId = chat.id;
   217→    }
   218→    const userMsg = await db.message.create({
   219→      data: {
   220→        chatId,
   221→        role: "user",
   222→        content: params.content,
   223→        sources: sources.length ? JSON.stringify(sources) : null,
   224→        attachments: attachments.length ? JSON.stringify(attachments) : null,
   225→      },
   226→    });
   227→    userMessageId = userMsg.id;
   228→  } else if (!params.temporary && params.chatId) {
   229→    chatId = params.chatId;
   230→  }
   231→
   232→  emit({
   233→    type: "meta",
   234→    chatId,
   235→    userMessageId,
   236→    title,
   237→    provider: provider.name,
   238→    model,
   239→    effort,
   240→  });
   241→
   242→  // ── 4a. Cache hit → 0 tokens spent ──
   243→  if (cached) {
   244→    // Contract enforcement applies to cached search answers too
   245→    // (older cache entries may predate the confidence-label rule).
   246→    if (sources.length && !confidenceOf(cached.content)) {
   247→      cached.content = await appendConfidenceLabel(cached.content);
   248→    }
   249→    // Cached answers type out live too — the answer never appears
   250→    // in a single shot, even when it cost 0 tokens.
   251→    await revealWords(
   252→      cached.content,
   253→      (text) => emit({ type: "delta", text }),
   254→      6,
   255→      () => params.signal?.aborted === true
   256→    );
   257→    let assistantMessageId: string | null = null;
   258→    if (!params.temporary && chatId) {
   259→      const msg = await db.message.create({
   260→        data: {
   261→          chatId,
   262→          role: "assistant",
   263→          content: cached.content,
   264→          model,
   265→          provider: provider.name,
   266→          keyLabel: "cache",
   267→          tokens: 0,
   268→          cached: true,
   269→        },
   270→      });
   271→      assistantMessageId = msg.id;
   272→      await db.chat.update({
   273→        where: { id: chatId },
   274→        data: { updatedAt: new Date(), model, provider: provider.name },
   275→      });
   276→    }
   277→    // Audit trail: cached hit, zero tokens.
   278→    await db.usageEvent.create({
   279→      data: {
   280→        keyId: null,
   281→        providerId: provider.id,
   282→        chatId,
   283→        model,
   284→        promptTokens: 0,
   285→        completionTokens: 0,
   286→        totalTokens: 0,
   287→        multiplier: provider.tokenMultiplier,
   288→        cached: true,
   289→      },
   290→    });
   291→    const done = {
   292→      type: "done",
   293→      chatId,
   294→      assistantMessageId,
   295→      content: cached.content,
   296→      cached: true,
   297→      tokens: 0,
   298→      rawTokens: 0,
   299→      servedBy: "cache" as const,
   300→      keyLabel: "cache",
   301→      providerName: provider.name,
   302→      model,
   303→      effort,
   304→      sources,
   305→      searchFailed: params.search && sources.length === 0,
   306→    };
   307→    emit(done);
   308→    return done;
   309→  }
   310→
   311→  // ── 4b + 5. Health-aware key dispatch, then cloud completion ──
   312→  // The dispatcher round-robins pooled keys, quarantines dead ones
   313→  // (auth failures), retries the next healthy key within the same
   314→  // turn, and only fails over to the Z.ai cloud when every key
   315→  // failed. Hard pool errors (no keys / limiter exhaustion) keep
   316→  // their original coded messages.
   317→  //
   318→  // `signal` is the CLIENT's connection: the moment it drops, the
   319→  // upstream fetch is aborted at the source — the pooled keys stop
   320→  // paying for tokens nobody will ever see.
   321→  const messages: ChatMessageData[] = [
   322→    { role: "system", content: systemPrompt },
   323→    ...expandHistory(params.history).slice(-20),
   324→    { role: "user", content: finalUserContent },
   325→  ];
   326→
   327→  const routed = await dispatchCompletion({
   328→    provider,
   329→    model,
   330→    messages,
   331→    onDelta: (text) => emit({ type: "delta", text }),
   332→    // Mid-stream provider death → the cloud re-emits from scratch;
   333→    // "restart" tells the client to clear the partial text first.
   334→    onRestart: () => emit({ type: "restart" }),
   335→    // Deep effort → the cloud failover THINKS before answering.
   336→    thinking: effort === "deep",
   337→    signal: params.signal,
   338→  });
   339→  const exec = routed.exec;
   340→  // The key that actually served — null when the Z.ai cloud did.
   341→  const key = routed.key;
   342→  // True when the CLIENT disconnected mid-generation: the partial
   343→  // text is final for this turn.
   344→  const aborted = routed.aborted === true;
   345→
   346→  // ── Grounded-answer contract enforcement ──
   347→  // On search turns the confidence label is mandatory. The directive
   348→  // asks for it, but model compliance is probabilistic — when the
   349→  // answer lacks the label, one cheap corrective pass appends it so
   350→  // the contract holds deterministically (cache fills afterwards,
   351→  // so corrected answers are reused corrected). Aborted turns skip
   352→  // it: no corrective cloud spend on an answer nobody is watching.
   353→  if (!aborted && sources.length && !confidenceOf(exec.content)) {
   354→    exec.content = await appendConfidenceLabel(exec.content);
   355→  }
   356→
   357→  // ── 6. Usage accounting (multiplier applies to effective totals) ──
   358→  // Only the key that ACTUALLY served the response is billed — a
   359→  // dead key that failed over to the cloud never gets charged.
   360→  const rawTotal =
   361→    exec.totalTokens ||
   362→    exec.promptTokens + exec.completionTokens ||
   363→    estimateTokens(finalUserContent) + estimateTokens(exec.content);
   364→
   365→  await db.usageEvent.create({
   366→    data: {
   367→      keyId: key?.id ?? null,
   368→      providerId: provider.id,
   369→      chatId,
   370→      model,
   371→      promptTokens: exec.promptTokens,
   372→      completionTokens: exec.completionTokens,
   373→      totalTokens: rawTotal,
   374→      multiplier: provider.tokenMultiplier,
   375→      cached: false,
   376→    },
   377→  });
   378→
   379→  // Roll the serving key's monthly accumulator forward (monthKey
   380→  // flip safe). Skipped entirely on cloud failover — no key served.
   381→  if (key) {
   382→    const now = new Date();
   383→    const mk = monthKey(now);
   384→    const monthAgg = await db.usageEvent.aggregate({
   385→      _sum: { totalTokens: true },
   386→      where: {
   387→        keyId: key.id,
   388→        createdAt: { gte: new Date(now.getFullYear(), now.getMonth(), 1) },
   389→      },
   390→    });
   391→    await db.apiKey.update({
   392→      where: { id: key.id },
   393→      data: {
   394→        lastUsedAt: now,
   395→        monthKey: mk,
   396→        tokensUsedThisMonth: monthAgg._sum.totalTokens ?? rawTotal,
   397→      },
   398→    });
   399→  }
   400→
   401→  // ── 7. Cache fill (24h TTL) ──
   402→  // A turn the user ABORTED never fills the cache: a truncated
   403→  // answer must never be served to a future identical question.
   404→  if (!aborted) {
   405→    await putCached({
   406→      cacheKey,
   407→      model,
   408→      provider: provider.name,
   409→      content: exec.content,
   410→      tokens: rawTotal,
   411→    });
   412→  }
   413→
   414→  // ── 8. Persist assistant message ──
   415→  // keyLabel is only recorded when a pooled key actually served the
   416→  // response — a cloud-failover answer never claims a key it didn't use.
   417→  const effectiveTokens = Math.round(rawTotal * provider.tokenMultiplier);
   418→  const servedKeyLabel = key?.label ?? null;
   419→  let assistantMessageId: string | null = null;
   420→  // The partial of an aborted turn is persisted too — the timeline
   421→  // keeps exactly what streamed before the user pressed stop (and
   422→  // an empty partial creates no ghost message).
   423→  if (!params.temporary && chatId && exec.content.trim()) {
   424→    const msg = await db.message.create({
   425→      data: {
   426→        chatId,
   427→        role: "assistant",
   428→        content: exec.content,
   429→        model,
   430→        provider: provider.name,
   431→        keyLabel: servedKeyLabel,
   432→        tokens: effectiveTokens,
   433→        cached: false,
   434→        effort,
   435→      },
   436→    });
   437→    assistantMessageId = msg.id;
   438→    await db.chat.update({
   439→      where: { id: chatId },
   440→      data: { updatedAt: new Date(), model, provider: provider.name },
   441→    });
   442→  }
   443→
   444→  const done: TurnResult = {
   445→    chatId,
   446→    userMessageId,
   447→    assistantMessageId,
   448→    title,
   449→    sources,
   450→    content: exec.content,
   451→    cached: false,
   452→    tokens: effectiveTokens,
   453→    rawTokens: rawTotal,
   454→    servedBy: exec.servedBy,
   455→    keyLabel: servedKeyLabel,
   456→    providerName: provider.name,
   457→    model,
   458→    effort,
   459→    searchFailed: params.search && sources.length === 0,
   460→    ...(aborted ? { aborted: true } : {}),
   461→  };
   462→  emit({ type: "done", ...done });
   463→  if (aborted) {
   464→    console.log(
   465→      `[gateway] turn cancelled by client — persisted ${exec.content.length} chars, ` +
   466→        `${effectiveTokens} tokens billed, cache skipped`
   467→    );
   468→  }
   469→  return done;
   470→}

   472→// ── Duel turn ─────────────────────────────────────────────────
   473→// TWO models race in parallel on one question: shared web search,
   474→// shared chat + user-message persistence, per-lane cache lookup,
   475→// health-aware dispatch, grounded-contract enforcement, usage
   476→// accounting and cache fill. Both lanes are billed honestly for
   477→// what they actually served. The WINNER is not persisted here —
   478→// the client calls POST /api/chat/duel/pick once the user chooses;
   479→// that endpoint stores the winner as the assistant message and the
   480→// loser as its runner-up reference.
   481→
   482→export interface DuelLaneResult {
   483→  lane: "a" | "b";
   484→  model: string;
   485→  content: string;
   486→  tokens: number; // effective (multiplied) tokens billed — 0 when cached
   487→  rawTokens: number;
   488→  cached: boolean;
   489→  servedBy: "provider" | "zai-cloud" | "cache" | null; // null on error
   490→  keyLabel: string | null;
   491→  error: string | null;
   492→}
   493→
   494→export interface DuelParams {
   495→  chatId?: string | null;
   496→  history: ChatMessageData[];
   497→  content: string;
   498→  model: string; // lane A — the user's selected model
   499→  challenger: string; // lane B
   500→  search: boolean;
   501→  temporary: boolean;
   502→  persistUser?: boolean;
   503→  /** Effort affects the duel's SHARED retrieval (deep → multi-round
   504→   *  search + thinking on the cloud path) but never the lane models
   505→   *  — a duel is explicitly about the two engines the user chose. */
   506→  effort?: EffortLevel;
   507→  /** Files the gateway read — shared by both lanes (the same
   508→   *  material, the same rules as a normal turn). */
   509→  attachments?: AttachmentRecord[];
   510→  /** Client disconnect cancels BOTH racing lanes at the source. */
   511→  signal?: AbortSignal;
   512→  emit: (ev: TurnEvent) => void;
   513→}
   514→
   515→export interface DuelTurnResult {
   516→  chatId: string | null;
   517→  userMessageId: string | null;
   518→  title: string | null;
   519→  sources: SearchSource[];
   520→  lanes: { a: DuelLaneResult; b: DuelLaneResult };
   521→  bothFailed: boolean;
   522→}
   523→
   524→export async function runDuelTurn(
   525→  params: DuelParams
   526→): Promise<DuelTurnResult> {
   527→  const { emit } = params;
   528→  const provider = await getActiveProvider();
   529→  const providerModels = JSON.parse(provider.models || "[]") as string[];
   530→  const modelA = params.model || providerModels[0] || "free/glm-5.3";
   531→  const modelB = params.challenger;
   532→  const effort: EffortLevel = params.effort ?? "medium";
   533→  const attachments = sanitizeAttachments(params.attachments);
   534→
   535→  if (!modelB || modelB === modelA) {
   536→    throw new Error("Duel needs two different models");
   537→  }
   538→  if (
   539→    providerModels.length &&
   540→    (!providerModels.includes(modelA) || !providerModels.includes(modelB))
   541→  ) {
   542→    throw new Error("Both duel models must be available on the active provider");
   543→  }
   544→
   545→  // ── 1. Shared web search — both lanes reason from the SAME retrieval ──
   546→  let sources: SearchSource[] = [];
   547→  if (params.search) {
   548→    if (effort === "deep") {
   549→      emit({ type: "search_status", status: "searching", depth: "deep" });
   550→      const outcome = await deepSearchDetailed(params.content, (detail) =>
   551→        emit({ type: "search_status", status: "searching", detail, depth: "deep" })
   552→      );
   553→      sources = outcome.results;
   554→    } else {
   555→      emit({ type: "search_status", status: "searching" });
   556→      const outcome = await webSearchDetailed(params.content);
   557→      sources = outcome.results;
   558→    }
   559→    emit({ type: "sources", sources });
   560→    emit({
   561→      type: "search_status",
   562→      status: sources.length ? "ok" : "failed",
   563→      depth: effort === "deep" ? "deep" : undefined,
   564→    });
   565→  }
   566→
   567→  const baseSystemPrompt =
   568→    params.search && sources.length ? SEARCH_SYSTEM_DIRECTIVE : BASE_SYSTEM_PROMPT;
   569→  const systemPrompt = `${baseSystemPrompt}\n\n${liveContextBlock()}`;
   570→  const finalUserContent =
   571→    params.content +
   572→    attachmentContextBlock(attachments) +
   573→    (sources.length ? searchContextBlock(sources) : "");
   574→
   575→  // ── 2. Shared persistence: chat row + user message (same as a
   576→  //      normal turn — the winner is appended later, at pick time) ──
   577→  let chatId = params.chatId ?? null;
   578→  let userMessageId: string | null = null;
   579→  let title: string | null = null;
   580→
   581→  if (!params.temporary && params.persistUser !== false) {
   582→    if (!chatId) {
   583→      title = deriveTitle(params.content);
   584→      const chat = await db.chat.create({
   585→        data: {
   586→          title,
   587→          model: modelA,
   588→          provider: provider.name,
   589→          isTemporary: false,
   590→        },
   591→      });
   592→      chatId = chat.id;
   593→    }
   594→    const userMsg = await db.message.create({
   595→      data: {
   596→        chatId,
   597→        role: "user",
   598→        content: params.content,
   599→        sources: sources.length ? JSON.stringify(sources) : null,
   600→        attachments: attachments.length ? JSON.stringify(attachments) : null,
   601→      },
   602→    });
   603→    userMessageId = userMsg.id;
   604→  } else if (!params.temporary && params.chatId) {
   605→    chatId = params.chatId;
   606→  }
   607→
   608→  emit({
   609→    type: "meta",
   610→    chatId,
   611→    userMessageId,
   612→    title,
   613→    provider: provider.name,
   614→    model: modelA,
   615→    effort,
   616→  });
   617→  emit({ type: "duel_start", a: modelA, b: modelB });
   618→
   619→  const messages: ChatMessageData[] = [
   620→    { role: "system", content: systemPrompt },
   621→    ...expandHistory(params.history).slice(-20),
   622→    { role: "user", content: finalUserContent },
   623→  ];
   624→
   625→  // ── 3. The race: two independent lanes over the shared context ──
   626→  const runLane = async (
   627→    lane: "a" | "b",
   628→    model: string
   629→  ): Promise<DuelLaneResult> => {
   630→    // Per-lane exact-match cache (the cache key includes the model).
   631→    const cacheKey = computeCacheKey(
   632→      finalUserContent,
   633→      model,
   634→      `${baseSystemPrompt}\n\n${liveContextBlockHourStable()}`
   635→    );
   636→    const cached = await getCached(cacheKey);
   637→    if (cached) {
   638→      // Contract enforcement applies to cached search answers too.
   639→      if (sources.length && !confidenceOf(cached.content)) {
   640→        cached.content = await appendConfidenceLabel(cached.content);
   641→      }
   642→      // Cached lane answers type out live too (word by word),
   643→      // exactly like a fresh lane.
   644→      await revealWords(
   645→        cached.content,
   646→        (text) => emit({ type: "delta", lane, text }),
   647→        6,
   648→        () => params.signal?.aborted === true
   649→      );
   650→      await db.usageEvent.create({
   651→        data: {
   652→          keyId: null,
   653→          providerId: provider.id,
   654→          chatId,
   655→          model,
   656→          promptTokens: 0,
   657→          completionTokens: 0,
   658→          totalTokens: 0,
   659→          multiplier: provider.tokenMultiplier,
   660→          cached: true,
   661→        },
   662→      });
   663→      const result: DuelLaneResult = {
   664→        lane,
   665→        model,
   666→        content: cached.content,
   667→        tokens: 0,
   668→        rawTokens: 0,
   669→        cached: true,
   670→        servedBy: "cache",
   671→        keyLabel: "cache",
   672→        error: null,
   673→      };
   674→      emit({ type: "lane_done", ...result });
   675→      return result;
   676→    }
   677→
   678→    try {
   679→      const routed = await dispatchCompletion({
   680→        provider,
   681→        model,
   682→        messages,
   683→        onDelta: (text) => emit({ type: "delta", lane, text }),
   684→        // Mid-stream death in THIS lane → the cloud re-emits this
   685→        // lane from scratch; clear only this lane's partial text.
   686→        onRestart: () => emit({ type: "restart", lane }),
   687→        // Deep effort → both lanes' cloud failover thinks too.
   688→        thinking: effort === "deep",
   689→        // Client disconnect cancels this lane's upstream too.
   690→        signal: params.signal,
   691→      });
   692→      const exec = routed.exec;
   693→      const key = routed.key;
   694→      const aborted = routed.aborted === true;
   695→
   696→      if (!aborted && sources.length && !confidenceOf(exec.content)) {
   697→        exec.content = await appendConfidenceLabel(exec.content);
   698→      }
   699→
   700→      // Per-lane billing — identical rules to a normal turn: only
   701→      // the key that actually served is charged; cloud lanes bill
   702→      // no key.
   703→      const rawTotal =
   704→        exec.totalTokens ||
   705→        exec.promptTokens + exec.completionTokens ||
   706→        estimateTokens(finalUserContent) + estimateTokens(exec.content);
   707→      await db.usageEvent.create({
   708→        data: {
   709→          keyId: key?.id ?? null,
   710→          providerId: provider.id,
   711→          chatId,
   712→          model,
   713→          promptTokens: exec.promptTokens,
   714→          completionTokens: exec.completionTokens,
   715→          totalTokens: rawTotal,
   716→          multiplier: provider.tokenMultiplier,
   717→          cached: false,
   718→        },
   719→      });
   720→      if (key) {
   721→        const now = new Date();
   722→        const mk = monthKey(now);
   723→        const monthAgg = await db.usageEvent.aggregate({
   724→          _sum: { totalTokens: true },
   725→          where: {
   726→            keyId: key.id,
   727→            createdAt: { gte: new Date(now.getFullYear(), now.getMonth(), 1) },
   728→          },
   729→        });
   730→        await db.apiKey.update({
   731→          where: { id: key.id },
   732→          data: {
   733→            lastUsedAt: now,
   734→            monthKey: mk,
   735→            tokensUsedThisMonth: monthAgg._sum.totalTokens ?? rawTotal,
   736→          },
   737→        });
   738→      }
   739→
   740→      // An aborted lane never fills the cache (truncated text must
   741→      // never be served to a future identical question).
   742→      if (!aborted) {
   743→        await putCached({
   744→          cacheKey,
   745→          model,
   746→          provider: provider.name,
   747→          content: exec.content,
   748→          tokens: rawTotal,
   749→        });
   750→      }
   751→
   752→      const result: DuelLaneResult = {
   753→        lane,
   754→        model,
   755→        content: exec.content,
   756→        tokens: Math.round(rawTotal * provider.tokenMultiplier),
   757→        rawTokens: rawTotal,
   758→        cached: false,
   759→        servedBy: exec.servedBy,
   760→        keyLabel: key?.label ?? null,
   761→        error: null,
   762→      };
   763→      emit({ type: "lane_done", ...result });
   764→      return result;
   765→    } catch (e) {
   766→      // One lane failing must never kill the other — the duel
   767→      // degrades to a single-lane answer the user can still keep.
   768→      const result: DuelLaneResult = {
   769→        lane,
   770→        model,
   771→        content: "",
   772→        tokens: 0,
   773→        rawTokens: 0,
   774→        cached: false,
   775→        servedBy: null,
   776→        keyLabel: null,
   777→        error: (e as Error).message || "Generation failed",
   778→      };
   779→      emit({ type: "lane_done", ...result });
   780→      return result;
   781→    }
   782→  };
   783→
   784→  const [a, b] = await Promise.all([
   785→    runLane("a", modelA),
   786→    runLane("b", modelB),
   787→  ]);
   788→
   789→  if (a.error && b.error) {
   790→    emit({
   791→      type: "error",
   792→      message: `Both models failed — A (${modelA}): ${a.error} · B (${modelB}): ${b.error}`,
   793→    });
   794→    return { chatId, userMessageId, title, sources, lanes: { a, b }, bothFailed: true };
   795→  }
   796→
   797→  // No assistant message is persisted here — the client picks the
   798→  // winner and /api/chat/duel/pick stores it (with the runner-up).
   799→  emit({
   800→    type: "done",
   801→    chatId,
   802→    userMessageId,
   803→    assistantMessageId: null,
   804→    duel: { a, b },
   805→    // Top-level fields mirror lane A for simple/legacy consumers.
   806→    content: a.content,
   807→    cached: a.cached,
   808→    tokens: a.tokens,
   809→    rawTokens: a.rawTokens,
   810→    servedBy: a.servedBy,
   811→    keyLabel: a.keyLabel,
   812→    providerName: provider.name,
   813→    model: modelA,
   814→    sources,
   815→    searchFailed: params.search && sources.length === 0,
   816→  });
   817→  return { chatId, userMessageId, title, sources, lanes: { a, b }, bothFailed: false };
   818→}
