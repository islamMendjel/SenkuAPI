   391→  async sendMessage(content) {
   392→    const state = get();
   393→    if (state.streaming || !content.trim()) return;
   394→    // A duel still awaiting its pick? The primary lane becomes the
   395→    // canonical answer so the next turn's history has one voice.
   396→    const pending = [...state.messages].reverse().find((m) => m.duel && !m.duelPicked);
   397→    if (pending) {
   398→      await get().pickDuelWinner(pending.id, "a");
   399→    }
   400→    const s2 = get();
   401→    const history: ChatMessageData[] = s2.messages.map((m) => ({
   402→      role: m.role,
   403→      content: m.content,
   404→    }));
   405→    // Resolve the challenger at send time — the model list may have
   406→    // changed since duel was armed.
   407→    let challenger = s2.challenger;
   408→    if (challenger === s2.model || !s2.models.includes(challenger ?? "")) {
   409→      challenger = autoChallenger(s2.models, s2.model);
   410→    }
   411→    const duel =
   412→      s2.duelEnabled && challenger ? { challenger } : undefined;
   413→    await streamTurn(set, get, {
   414→      chatId: s2.temporary ? null : s2.currentChatId,
   415→      history,
   416→      content,
   417→      persistUser: true,
   418→      duel,
   419→    });
   420→  },

   422→  async editMessage(messageId, content) {
   423→    const state = get();
   424→    if (state.streaming || !content.trim()) return;
   425→    const idx = state.messages.findIndex((m) => m.id === messageId);
   426→    if (idx === -1) return;
   427→    if (state.temporary || !state.currentChatId) {
   428→      // Local-only timeline: truncate everything after the edited message.
   429→      const kept = state.messages
   430→        .slice(0, idx + 1)
   431→        .map((m) => (m.id === messageId ? { ...m, content: content.trim() } : m));
   432→      set({ messages: kept });
   433→      const history: ChatMessageData[] = kept.slice(0, -1).map((m) => ({
   434→        role: m.role,
   435→        content: m.content,
   436→      }));
   437→      await streamTurn(set, get, {
   438→        chatId: null,
   439→        history,
   440→        content: content.trim(),
   441→        persistUser: false,
   442→        temporary: true,
   443→      });
   444→      return;
   445→    }
   446→
   447→    const res = await fetch("/api/chat/truncate", {
   448→      method: "POST",
   449→      headers: { "Content-Type": "application/json" },
   450→      body: JSON.stringify({ messageId, content: content.trim() }),
   451→    });
   452→    const data = await res.json();
   453→    if (!Array.isArray(data.messages)) return;
   454→    const kept: UIMessage[] = data.messages.map((m: UIMessage) => ({
   455→      ...m,
   456→      sources: m.sources ?? null,
   457→    }));
   458→    set({ messages: kept });
   459→    const history: ChatMessageData[] = kept.slice(0, -1).map((m) => ({
   460→      role: m.role,
   461→      content: m.content,
   462→    }));
   463→    await streamTurn(set, get, {
   464→      chatId: state.currentChatId,
   465→      history,
   466→      content: content.trim(),
   467→      persistUser: false,
   468→    });
   469→  },

   471→  async retryLast(modelOverride) {
   472→    const state = get();
   473→    if (state.streaming) return;
   474→
   475→    // The most recent failed assistant reply.
   476→    let failedIdx = -1;
   477→    for (let i = state.messages.length - 1; i >= 0; i--) {
   478→      const m = state.messages[i];
   479→      if (m.role === "assistant" && m.error) {
   480→        failedIdx = i;
   481→        break;
   482→      }
   483→    }
   484→    if (failedIdx === -1) return;
   485→
   486→    // The user message that failed turn was answering.
   487→    let userIdx = -1;
   488→    for (let i = failedIdx - 1; i >= 0; i--) {
   489→      if (state.messages[i].role === "user") {
   490→        userIdx = i;
   491→        break;
   492→      }
   493→    }
   494→    if (userIdx === -1) return;
   495→    const userMsg = state.messages[userIdx];
   496→
   497→    const history: ChatMessageData[] = state.messages
   498→      .slice(0, userIdx)
   499→      .map((m) => ({ role: m.role, content: m.content }));
   500→
   501→    // Drop the failed bubble from the timeline.
   502→    set({ messages: state.messages.slice(0, failedIdx) });
   503→
   504→    // If the user message never received a server id, the turn died
   505→    // before persistence — persist it on retry. Otherwise it is
   506→    // already in the DB and must not be duplicated.
   507→    const persisted = !userMsg.id.startsWith("u-");
   508→    await streamTurn(set, get, {
   509→      chatId: state.temporary ? null : state.currentChatId,
   510→      history,
   511→      content: userMsg.content,
   512→      persistUser: !persisted,
   513→      appendUserBubble: false, // the original user bubble stays
   514→      temporary: state.temporary,
   515→      model: modelOverride,
   516→    });
   517→  },

   519→  async pickDuelWinner(messageId, laneId) {
   520→    const s = get();
   521→    const msg = s.messages.find((m) => m.id === messageId);
   522→    if (!msg?.duel || msg.duelPicked) return;
   523→    const winner = msg.duel[laneId];
   524→    const loser = msg.duel[laneId === "a" ? "b" : "a"];
   525→    if (!winner.content.trim()) return; // nothing to keep
   526→    const providerName = s.limiter?.providerName ?? null;
   527→
   528→    // Local commit first — the pick must feel instant.
   529→    set({
   530→      messages: s.messages.map((m) =>
   531→        m.id === messageId
   532→          ? {
   533→              ...m,
   534→              content: winner.content,
   535→              model: winner.model,
   536→              tokens: winner.tokens,
   537→              cached: winner.cached,
   538→              servedBy: winner.servedBy,
   539→              keyLabel: winner.keyLabel,
   540→              provider: providerName,
   541→              duel: null,
   542→              duelPicked: laneId,
   543→              runnerUp: loser.content.trim()
   544→                ? { content: loser.content, model: loser.model }
   545→                : null,
   546→              streaming: false,
   547→              phase: null,
   548→            }
   549→          : m
   550→      ),
   551→    });
   552→
   553→    // Persist for saved chats (temporary duels stay local-only).
   554→    if (!s.temporary && s.currentChatId) {
   555→      try {
   556→        const res = await fetch("/api/chat/duel/pick", {
   557→          method: "POST",
   558→          headers: { "Content-Type": "application/json" },
   559→          body: JSON.stringify({
   560→            chatId: s.currentChatId,
   561→            winner: {
   562→              content: winner.content,
   563→              model: winner.model,
   564→              tokens: winner.tokens,
   565→              cached: winner.cached,
   566→              servedBy: winner.servedBy,
   567→              keyLabel: winner.keyLabel,
   568→            },
   569→            runnerUp: loser.content.trim()
   570→              ? { content: loser.content, model: loser.model }
   571→              : null,
   572→            effort: s.effort,
   573→          }),
   574→        });
   575→        const data = (await res.json()) as { id?: string; error?: string };
   576→        if (typeof data.id === "string" && data.id) {
   577→          // Reconcile with the server id so later edits/truncates hit
   578→          // the real message row.
   579→          set((st) => ({
   580→            messages: st.messages.map((m) =>
   581→              m.id === messageId ? { ...m, id: data.id as string } : m
   582→            ),
   583→          }));
   584→        }
   585→      } catch {
   586→        // keep the local pick — the answer stays in the timeline
   587→      }
   588→    }
   589→  },
   590→
   591→  stopStreaming: () => {
   592→    streamController?.abort();
   593→    streamController = null;
   594→  },
   595→}));

   598→type SetState = (
   599→  partial: Partial<SenkuState> | ((s: SenkuState) => Partial<SenkuState>)
   600→) => void;
   601→type GetState = () => SenkuState;
   602→
   603→interface TurnOptions {
   604→  chatId: string | null;
   605→  history: ChatMessageData[];
   606→  content: string;
   607→  persistUser: boolean;
   608→  temporary?: boolean;
   609→  /** Overrides the store's active model (error-card "switch model"). */
   610→  model?: string;
   611→  /** Force whether a new local user bubble is appended. Defaults to
   612→      the persistUser flag — retries keep the original bubble but may
   613→      still need server-side persistence, so the two are decoupled. */
   614→  appendUserBubble?: boolean;
   615→  /** Duel: race the active model against `duel.challenger`. */
   616→  duel?: { challenger: string };
   617→}
   618→
   619→async function streamTurn(
   620→  set: SetState,
   621→  get: GetState,
   622→  opts: TurnOptions
   623→): Promise<void> {
   624→  const state = get();
   625→  const temporary = opts.temporary ?? state.temporary;
   626→  const model = opts.model ?? state.model;
   627→  const effort = state.effort;
   628→  const userId = `u-${Date.now()}`;
   629→  const assistantId = `a-${Date.now()}`;
   630→  const search = state.searchEnabled;
   631→  const duel = opts.duel;
   632→  const makeLane = (id: "a" | "b", laneModel: string): DuelLane => ({
   633→    id,
   634→    model: laneModel,
   635→    content: "",
   636→    streaming: true,
   637→    error: null,
   638→    phase: search ? ("searching" as const) : ("thinking" as const),
   639→    tokens: 0,
   640→    cached: false,
   641→    servedBy: null,
   642→    keyLabel: null,
   643→  });
   644→  // Regenerate-after-edit turns keep the (already updated) user message
   645→  // in the timeline — only fresh sends append a new local user bubble.
   646→  // Retries decouple the two: the bubble stays, persistence is decided
   647→  // by whether the failed turn ever reached the server.
   648→  const appendUserBubble = opts.appendUserBubble ?? (opts.persistUser !== false);
   649→  // Set when this turn creates a brand-new persisted chat — the
   650→  // raw derived title then gets refined by a cheap model pass.
   651→  let createdChatId: string | null = null;
   652→
   653→  set({
   654→    streaming: true,
   655→    messages: [
   656→      ...state.messages,
   657→      ...(appendUserBubble
   658→        ? [
   659→            {
   660→              id: userId,
   661→              role: "user" as const,
   662→              content: opts.content,
   663→              sources: search ? [] : null,
   664→            },
   665→          ]
   666→        : []),
   667→      {
   668→        id: assistantId,
   669→        role: "assistant" as const,
   670→        content: "",
   671→        streaming: true,
   672→        model,
   673→        provider: state.limiter?.providerName ?? null,
   674→        phase: search ? "searching" : "thinking",
   675→        effort,
   676→        ...(duel
   677→          ? {
   678→              duel: {
   679→                a: makeLane("a", model),
   680→                b: makeLane("b", duel.challenger),
   681→              },
   682→              duelPicked: null,
   683→            }
   684→          : {}),
   685→      },
   686→    ],
   687→  });
   688→
   689→  const updateAssistant = (patch: Partial<UIMessage>) => {
   690→    set((s) => ({
   691→      messages: s.messages.map((m) =>
   692→        m.id === assistantId ? { ...m, ...patch } : m
   693→      ),
   694→    }));
   695→  };
   696→  // Duel-only: patch one racing lane of the assistant message.
   697→  const updateLane = (lane: "a" | "b", patch: Partial<DuelLane>) => {
   698→    set((s) => ({
   699→      messages: s.messages.map((m) =>
   700→        m.id === assistantId && m.duel
   701→          ? { ...m, duel: { ...m.duel, [lane]: { ...m.duel[lane], ...patch } } }
   702→          : m
   703→      ),
   704→    }));
   705→  };
   706→  const updateUser = (patch: Partial<UIMessage>) => {
   707→    set((s) => {
   708→      const idx = appendUserBubble
   709→        ? s.messages.findIndex((m) => m.id === currentUserMsgId)
   710→        : (() => {
   711→            for (let i = s.messages.length - 1; i >= 0; i--) {
   712→              if (s.messages[i].role === "user") return i;
   713→            }
   714→            return -1;
   715→          })();
   716→      if (idx === -1) return {};
   717→      return {
   718→        messages: s.messages.map((m, i) => (i === idx ? { ...m, ...patch } : m)),
   719→      };
   720→    });
   721→  };
   722→
   723→  const controller = new AbortController();
   724→  streamController = controller;
   725→  // Reconciled with the server-assigned id as soon as "meta" arrives so
   726→  // later edits hit /api/chat/truncate with the real message id.
   727→  let currentUserMsgId = userId;
   728→
   729→  try {
   730→    const res = await fetch("/api/chat/send", {
   731→      method: "POST",
   732→      headers: { "Content-Type": "application/json" },
   733→      signal: controller.signal,
   734→      body: JSON.stringify({
   735→        chatId: opts.chatId,
   736→        history: opts.history,
   737→        content: opts.content,
   738→        model,
   739→        search,
   740→        temporary,
   741→        persistUser: opts.persistUser,
   742→        effort,
   743→        ...(duel ? { duel: true, challengerModel: duel.challenger } : {}),
   744→      }),
   745→    });
   746→
   747→    if (!res.ok || !res.body) {
   748→      const errText = await res.text().catch(() => "");
   749→      throw new Error(errText || `Request failed (${res.status})`);
   750→    }
   751→
   752→    const reader = res.body.getReader();
   753→    const decoder = new TextDecoder();
   754→    let buffer = "";
   755→    let assistantContent = "";
   756→    // Duel lanes accumulate independently; untagged deltas (normal
   757→    // turns) fall through to assistantContent only.
   758→    const laneContents: Record<"a" | "b", string> = { a: "", b: "" };
   759→
   760→    while (true) {
   761→      const { done, value } = await reader.read();
   762→      if (done) break;
   763→      buffer += decoder.decode(value, { stream: true });
   764→      const parts = buffer.split("\n\n");
   765→      buffer = parts.pop() ?? "";
   766→      for (const part of parts) {
   767→        const line = part.trim();
   768→        if (!line.startsWith("data:")) continue;
   769→        let ev: Record<string, unknown>;
   770→        try {
   771→          ev = JSON.parse(line.slice(5).trim());
   772→        } catch {
   773→          continue;
   774→        }
   775→        if (ev.type === "meta") {
   776→          if (!temporary && typeof ev.chatId === "string" && ev.chatId) {
   777→            set({ currentChatId: ev.chatId });
   778→            if (opts.chatId == null) createdChatId = ev.chatId;
   779→          }
   780→          // The effort dial may route the turn to a different pooled
   781→          // engine (fast → flash tier, deep → frontier tier) — the
   782→          // server's model is the truth the provenance must show.
   783→          if (typeof ev.model === "string" && ev.model) {
   784→            updateAssistant({ model: ev.model });
   785→          }
   786→          if (
   787→            typeof ev.userMessageId === "string" &&
   788→            ev.userMessageId &&
   789→            appendUserBubble
   790→          ) {
   791→            const serverId = ev.userMessageId;
   792→            // patch FIRST (message still carries the local id), then move
   793→            // the cursor so later patches (sources, …) target the new id
   794→            updateUser({ id: serverId });
   795→            currentUserMsgId = serverId;
   796→          }
   797→        } else if (ev.type === "duel_start") {
   798→          // Reconcile lane models with the server's normalization.
   799→          if (duel) {
   800→            if (typeof ev.a === "string" && ev.a) updateLane("a", { model: ev.a });
   801→            if (typeof ev.b === "string" && ev.b) updateLane("b", { model: ev.b });
   802→          }
   803→        } else if (ev.type === "search_status") {
   804→          const status = String(ev.status ?? "");
   805→          const detail =
   806→            typeof ev.detail === "string" && ev.detail ? ev.detail : null;
   807→          if (status === "searching") {
   808→            updateAssistant({ phase: "searching", searchDetail: detail });
   809→            if (duel) {
   810→              updateLane("a", { phase: "searching" });
   811→              updateLane("b", { phase: "searching" });
   812→            }
   813→          } else if (status === "ok") {
   814→            updateAssistant({ phase: "thinking", searchFailed: false, searchDetail: null });
   815→            if (duel) {
   816→              updateLane("a", { phase: "thinking" });
   817→              updateLane("b", { phase: "thinking" });
   818→            }
   819→          } else if (status === "failed") {
   820→            updateAssistant({ phase: "thinking", searchFailed: true, searchDetail: null });
   821→            if (duel) {
   822→              updateLane("a", { phase: "thinking" });
   823→              updateLane("b", { phase: "thinking" });
   824→            }
   825→          }
   826→        } else if (ev.type === "sources") {
   827→          const sources = (ev.sources ?? []) as SearchSource[];
   828→          if (sources.length) updateUser({ sources });
   829→        } else if (ev.type === "restart") {
   830→          // A provider died mid-stream; the cloud re-generates that
   831→          // lane from scratch. Discard only the affected lane's
   832→          // partial text (untagged = normal single-lane turn).
   833→          const lane = ev.lane === "b" ? "b" : "a";
   834→          if (duel) {
   835→            laneContents[lane] = "";
   836→            updateLane(lane, { content: "", phase: "thinking" });
   837→          } else {
   838→            assistantContent = "";
   839→            updateAssistant({ content: "", phase: "thinking" });
   840→          }
   841→        } else if (ev.type === "delta") {
   842→          const lane = ev.lane === "b" ? "b" : "a";
   843→          if (duel) {
   844→            laneContents[lane] += String(ev.text ?? "");
   845→            updateLane(lane, { content: laneContents[lane], phase: null });
   846→          } else {
   847→            assistantContent += String(ev.text ?? "");
   848→            updateAssistant({ content: assistantContent, phase: null });
   849→          }
   850→        } else if (ev.type === "lane_done") {
   851→          // Final per-lane stats from the gateway — server truth wins.
   852→          const lane = ev.lane === "b" ? "b" : "a";
   853→          if (duel) {
   854→            updateLane(lane, {
   855→              content: (ev.content as string) ?? laneContents[lane],
   856→              streaming: false,
   857→              phase: null,
   858→              tokens: (ev.tokens as number) ?? 0,
   859→              cached: ev.cached === true,
   860→              servedBy: (ev.servedBy as string) ?? null,
   861→              keyLabel: (ev.keyLabel as string) ?? null,
   862→              error: (ev.error as string) ?? null,
   863→            });
   864→          }
   865→        } else if (ev.type === "done") {
   866→          if (duel && ev.duel && typeof ev.duel === "object") {
   867→            const lanes = ev.duel as {
   868→              a: { content: string; error: string | null; tokens: number; cached: boolean; servedBy: string | null; keyLabel: string | null; model: string };
   869→              b: { content: string; error: string | null; tokens: number; cached: boolean; servedBy: string | null; keyLabel: string | null; model: string };
   870→            };
   871→            updateAssistant({
   872→              streaming: false,
   873→              phase: null,
   874→              searchFailed: ev.searchFailed === true,
   875→            });
   876→            updateLane("a", {
   877→              content: lanes.a.content,
   878→              model: lanes.a.model,
   879→              streaming: false,
   880→              phase: null,
   881→              tokens: lanes.a.tokens,
   882→              cached: lanes.a.cached,
   883→              servedBy: lanes.a.servedBy,
   884→              keyLabel: lanes.a.keyLabel,
   885→              error: lanes.a.error,
   886→            });
   887→            updateLane("b", {
   888→              content: lanes.b.content,
   889→              model: lanes.b.model,
   890→              streaming: false,
   891→              phase: null,
   892→              tokens: lanes.b.tokens,
   893→              cached: lanes.b.cached,
   894→              servedBy: lanes.b.servedBy,
   895→              keyLabel: lanes.b.keyLabel,
   896→              error: lanes.b.error,
   897→            });
   898→            // One lane died → the duel silently degrades: keep the
   899→            // surviving lane as the answer (no pick needed).
   900→            if (lanes.a.error && !lanes.b.error && lanes.b.content.trim()) {
   901→              void get().pickDuelWinner(assistantId, "b");
   902→            } else if (lanes.b.error && !lanes.a.error && lanes.a.content.trim()) {
   903→              void get().pickDuelWinner(assistantId, "a");
   904→            }
   905→            // Both lanes healthy → the message waits for the user's pick.
   906→          } else {
   907→            updateAssistant({
   908→              id: (ev.assistantMessageId as string) ?? assistantId,
   909→              content: (ev.content as string) ?? assistantContent,
   910→              streaming: false,
   911→              cached: Boolean(ev.cached),
   912→              tokens: (ev.tokens as number) ?? 0,
   913→              servedBy: (ev.servedBy as string) ?? null,
   914→              keyLabel: (ev.keyLabel as string) ?? null,
   915→              provider: (ev.providerName as string) ?? null,
   916→              model: (ev.model as string) ?? model,
   917→              effort:
   918→                ev.effort === "fast" || ev.effort === "deep" || ev.effort === "medium"
   919→                  ? ev.effort
   920→                  : effort,
   921→              phase: null,
   922→              searchDetail: null,
   923→              searchFailed: ev.searchFailed === true,
   924→            });
   925→          }
   926→        } else if (ev.type === "error") {
   927→          updateAssistant({
   928→            streaming: false,
   929→            error: String(ev.message ?? "Generation failed"),
   930→            content: "",
   931→            phase: null,
   932→          });
   933→        }
   934→      }
   935→    }
   936→  } catch (e) {
   937→    const err = e as Error;
   938→    if (err.name === "AbortError") {
   939→      // User pressed stop: a stopped duel keeps both partial lanes —
   940→      // the user can still pick either one as the answer.
   941→      if (duel) {
   942→        updateAssistant({ streaming: false, phase: null });
   943→        updateLane("a", { streaming: false, phase: null });
   944→        updateLane("b", { streaming: false, phase: null });
   945→      } else {
   946→        updateAssistant({ streaming: false, phase: null });
   947→      }
   948→    } else {
   949→      updateAssistant({
   950→        streaming: false,
   951→        error: err.message,
   952→        content: "",
   953→        phase: null,
   954→      });
   955→    }
   956→  } finally {
   957→    streamController = null;
   958→    set({ streaming: false });
   959→    if (!temporary) {
   960→      // Await the refresh so the raw derived title is in the list
   961→      // BEFORE the refined one patches it (no clobber race).
   962→      await get().refreshChats();
   963→      if (createdChatId) void refineTitle(set, createdChatId);
   964→    }
   965→    void get().refreshUsage();
   966→  }
   967→}
   968→
   969→/**
   970→ * Replace a brand-new chat's raw derived title ("who r u?") with a
   971→ * proper one from a cheap cloud model pass: "Who Are You?". Fire and
   972→ * forget — failure keeps the derived title.
   973→ */
   974→async function refineTitle(set: SetState, chatId: string): Promise<void> {
   975→  try {
   976→    const res = await fetch("/api/chat/title", {
   977→      method: "POST",
   978→      headers: { "Content-Type": "application/json" },
   979→      body: JSON.stringify({ chatId }),
   980→    });
   981→    if (!res.ok) return;
   982→    const data = (await res.json()) as { title?: string };
   983→    const title = typeof data.title === "string" ? data.title.trim() : "";
   984→    if (!title) return;
   985→    set((s) => ({
   986→      chats: s.chats.map((c) => (c.id === chatId ? { ...c, title } : c)),
   987→    }));
   988→  } catch {
   989→    // keep the derived title
   990→  }
   991→}
