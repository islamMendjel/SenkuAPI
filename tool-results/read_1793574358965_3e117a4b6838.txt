   234→   234→/** One quiet provenance line: "GLM 5.3 · APInex pool · 1.9k tokens · cache: hit". */
   235→   235→function provenanceLine(m: UIMessage): string {
   236→   236→  const parts: string[] = [];
   237→   237→  // The effort dial leaves its mark — fast/deep turns say so.
   238→   238→  if (m.effort === "fast") parts.push("Fast mode");
   239→   239→  if (m.effort === "deep") parts.push("Deep research");
   240→   240→  if (m.model) parts.push(modelMeta(m.model).name);
   241→   241→  if (m.provider) {
   242→   242→    parts.push(
   243→   243→      m.servedBy === "zai-cloud"
   244→   244→        ? `${m.provider} · cloud failover`
   245→   245→        : `${m.provider} pool`
   246→   246→    );
   247→   247→  }
   248→   248→  if (typeof m.tokens === "number" && m.tokens > 0) {
   249→   249→    parts.push(`${formatCompact(m.tokens)} tokens`);
   250→   250→  }
   251→   251→  parts.push(m.cached ? "cache: hit" : "cache: miss");
   252→   252→  return parts.join(" · ");
   253→   253→}

   257→   257→  if (!confidenceOf(content)) return content;
   258→   258→  const lines = content.trimEnd().split("\n");
   259→   259→  return lines.slice(0, -1).join("\n").trimEnd();
   260→   260→}

   379→   379→/** A duel turn awaiting its pick: two models racing side by side
   380→   380→    (stacked on mobile), each with its own Keep button. */
   381→   381→function DuelMessage({ message }: { message: UIMessage }) {
   382→   382→  const { pickDuelWinner } = useSenku();
   383→   383→  const { toast } = useToast();
   384→   384→  const duel = message.duel;
   385→   385→  if (!duel) return null;
   386→   386→  // Picks unlock once the turn finishes (or is stopped) — picking
   387→   387→  // mid-stream would race the lane_done/done reconciliation.
   388→   388→  const canPick = !message.streaming;
   389→   389→
   390→   390→  const keep = (lane: "a" | "b") => {
   391→   391→    const meta = modelMeta(duel[lane].model);
   392→   392→    void pickDuelWinner(message.id, lane).then(() => {
   393→   393→      toast({
   394→   394→        description: `Kept ${meta.name} — the runner-up stays under the answer.`,
   395→   395→      });
   396→   396→    });
   397→   397→  };
   398→   398→
   399→   399→  return (
   400→   400→    <div>
   401→   401→      <div className="mb-2 flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-amber-300/70">
   402→   402→        <Swords className="h-3 w-3" aria-hidden />
   403→   403→        Duel — pick the keeper
   404→   404→      </div>
   405→   405→      <div className="grid grid-cols-1 items-start gap-3 md:grid-cols-2">
   406→   406→        <DuelLaneCard lane={duel.a} canPick={canPick} onKeep={() => keep("a")} />
   407→   407→        <DuelLaneCard lane={duel.b} canPick={canPick} onKeep={() => keep("b")} />
   408→   408→      </div>
   409→   409→      <p className="mt-2 text-[11px] text-zinc-600">
   410→   410→        Both answers are billed separately ·{" "}
   411→   411→        {canPick
   412→   412→          ? "pick one to keep it in the chat"
   413→   413→          : "generating both answers…"}
   414→   414→      </p>
   415→   415→    </div>
   416→   416→  );
   417→   417→}

   419→   419→function AssistantMessage({
   420→   420→  message,
   421→   421→  speech,
   422→   422→  onToggleSpeech,
   423→   423→}: {
   424→   424→  message: UIMessage;
   425→   425→  speech: SpeechState;
   426→   426→  onToggleSpeech: (m: UIMessage) => void;
   427→   427→}) {
   428→   428→  const { models, model, setModel, retryLast, streaming } = useSenku();
   429→   429→  const isThinking = message.streaming && !message.content && !message.error;
   430→   430→  const isSpeakingThis = speech?.id === message.id;
   431→   431→  const speechLoading = isSpeakingThis && speech.phase === "loading";
   432→   432→  const speechPlaying = isSpeakingThis && speech.phase === "playing";
   433→   433→
   434→   434→  // Error recovery: the next model in the list (wraps around).
   435→   435→  const currentIdx = models.indexOf(model);
   436→   436→  const switchTarget =
   437→   437→    models.length > 1
   438→   438→      ? models[((currentIdx === -1 ? 0 : currentIdx) + 1) % models.length]
   439→   439→      : null;
   440→   440→
   441→   441→  // Grounded-answer contract: the trailing confidence label becomes
   442→   442→  // a badge once streaming completes (while streaming, the final
   443→   443→  // line may still be half-written — render it raw).
   444→   444→  const confidence = message.error ? null : confidenceOf(message.content);
   445→   445→  const displayContent =
   446→   446→    !message.streaming && confidence
   447→   447→      ? withoutConfidenceLine(message.content)
   448→   448→      : message.content;
   449→   449→

   627→   627→export function ChatTimeline() {
   628→   628→  const { messages } = useSenku();
   629→   629→  const { toast } = useToast();
   630→   630→  const bottomRef = useRef<HTMLDivElement>(null);
   631→   631→
   632→   632→  // ── Read-aloud (TTS) playback state ──
   633→   633→  // One audio element for the whole timeline; only one message can
   634→   634→  // speak at a time. A token guards against stale fetches after a stop.
   635→   635→  const [speech, setSpeech] = useState<SpeechState>(null);
   636→   636→  const audioRef = useRef<HTMLAudioElement | null>(null);
   637→   637→  const audioUrlRef = useRef<string | null>(null);
   638→   638→  const speechTokenRef = useRef(0);
   639→   639→
   640→   640→  const stopSpeech = () => {
   641→   641→    speechTokenRef.current += 1;
   642→   642→    if (audioRef.current) {
   643→   643→      audioRef.current.pause();
   644→   644→      audioRef.current.removeAttribute("src");
   645→   645→    }
   646→   646→    if (audioUrlRef.current) {
   647→   647→      URL.revokeObjectURL(audioUrlRef.current);
   648→   648→      audioUrlRef.current = null;
   649→   649→    }
   650→   650→    setSpeech(null);
   651→   651→  };
   652→   652→
   653→   653→  const toggleSpeech = async (m: UIMessage) => {
   654→   654→    if (speech?.id === m.id) {
   655→   655→      stopSpeech();
   656→   656→      return;
   657→   657→    }
   658→   658→    stopSpeech();
   659→   659→    const token = ++speechTokenRef.current;
   660→   660→    setSpeech({ id: m.id, phase: "loading" });
   661→   661→    try {
   662→   662→      const res = await fetch("/api/voice/speak", {
   663→   663→        method: "POST",
   664→   664→        headers: { "Content-Type": "application/json" },
   665→   665→        body: JSON.stringify({ text: m.content }),
   666→   666→      });
   667→   667→      if (!res.ok) {
   668→   668→        const data = await res.json().catch(() => ({}));
   669→   669→        throw new Error(data.error || "Read-aloud is unavailable right now.");
   670→   670→      }
   671→   671→      if (token !== speechTokenRef.current) return; // user moved on
   672→   672→      const blob = await res.blob();
   673→   673→      const url = URL.createObjectURL(blob);
   674→   674→      audioUrlRef.current = url;
   675→   675→      const audio = (audioRef.current ??= new Audio());
   676→   676→      audio.src = url;
   677→   677→      audio.onended = () => {
   678→   678→        if (token === speechTokenRef.current) stopSpeech();
   679→   679→      };
   680→   680→      audio.onerror = () => {
   681→   681→        if (token === speechTokenRef.current) stopSpeech();
   682→   682→      };
   683→   683→      await audio.play();
   684→   684→      if (token !== speechTokenRef.current) return;
   685→   685→      setSpeech({ id: m.id, phase: "playing" });
   686→   686→    } catch (e) {
   687→   687→      if (token === speechTokenRef.current) {
   688→   688→        setSpeech(null);
   689→   689→        if ((e as Error).name !== "NotAllowedError") {
   690→   690→          toast({
   691→   691→            description: (e as Error).message || "Could not play the audio.",
   692→   692→            variant: "destructive",
   693→   693→          });
   694→   694→        }
   695→   695→      }
   696→   696→    }
   697→   697→  };
   698→   698→
   699→   699→  // stop playback when the timeline unmounts or the chat changes
   700→   700→  useEffect(() => {
   701→   701→    return () => {
   702→   702→      speechTokenRef.current += 1;
   703→   703→      audioRef.current?.pause();
   704→   704→      if (audioUrlRef.current) URL.revokeObjectURL(audioUrlRef.current);
   705→   705→    };
   706→   706→  }, []);
   707→   707→
   708→   708→  useEffect(() => {
   709→   709→    bottomRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
   710→   710→  }, [messages]);
   711→   711→
   712→   712→  return (
   713→   713→    <div className="senku-scroll flex-1 overflow-y-auto">
   714→   714→      <div className="mx-auto w-full max-w-3xl space-y-6 px-4 pb-6 pt-4">
   715→   715→        {messages.map((m) =>
   716→   716→          m.role === "user" ? (
   717→   717→            <UserMessage key={m.id} message={m} />
   718→   718→          ) : (
   719→   719→            <AssistantMessage
   720→   720→              key={m.id}
   721→   721→              message={m}
   722→   722→              speech={speech}
   723→   723→              onToggleSpeech={(msg) => void toggleSpeech(msg)}
   724→   724→            />
   725→   725→          )
   726→   726→        )}
   727→   727→        <div ref={bottomRef} />
   728→   728→      </div>
   729→   729→    </div>
   730→   730→  );
   731→   731→}
