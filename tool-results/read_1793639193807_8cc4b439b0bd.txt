    43→function formatChars(n: number): string {
    44→  if (n >= 1000) return `${(n / 1000).toFixed(1)}k chars`;
    45→  return `${n} chars`;
    46→}

    48→function kindIcon(p: PendingAttachment) {
    49→  return p.kindGuess === "image";
    50→}

    52→/** One staged file: icon + name + live extraction status. The chip
    53→ *  tells the user exactly what the model will receive — chars read,
    54→ *  vision pass, truncation — or why the file can't be sent. */
    55→function AttachmentChip({
    56→  p,
    57→  onRemove,
    58→  disabled,
    59→}: {
    60→  p: PendingAttachment;
    61→  onRemove: () => void;
    62→  disabled: boolean;
    63→}) {
    64→  const isImage = kindIcon(p);
    65→  return (
    66→    <div
    67→      className={`flex w-full items-center gap-2 rounded-xl border px-2.5 py-2 text-left ${
    68→        p.status === "error"
    69→          ? "border-red-500/30 bg-red-500/5"
    70→          : "border-zinc-700/70 bg-[#1d1d20]"
    71→      }`}
    72→      aria-label={
    73→        p.status === "ready"
    74→          ? `Attachment ${p.name}, read by the gateway`
    75→          : `Attachment ${p.name}`
    76→      }
    77→    >
    78→      {isImage ? (
    79→        <FileImage
    80→          className={`h-4 w-4 shrink-0 ${
    81→            p.status === "error" ? "text-red-400/80" : "text-zinc-400"
    82→          }`}
    83→          aria-hidden
    84→        />
    85→      ) : (
    86→        <FileText
    87→          className={`h-4 w-4 shrink-0 ${
    88→            p.status === "error" ? "text-red-400/80" : "text-zinc-400"
    89→          }`}
    90→          aria-hidden
    91→        />
    92→      )}
    93→      <span className="min-w-0 flex-1">
    94→        <span className="block truncate text-[12.5px] font-medium text-zinc-200">
    95→          {p.name}
    96→        </span>
    97→        {p.status === "extracting" && (
    98→          <span className="flex items-center gap-1.5 text-[11px] text-zinc-500">
    99→            <Loader2 className="h-3 w-3 animate-spin" aria-hidden />
   100→            reading…
   101→          </span>
   102→        )}
   103→        {p.status === "ready" && p.record && (
   104→          <span
   105→            className="flex items-center gap-1 text-[11px] text-emerald-300/80"
   106→            title={
   107→              p.record.truncated
   108→                ? `Sent to the model: first ${formatChars(p.record.chars)} of ${p.record.totalChars} — the rest was cut to fit`
   109→                : `Sent to the model: ${p.record.chars} characters${p.record.kind === "image" ? " from a vision pass (OCR + description)" : " read from the file"}`
   110→            }
   111→          >
   112→            {p.record.kind === "image" ? (
   113→              <Eye className="h-3 w-3 shrink-0" aria-hidden />
   114→            ) : (
   115→              <Check className="h-3 w-3 shrink-0" aria-hidden />
   116→            )}
   117→            {p.record.kind === "image" ? "vision pass · " : "read · "}
   118→            {formatChars(p.record.chars)}
   119→            {p.record.truncated ? " (truncated)" : ""}
   120→          </span>
   121→        )}
   122→        {p.status === "error" && (
   123→          <span className="flex items-start gap-1 text-[11px] leading-snug text-red-300/80">
   124→            <TriangleAlert className="mt-px h-3 w-3 shrink-0" aria-hidden />
   125→            {p.error} — not sent
   126→          </span>
   127→        )}
   128→      </span>
   129→      <button
   130→        onClick={onRemove}
   131→        disabled={disabled}
   132→        aria-label={`Remove ${p.name}`}
   133→        className="flex h-6 w-6 shrink-0 items-center justify-center rounded-md text-zinc-500 transition-colors hover:bg-white/10 hover:text-zinc-200 disabled:opacity-40"
   134→      >
   135→        <X className="h-3.5 w-3.5" />
   136→      </button>
   137→    </div>
   138→  );
   139→}

   175→function formatElapsed(ms: number): string {
   176→  const total = Math.floor(ms / 1000);
   177→  return `${Math.floor(total / 60)}:${String(total % 60).padStart(2, "0")}`;
   178→}

   180→export function InputBar() {
   181→  const {
   182→    searchEnabled,
   183→    toggleSearch,
   184→    temporary,
   185→    model,
   186→    models,
   187→    setModel,
   188→    syncModels,
   189→    streaming,
   190→    sendMessage,

   193→    duelEnabled,

   198→    setEffort,

   203→  const { toast } = useToast();
   204→
   205→  const [value, setValue] = useState("");

   207→  const [modelOpen, setModelOpen] = useState(false);
   208→  const [challengerOpen, setChallengerOpen] = useState(false);
   209→  const [effortOpen, setEffortOpen] = useState(false);
   210→  const [syncing, setSyncing] = useState(false);
   211→  const [transcribing, setTranscribing] = useState(false);
   212→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   213→  const fileRef = useRef<HTMLInputElement>(null);
   214→  const recorder = useVoiceRecorder();
   215→  const waveBarsRef = useRef<(HTMLDivElement | null)[]>([]);
   216→  const waveLevelsRef = useRef<number[]>(new Array(WAVE_BARS).fill(4));
   217→
   218→  const providerName = limiter?.providerName ?? "";
   219→  const extractingAttachments = pendingAttachments.some(
   220→    (p) => p.status === "extracting"
   221→  );
   222→  const canSend =
   223→    value.trim().length > 0 && !streaming && !extractingAttachments;
   224→  const isRecording = recorder.state === "recording";
   225→  const isProcessing = recorder.state === "processing" || transcribing;
   226→  // A duel needs a second model to race against.
   227→  const duelAvailable = models.filter((m) => m !== model).length > 0;
   228→
   229→  // auto-grow textarea
   230→  useEffect(() => {
   231→    const el = textareaRef.current;
   232→    if (!el) return;
   233→    el.style.height = "0px";
   234→    el.style.height = Math.min(el.scrollHeight, 208) + "px";
   235→  }, [value]);
   236→
   237→  // surface recorder permission/device errors
   238→  useEffect(() => {
   239→    if (recorder.error) {
   240→      toast({ description: recorder.error, variant: "destructive" });
   241→    }
   242→  }, [recorder.error, toast]);
   243→
   244→  // live waveform: push the smoothed mic level through a rolling window
   245→  // and paint bar heights directly on the DOM (no re-render per frame)
   246→  useEffect(() => {
   247→    if (recorder.state !== "recording") return;
   248→    let raf = 0;
   249→    const loop = () => {
   250→      const level = recorder.levelRef.current;
   251→      const next = waveLevelsRef.current
   252→        .slice(1)
   253→        .concat(4 + level * level * 42 + Math.random() * 2);
   254→      waveLevelsRef.current = next;
   255→      for (let i = 0; i < WAVE_BARS; i++) {
   256→        const el = waveBarsRef.current[i];
   257→        if (el) el.style.height = `${Math.max(4, next[i]).toFixed(1)}px`;
   258→      }
   259→      raf = requestAnimationFrame(loop);
   260→    };
   261→    raf = requestAnimationFrame(loop);
   262→    return () => cancelAnimationFrame(raf);
   263→  }, [recorder.state, recorder.levelRef]);
   264→
   265→  const submit = () => {
   266→    if (!canSend) return;
   267→    const content = value.trim();
   268→    setValue("");
   269→    void sendMessage(content);
   270→  };
   271→
   272→  // stop → transcribe (server ASR) → land the text in the input
   273→  const finishDictation = async () => {
   274→    recorder.stop();
   275→    setTranscribing(true);
   276→    try {
   277→      const wav = await recorder.takeResult();
   278→      if (!wav) {
   279→        toast({ description: "Didn't catch that — try speaking a bit louder." });
   280→        return;
   281→      }
   282→      const res = await fetch("/api/voice/transcribe", {
   283→        method: "POST",
   284→        headers: { "Content-Type": "application/json" },
   285→        body: JSON.stringify({ audio: wav }),
   286→      });
   287→      const data = await res.json();
   288→      if (!res.ok) throw new Error(data.error || "Transcription failed");
   289→      const text = String(data.text ?? "").trim();
   290→      if (!text) {
   291→        toast({ description: "Didn't catch that — try speaking a bit louder." });
   292→        return;
   293→      }
   294→      setValue((v) => (v ? `${v} ${text}` : text));
   295→      requestAnimationFrame(() => {
   296→        const el = textareaRef.current;
   297→        if (el) {
   298→          el.focus();
   299→          el.setSelectionRange(el.value.length, el.value.length);
   300→        }
   301→      });
   302→    } catch (e) {
   303→      toast({
   304→        description: (e as Error).message || "Transcription failed.",
   305→        variant: "destructive",
   306→      });
   307→    } finally {
   308→      setTranscribing(false);
   309→    }
   310→  };
   311→
   312→  const onMicClick = () => {
   313→    if (recorder.state === "recording") {
   314→      void finishDictation();
   315→    } else if (recorder.state === "idle") {
   316→      void recorder.start();
   317→    }
   318→  };
   319→
   320→  const onUpload = () => {
   321→    setAttachOpen(false);
   322→    fileRef.current?.click();
   323→  };
   324→
   325→  const onFilePicked = (e: React.ChangeEvent<HTMLInputElement>) => {
   326→    const files = Array.from(e.target.files ?? []);
   327→    e.target.value = "";
   328→    if (!files.length) return;
   329→    const room = ATTACHMENT_CAP - pendingAttachments.length;
   330→    if (room <= 0) {
   331→      toast({
   332→        description: `Up to ${ATTACHMENT_CAP} files per message.`,
   333→        variant: "destructive",
   334→      });
   335→      return;
   336→    }
   337→    const accepted = files.slice(0, room);
   338→    if (files.length > accepted.length) {
   339→      toast({
   340→        description: `Only ${accepted.length} of ${files.length} files added — up to ${ATTACHMENT_CAP} per message.`,
   341→      });
   342→    }
   343→    void addAttachments(accepted).then(() => {
   344→      /* per-file results are shown on the chips */
   345→    });
   346→  };
   347→
   348→  const activeMeta = modelMeta(model);
   349→
   350→  return (
   351→    <div className="relative mx-auto w-full max-w-3xl shrink-0 px-4">
   352→      <input
   353→        ref={fileRef}
   354→        type="file"
   355→        multiple
   356→        hidden
   357→        onChange={onFilePicked}
   358→        aria-hidden
   359→      />
   360→
   361→      <div className="rounded-[26px] border border-zinc-700/70 bg-[#262627] shadow-[0_8px_32px_rgba(0,0,0,0.45)] transition-colors focus-within:border-zinc-500">
   362→        {isRecording ? (
   363→          /* live waveform strip replaces the textarea while dictating */
   364→          <div className="senku-rec-in flex h-[52px] items-center gap-3 px-4" aria-live="polite">
   365→            <span className="flex shrink-0 items-center gap-2 tabular-nums text-[13px] text-red-400">
   366→              <span className="senku-rec-dot inline-block h-2 w-2 rounded-full bg-red-400" />
   367→              {formatElapsed(recorder.elapsedMs)}
   368→            </span>
   369→            <div className="flex h-10 min-w-0 flex-1 items-center justify-center gap-[3px]" aria-hidden>
   370→              {Array.from({ length: WAVE_BARS }).map((_, i) => (
   371→                <div
   372→                  key={i}
   373→                  ref={(el) => {
   374→                    waveBarsRef.current[i] = el;
   375→                  }}
   376→                  className="senku-wave-bar"
   377→                  style={{ height: 4 }}
   378→                />
   379→              ))}
   380→            </div>
   381→            <span className="hidden shrink-0 text-[11px] text-zinc-500 sm:block">
   382→              tap ◼ to stop
   383→            </span>
   384→          </div>
   385→        ) : isProcessing ? (
   386→          <div className="flex h-[52px] items-center px-4" aria-live="polite">
   387→            <span className="senku-thinking-label text-[14px] font-medium">
   388→              Transcribing…
   389→            </span>
   390→          </div>
   391→        ) : (
   392→          <textarea
   393→            ref={textareaRef}
   394→            value={value}
   395→            onChange={(e) => setValue(e.target.value)}
   396→            onKeyDown={(e) => {
   397→              if (e.key === "Enter" && !e.shiftKey) {
   398→                e.preventDefault();
   399→                submit();
   400→              }
   401→            }}
   402→            rows={1}
   403→            placeholder="Message SenkuAPI..."
   404→            aria-label="Message SenkuAPI"
   405→            className="senku-scroll max-h-52 w-full resize-none bg-transparent px-4 pb-1 pt-3.5 text-[15px] leading-relaxed text-zinc-100 outline-none placeholder:text-zinc-500"
   406→          />
   407→        )}

   409→        {/* staged attachments — one chip per file, live extraction
