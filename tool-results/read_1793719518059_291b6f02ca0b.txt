    42→    | "meta"

    54→export interface TurnParams {
    55→  chatId?: string | null;
    56→  history: ChatMessageData[]; // prior turns (already persisted)
    57→  content: string; // new user message
    58→  model: string;
    59→  search: boolean;
    60→  temporary: boolean; // temporary chats are never persisted
    61→  persistUser?: boolean; // false when regenerating after an edit (message already updated)
    62→  /** Effort dial. "medium" (default) = byte-identical legacy
    63→   *  behavior; "fast" routes to the fastest pooled engine with a
    64→   *  brevity directive; "deep" routes to the strongest pooled
    65→   *  engine with thinking + multi-round audited retrieval. */
    66→  effort?: EffortLevel;
    67→  /** Files the gateway ALREADY read (extraction happened at pick
    68→   *  time). Their text is appended to the request context — never
    69→   *  to the stored message content — and persisted on the user
    70→   *  message row so follow-up turns and reloads keep the context. */
    71→  attachments?: AttachmentRecord[];
    72→  /** One-shot response style (Run 17 slash commands): "code" →
    73→   *  code-focused directive, "concise" → ultra-brief directive.
    74→   *  Appended to the system prompt (and thus the cache key) — the
    75→   *  /code and /concise variants of one question cache separately. */
    76→  style?: TurnStyle;
    77→  /** Fires when the CLIENT disconnects (Stop button, tab close).
    78→   *  The upstream generation is cancelled at the source, the
    79→   *  partial reply is persisted, and the truncated answer NEVER
    80→   *  enters the response cache. */
    81→  signal?: AbortSignal;
    82→  emit: (ev: TurnEvent) => void;
    83→}

    85→export interface TurnResult {
    86→  chatId: string | null;
    87→  userMessageId: string | null;
    88→  assistantMessageId: string | null;
    89→  title: string | null;
    90→  sources: SearchSource[];
    91→  content: string;
    92→  cached: boolean;
    93→  tokens: number; // effective (multiplied) tokens billed — 0 when cached
    94→  rawTokens: number;
    95→  servedBy: "provider" | "zai-cloud" | "cache";
    96→  keyLabel: string | null;
    97→  providerName: string;
    98→  model: string;
    99→  /** which effort level served this turn */
   100→  effort: EffortLevel;
   101→  /** true when the token count is a chars/4 estimate because the
   102→   *  upstream did not report usage (Run 17 — the UI marks it) */
   103→  tokensEstimated: boolean;
   104→  /** true when search was requested but every engine failed */
   105→  searchFailed: boolean;
   106→  /** true when the client disconnected mid-generation — content is
   107→   *  the partial text that actually streamed */
   108→  aborted?: boolean;
   109→}

   111→export function deriveTitle(content: string): string {
   112→  const clean = content
   113→    .replace(/\s+/g, " ")
   114→    .replace(/[^\p{L}\p{N} ?!.,'’-]/gu, "")
   115→    .trim();
   116→  const words = clean.split(" ").filter(Boolean).slice(0, 6).join(" ");
   117→  return (words || "New chat").slice(0, 60);
   118→}
   119→
   120→function monthKey(d = new Date()): string {
   121→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   122→}
   123→
   124→/** History rows keep their attachment extracts — a follow-up
   125→ *  question ("what was the revenue figure?") must still see the
   126→ *  file from the earlier turn, exactly like the incumbents. */
   127→function expandHistory(history: ChatMessageData[]): ChatMessageData[] {
   128→  return history.map((m) =>
   129→    m.role === "user" && m.attachments?.length
   130→      ? { ...m, content: m.content + attachmentContextBlock(m.attachments) }
   131→      : m
   132→  );
   133→}

   135→export async function runChatTurn(params: TurnParams): Promise<TurnResult> {
   136→  const { emit } = params;
   137→  const provider = await getActiveProvider();
   138→  const providerModels = JSON.parse(provider.models || "[]") as string[];
   139→  const requestedModel =
   140→    params.model || providerModels[0] || "free/glm-5.3";
   141→  const effort: EffortLevel = params.effort ?? "medium";
   142→  // The effort dial routes across the POOL: fast → the fastest
   143→  // engine the provider offers, deep → the strongest. Medium keeps
   144→  // the user's selection untouched (legacy behavior).
   145→  const model = effortModel(providerModels, requestedModel, effort);
   146→  // Untrusted client payload — rebuilt field-by-field server-side.
   147→  const attachments = sanitizeAttachments(params.attachments);
   148→
   149→  // ── 1. Web search ──
   150→  // Medium/fast: strict discipline (≤3 snippets, ≤300 chars), one
   151→  // pass. Deep: multi-round iterative retrieval — plan queries,
   152→  // search the question verbatim, then the planned angles in
   153→  // parallel, merged URL-deduped (≤8 sources), with live stage
   154→  // labels streamed to the client.
   155→  let sources: SearchSource[] = [];
   156→  let searchEngine: string | undefined;
   157→  if (params.search) {
   158→    if (effort === "deep") {
   159→      emit({ type: "search_status", status: "searching", depth: "deep" });
   160→      const outcome = await deepSearchDetailed(params.content, (detail) =>
   161→        emit({ type: "search_status", status: "searching", detail, depth: "deep" })
   162→      );
   163→      sources = outcome.results;
   164→      searchEngine = "deep";
   165→    } else {
   166→      emit({ type: "search_status", status: "searching" });
   167→      const outcome = await webSearchDetailed(params.content);
   168→      sources = outcome.results;
   169→      searchEngine = outcome.source;
   170→    }
   171→    emit({ type: "sources", sources });
   172→    emit({
   173→      type: "search_status",
   174→      status: sources.length ? "ok" : "failed",
   175→      engine: searchEngine,
   176→      depth: effort === "deep" ? "deep" : undefined,
   177→    });
   178→  }
   179→
   180→  // Effort directives append AFTER the base contract. Medium adds
   181→  // NOTHING — the prompt stays byte-identical to pre-dial turns so
   182→  // the 24h response cache still hits. Fast/deep additions also
   183→  // namespace the cache key per effort level for free.
   184→  const effortAddon =
   185→    effort === "fast"
   186→      ? FAST_SYSTEM_ADDON
   187→      : effort === "deep"
   188→        ? sources.length
   189→          ? DEEP_RESEARCH_ADDON
   190→          : DEEP_THINKING_ADDON
   191→        : "";
   192→  // One-shot response style (/code, /concise) — appended AFTER the
   193→  // effort addon. Medium + no style stays byte-identical to legacy
   194→  // turns so the 24h response cache still hits for them.
   195→  const styleAddon =
   196→    params.style === "code"
   197→      ? CODE_STYLE_ADDON
   198→      : params.style === "concise"
   199→        ? CONCISE_STYLE_ADDON
   200→        : "";
   201→  // Run 28 — the user's persistent memory. Fetched once per turn,
   202→  // appended to the BASE prompt so both the live system prompt and
   203→  // the hour-stable cache-key prompt carry it (editing memory
   204→  // naturally invalidates stale cached answers). No memories →
   205→  // empty block → byte-identical prompts → old cache entries still hit.
   206→  const memBlock = memoryBlock(await getMemories());
   207→  const baseSystemPrompt =
   208→    (params.search && sources.length ? SEARCH_SYSTEM_DIRECTIVE : BASE_SYSTEM_PROMPT) +
   209→    (effortAddon ? `\n\n${effortAddon}` : "") +
   210→    (styleAddon ? `\n\n${styleAddon}` : "") +
   211→    (memBlock ? `\n\n${memBlock}` : "");
   212→  // The model always knows the real "now" (user's timezone).
   213→  const systemPrompt = `${baseSystemPrompt}\n\n${liveContextBlock()}`;
   214→
   215→  const finalUserContent =
   216→    params.content +
   217→    attachmentContextBlock(attachments) +
   218→    (sources.length ? searchContextBlock(sources) : "");
   219→
   220→  // ── 2. Exact-match cache (prompt + model + hour-stable system prompt) ──
   221→  // The clock block is hour-stable here so repeat questions within the
   222→  // same hour still hit the cache, while the model sees second-level time.
   223→  const cacheKey = computeCacheKey(
   224→    finalUserContent,
   225→    model,
   226→    `${baseSystemPrompt}\n\n${liveContextBlockHourStable()}`
   227→  );
   228→  const cached = await getCached(cacheKey);
   229→
   230→  // ── 3. Persistence: chat row + user message ──
   231→  let chatId = params.chatId ?? null;
   232→  let userMessageId: string | null = null;
   233→  let title: string | null = null;
   234→
   235→  if (!params.temporary && params.persistUser !== false) {
   236→    if (!chatId) {
   237→      title = deriveTitle(params.content);
   238→      const chat = await db.chat.create({
   239→        data: {
   240→          title,
   241→          model,
   242→          provider: provider.name,
   243→          isTemporary: false,
   244→        },
   245→      });
   246→      chatId = chat.id;
   247→    }
   248→    const userMsg = await db.message.create({
   249→      data: {
   250→        chatId,
   251→        role: "user",
   252→        content: params.content,
   253→        sources: sources.length ? JSON.stringify(sources) : null,
   254→        attachments: attachments.length ? JSON.stringify(attachments) : null,
   255→      },
   256→    });
   257→    userMessageId = userMsg.id;
   258→  } else if (!params.temporary && params.chatId) {
   259→    chatId = params.chatId;
   260→  }
   261→
   262→  emit({
   263→    type: "meta",
   264→    chatId,
   265→    userMessageId,
   266→    title,
   267→    provider: provider.name,
   268→    model,
   269→    effort,
   270→  });
   271→
   272→  // ── 4a. Cache hit → 0 tokens spent ──
   273→  if (cached) {
   274→    // Contract enforcement applies to cached search answers too
   275→    // (older cache entries may predate the confidence-label rule).
   276→    if (sources.length && !confidenceOf(cached.content)) {
   277→      cached.content = await appendConfidenceLabel(cached.content);
   278→    }
   279→    // Cached answers type out live too — the answer never appears
   280→    // in a single shot, even when it cost 0 tokens.
   281→    await revealWords(
   282→      cached.content,
   283→      (text) => emit({ type: "delta", text }),
   284→      6,
   285→      () => params.signal?.aborted === true
   286→    );
   287→    let assistantMessageId: string | null = null;
   288→    if (!params.temporary && chatId) {
   289→      const msg = await db.message.create({
   290→        data: {
   291→          chatId,
   292→          role: "assistant",
   293→          content: cached.content,
   294→          model,
   295→          provider: provider.name,
   296→          keyLabel: "cache",
   297→          tokens: 0,
   298→          cached: true,
   299→        },
   300→      });
   301→      assistantMessageId = msg.id;
   302→      await db.chat.update({
   303→        where: { id: chatId },
   304→        data: { updatedAt: new Date(), model, provider: provider.name },
   305→      });
   306→    }
   307→    // Audit trail: cached hit, zero tokens.
   308→    await db.usageEvent.create({
   309→      data: {
   310→        keyId: null,
   311→        providerId: provider.id,
   312→        chatId,
   313→        model,
   314→        promptTokens: 0,
   315→        completionTokens: 0,
   316→        totalTokens: 0,
   317→        multiplier: provider.tokenMultiplier,
   318→        cached: true,
   319→      },
   320→    });
   321→    const done = {
   322→      type: "done",
   323→      chatId,
   324→      // Run 34 correctness fix: the cache-hit path persisted the
   325→      // user message (step 3) but never returned its id — callers
   326→      // reading TurnResult.userMessageId (the Ritual scheduler
   327→      // stamps its origin on that row) got undefined on every
   328→      // cached turn. The field was always in the contract; it is
   329→      // now actually carried.
   330→      userMessageId,
   331→      assistantMessageId,
   332→      content: cached.content,
   333→      cached: true,
   334→      tokens: 0,
   335→      rawTokens: 0,
   336→      // Cache hits bill zero tokens — nothing was estimated.
   337→      tokensEstimated: false,
   338→      servedBy: "cache" as const,
   339→      keyLabel: "cache",
   340→      providerName: provider.name,
   341→      model,
   342→      effort,
   343→      sources,
   344→      searchFailed: params.search && sources.length === 0,
   345→    };
   346→    emit(done);
   347→    return done;
   348→  }
   349→
   350→  // ── 4b + 5. Health-aware key dispatch, then cloud completion ──
   351→  // The dispatcher round-robins pooled keys, quarantines dead ones
   352→  // (auth failures), retries the next healthy key within the same
   353→  // turn, and only fails over to the Z.ai cloud when every key
   354→  // failed. Hard pool errors (no keys / limiter exhaustion) keep
   355→  // their original coded messages.
   356→  //
   357→  // `signal` is the CLIENT's connection: the moment it drops, the
   358→  // upstream fetch is aborted at the source — the pooled keys stop
   359→  // paying for tokens nobody will ever see.
   360→  const messages: ChatMessageData[] = [
   361→    { role: "system", content: systemPrompt },
   362→    ...expandHistory(params.history).slice(-20),
   363→    { role: "user", content: finalUserContent },
   364→  ];
   365→
   366→  const routed = await dispatchCompletion({
   367→    provider,
   368→    model,
   369→    messages,
   370→    onDelta: (text) => emit({ type: "delta", text }),
   371→    // Mid-stream provider death → the cloud re-emits from scratch;
   372→    // "restart" tells the client to clear the partial text first.
   373→    onRestart: () => emit({ type: "restart" }),
   374→    // Deep effort → the cloud failover THINKS before answering.
   375→    thinking: effort === "deep",
   376→    signal: params.signal,
   377→  });
   378→  const exec = routed.exec;
   379→  // The key that actually served — null when the Z.ai cloud did.
   380→  const key = routed.key;
   381→  // True when the CLIENT disconnected mid-generation: the partial
   382→  // text is final for this turn.
   383→  const aborted = routed.aborted === true;
   384→
   385→  // ── Grounded-answer contract enforcement ──
   386→  // On search turns the confidence label is mandatory. The directive
   387→  // asks for it, but model compliance is probabilistic — when the
   388→  // answer lacks the label, one cheap corrective pass appends it so
   389→  // the contract holds deterministically (cache fills afterwards,
   390→  // so corrected answers are reused corrected). Aborted turns skip
   391→  // it: no corrective cloud spend on an answer nobody is watching.
   392→  if (!aborted && sources.length && !confidenceOf(exec.content)) {
   393→    exec.content = await appendConfidenceLabel(exec.content);
   394→  }
   395→
   396→  // ── 6. Usage accounting (multiplier applies to effective totals) ──
   397→  // Only the key that ACTUALLY served the response is billed — a
   398→  // dead key that failed over to the cloud never gets charged.
   399→  const rawTotal =
   400→    exec.totalTokens ||
   401→    exec.promptTokens + exec.completionTokens ||
   402→    estimateTokens(finalUserContent) + estimateTokens(exec.content);
   403→  // Honesty marker: without upstream usage stats the counts are
   404→  // chars/4 estimates — the UI must never present them as exact.
   405→  const tokensEstimated = exec.usageReported !== true && rawTotal > 0;
   406→
   407→  await db.usageEvent.create({
   408→    data: {
   409→      keyId: key?.id ?? null,
   410→      providerId: provider.id,
   411→      chatId,
   412→      model,
   413→      promptTokens: exec.promptTokens,
   414→      completionTokens: exec.completionTokens,
   415→      totalTokens: rawTotal,
   416→      multiplier: provider.tokenMultiplier,
   417→      cached: false,
   418→    },
   419→  });
   420→
   421→  // Roll the serving key's monthly accumulator forward (monthKey
   422→  // flip safe). Skipped entirely on cloud failover — no key served.
   423→  if (key) {
   424→    const now = new Date();
   425→    const mk = monthKey(now);
   426→    const monthAgg = await db.usageEvent.aggregate({
   427→      _sum: { totalTokens: true },
   428→      where: {
   429→        keyId: key.id,
   430→        createdAt: { gte: new Date(now.getFullYear(), now.getMonth(), 1) },
   431→      },
   432→    });
   433→    await db.apiKey.update({
   434→      where: { id: key.id },
   435→      data: {
   436→        lastUsedAt: now,
   437→        monthKey: mk,
   438→        tokensUsedThisMonth: monthAgg._sum.totalTokens ?? rawTotal,
   439→      },
   440→    });
   441→  }
   442→
   443→  // ── 7. Cache fill (24h TTL) ──
   444→  // A turn the user ABORTED never fills the cache: a truncated
   445→  // answer must never be served to a future identical question.
   446→  if (!aborted) {
   447→    await putCached({
   448→      cacheKey,
   449→      model,
   450→      provider: provider.name,
   451→      content: exec.content,
   452→      tokens: rawTotal,
   453→    });
   454→  }
   455→
   456→  // ── 8. Persist assistant message ──
   457→  // keyLabel is only recorded when a pooled key actually served the
   458→  // response — a cloud-failover answer never claims a key it didn't use.
   459→  const effectiveTokens = Math.round(rawTotal * provider.tokenMultiplier);
   460→  const servedKeyLabel = key?.label ?? null;
   461→  let assistantMessageId: string | null = null;
   462→  // The partial of an aborted turn is persisted too — the timeline
   463→  // keeps exactly what streamed before the user pressed stop (and
   464→  // an empty partial creates no ghost message).
   465→  if (!params.temporary && chatId && exec.content.trim()) {
   466→    const msg = await db.message.create({
   467→      data: {
   468→        chatId,
   469→        role: "assistant",
   470→        content: exec.content,
   471→        model,
   472→        provider: provider.name,
   473→        keyLabel: servedKeyLabel,
   474→        tokens: effectiveTokens,
   475→        tokensEstimated,
   476→        cached: false,
   477→        effort,
   478→      },
   479→    });
   480→    assistantMessageId = msg.id;
   481→    await db.chat.update({
   482→      where: { id: chatId },
   483→      data: { updatedAt: new Date(), model, provider: provider.name },
   484→    });
   485→  }
   486→
   487→  const done: TurnResult = {
   488→    chatId,
   489→    userMessageId,
   490→    assistantMessageId,
   491→    title,
   492→    sources,
   493→    content: exec.content,
   494→    cached: false,
   495→    tokens: effectiveTokens,
   496→    rawTokens: rawTotal,
   497→    servedBy: exec.servedBy,
   498→    keyLabel: servedKeyLabel,
   499→    providerName: provider.name,
   500→    model,
   501→    effort,
   502→    tokensEstimated,
   503→    searchFailed: params.search && sources.length === 0,
   504→    ...(aborted ? { aborted: true } : {}),
   505→  };
   506→  emit({ type: "done", ...done });
   507→  if (aborted) {
   508→    console.log(
   509→      `[gateway] turn cancelled by client — persisted ${exec.content.length} chars, ` +
   510→        `${effectiveTokens} tokens billed, cache skipped`
   511→    );
   512→  }
   513→  return done;
   514→}

   516→// ── Duel turn ─────────────────────────────────────────────────
   517→// TWO models race in parallel on one question: shared web search,
   518→// shared chat + user-message persistence, per-lane cache lookup,
   519→// health-aware dispatch, grounded-contract enforcement, usage
   520→// accounting and cache fill. Both lanes are billed honestly for
   521→// what they actually served. The WINNER is not persisted here —
   522→// the client calls POST /api/chat/duel/pick once the user chooses;
   523→// that endpoint stores the winner as the assistant message and the
   524→// loser as its runner-up reference.
   525→
   526→export interface DuelLaneResult {
   527→  lane: "a" | "b";
   528→  model: string;
   529→  content: string;
   530→  tokens: number; // effective (multiplied) tokens billed — 0 when cached
   531→  rawTokens: number;
   532→  cached: boolean;
   533→  servedBy: "provider" | "zai-cloud" | "cache" | null; // null on error
   534→  keyLabel: string | null;
   535→  error: string | null;
   536→}
   537→
   538→export interface DuelParams {
   539→  chatId?: string | null;
   540→  history: ChatMessageData[];
   541→  content: string;
   542→  model: string; // lane A — the user's selected model
   543→  challenger: string; // lane B
   544→  search: boolean;
   545→  temporary: boolean;
   546→  persistUser?: boolean;
   547→  /** Effort affects the duel's SHARED retrieval (deep → multi-round
   548→   *  search + thinking on the cloud path) but never the lane models
   549→   *  — a duel is explicitly about the two engines the user chose. */
   550→  effort?: EffortLevel;
   551→  /** Files the gateway read — shared by both lanes (the same
   552→   *  material, the same rules as a normal turn). */
   553→  attachments?: AttachmentRecord[];
   554→  /** Client disconnect cancels BOTH racing lanes at the source. */
   555→  signal?: AbortSignal;
   556→  emit: (ev: TurnEvent) => void;
   557→}
   558→
   559→export interface DuelTurnResult {
   560→  chatId: string | null;
   561→  userMessageId: string | null;
   562→  title: string | null;
   563→  sources: SearchSource[];
   564→  lanes: { a: DuelLaneResult; b: DuelLaneResult };
   565→  bothFailed: boolean;
   566→}
   567→
   568→export async function runDuelTurn(
   569→  params: DuelParams
   570→): Promise<DuelTurnResult> {
   571→  const { emit } = params;
   572→  const provider = await getActiveProvider();
   573→  const providerModels = JSON.parse(provider.models || "[]") as string[];
   574→  const modelA = params.model || providerModels[0] || "free/glm-5.3";
   575→  const modelB = params.challenger;
   576→  const effort: EffortLevel = params.effort ?? "medium";
   577→  const attachments = sanitizeAttachments(params.attachments);
   578→
   579→  if (!modelB || modelB === modelA) {
   580→    throw new Error("Duel needs two different models");
   581→  }
   582→  if (
   583→    providerModels.length &&
   584→    (!providerModels.includes(modelA) || !providerModels.includes(modelB))
   585→  ) {
   586→    throw new Error("Both duel models must be available on the active provider");
   587→  }
   588→
   589→  // ── 1. Shared web search — both lanes reason from the SAME retrieval ──
   590→  let sources: SearchSource[] = [];
   591→  if (params.search) {
   592→    if (effort === "deep") {
   593→      emit({ type: "search_status", status: "searching", depth: "deep" });
   594→      const outcome = await deepSearchDetailed(params.content, (detail) =>
   595→        emit({ type: "search_status", status: "searching", detail, depth: "deep" })
   596→      );
   597→      sources = outcome.results;
   598→    } else {
   599→      emit({ type: "search_status", status: "searching" });
   600→      const outcome = await webSearchDetailed(params.content);
   601→      sources = outcome.results;
   602→    }
   603→    emit({ type: "sources", sources });
   604→    emit({
   605→      type: "search_status",
   606→      status: sources.length ? "ok" : "failed",
   607→      depth: effort === "deep" ? "deep" : undefined,
   608→    });
   609→  }
   610→
   611→  // Run 28 — duels run with the same persistent memory as normal
   612→  // turns (appended to the base prompt → both the live prompt and
   613→  // each lane's cache key include it; empty memory = byte-identical).
   614→  const memBlock = memoryBlock(await getMemories());
   615→  const baseSystemPrompt =
   616→    (params.search && sources.length ? SEARCH_SYSTEM_DIRECTIVE : BASE_SYSTEM_PROMPT) +
   617→    (memBlock ? `\n\n${memBlock}` : "");
   618→  const systemPrompt = `${baseSystemPrompt}\n\n${liveContextBlock()}`;
   619→  const finalUserContent =
   620→    params.content +
   621→    attachmentContextBlock(attachments) +
   622→    (sources.length ? searchContextBlock(sources) : "");
   623→
   624→  // ── 2. Shared persistence: chat row + user message (same as a
   625→  //      normal turn — the winner is appended later, at pick time) ──
   626→  let chatId = params.chatId ?? null;
   627→  let userMessageId: string | null = null;
   628→  let title: string | null = null;
   629→
   630→  if (!params.temporary && params.persistUser !== false) {
   631→    if (!chatId) {
   632→      title = deriveTitle(params.content);
   633→      const chat = await db.chat.create({
   634→        data: {
   635→          title,
   636→          model: modelA,
   637→          provider: provider.name,
   638→          isTemporary: false,
   639→        },
   640→      });
   641→      chatId = chat.id;
   642→    }
   643→    const userMsg = await db.message.create({
   644→      data: {
   645→        chatId,
   646→        role: "user",
   647→        content: params.content,
   648→        sources: sources.length ? JSON.stringify(sources) : null,
   649→        attachments: attachments.length ? JSON.stringify(attachments) : null,
   650→      },
   651→    });
   652→    userMessageId = userMsg.id;
   653→  } else if (!params.temporary && params.chatId) {
   654→    chatId = params.chatId;
   655→  }
   656→
   657→  emit({
   658→    type: "meta",
   659→    chatId,
   660→    userMessageId,
   661→    title,
   662→    provider: provider.name,
   663→    model: modelA,
   664→    effort,
   665→  });
   666→  emit({ type: "duel_start", a: modelA, b: modelB });
   667→
   668→  const messages: ChatMessageData[] = [
   669→    { role: "system", content: systemPrompt },
   670→    ...expandHistory(params.history).slice(-20),
   671→    { role: "user", content: finalUserContent },
   672→  ];
   673→
   674→  // ── 3. The race: two independent lanes over the shared context ──
   675→  const runLane = async (
   676→    lane: "a" | "b",
   677→    model: string
   678→  ): Promise<DuelLaneResult> => {
   679→    // Per-lane exact-match cache (the cache key includes the model).
   680→    const cacheKey = computeCacheKey(
   681→      finalUserContent,
   682→      model,
   683→      `${baseSystemPrompt}\n\n${liveContextBlockHourStable()}`
   684→    );
   685→    const cached = await getCached(cacheKey);
   686→    if (cached) {
   687→      // Contract enforcement applies to cached search answers too.
   688→      if (sources.length && !confidenceOf(cached.content)) {
   689→        cached.content = await appendConfidenceLabel(cached.content);
   690→      }
   691→      // Cached lane answers type out live too (word by word),
   692→      // exactly like a fresh lane.
   693→      await revealWords(
   694→        cached.content,
   695→        (text) => emit({ type: "delta", lane, text }),
   696→        6,
   697→        () => params.signal?.aborted === true
   698→      );
   699→      await db.usageEvent.create({
   700→        data: {
   701→          keyId: null,
   702→          providerId: provider.id,
   703→          chatId,
   704→          model,
   705→          promptTokens: 0,
   706→          completionTokens: 0,
   707→          totalTokens: 0,
   708→          multiplier: provider.tokenMultiplier,
   709→          cached: true,
   710→        },
   711→      });
   712→      const result: DuelLaneResult = {
   713→        lane,
   714→        model,
   715→        content: cached.content,
   716→        tokens: 0,
   717→        rawTokens: 0,
   718→        cached: true,
   719→        servedBy: "cache",
   720→        keyLabel: "cache",
   721→        error: null,
   722→      };
   723→      emit({ type: "lane_done", ...result });
   724→      return result;
   725→    }
   726→
   727→    try {
   728→      const routed = await dispatchCompletion({
   729→        provider,
   730→        model,
   731→        messages,
   732→        onDelta: (text) => emit({ type: "delta", lane, text }),
   733→        // Mid-stream death in THIS lane → the cloud re-emits this
   734→        // lane from scratch; clear only this lane's partial text.
   735→        onRestart: () => emit({ type: "restart", lane }),
   736→        // Deep effort → both lanes' cloud failover thinks too.
   737→        thinking: effort === "deep",
   738→        // Client disconnect cancels this lane's upstream too.
   739→        signal: params.signal,
   740→      });
   741→      const exec = routed.exec;
   742→      const key = routed.key;
   743→      const aborted = routed.aborted === true;
   744→
   745→      if (!aborted && sources.length && !confidenceOf(exec.content)) {
   746→        exec.content = await appendConfidenceLabel(exec.content);
   747→      }
   748→
   749→      // Per-lane billing — identical rules to a normal turn: only
   750→      // the key that actually served is charged; cloud lanes bill
   751→      // no key.
   752→      const rawTotal =
   753→        exec.totalTokens ||
   754→        exec.promptTokens + exec.completionTokens ||
   755→        estimateTokens(finalUserContent) + estimateTokens(exec.content);
   756→      await db.usageEvent.create({
   757→        data: {
   758→          keyId: key?.id ?? null,
   759→          providerId: provider.id,
   760→          chatId,
   761→          model,
   762→          promptTokens: exec.promptTokens,
   763→          completionTokens: exec.completionTokens,
   764→          totalTokens: rawTotal,
   765→          multiplier: provider.tokenMultiplier,
   766→          cached: false,
   767→        },
   768→      });
   769→      if (key) {
   770→        const now = new Date();
   771→        const mk = monthKey(now);
   772→        const monthAgg = await db.usageEvent.aggregate({
   773→          _sum: { totalTokens: true },
   774→          where: {
   775→            keyId: key.id,
   776→            createdAt: { gte: new Date(now.getFullYear(), now.getMonth(), 1) },
   777→          },
   778→        });
   779→        await db.apiKey.update({
   780→          where: { id: key.id },
   781→          data: {
   782→            lastUsedAt: now,
   783→            monthKey: mk,
   784→            tokensUsedThisMonth: monthAgg._sum.totalTokens ?? rawTotal,
   785→          },
   786→        });
   787→      }
   788→
   789→      // An aborted lane never fills the cache (truncated text must
   790→      // never be served to a future identical question).
   791→      if (!aborted) {
   792→        await putCached({
   793→          cacheKey,
   794→          model,
   795→          provider: provider.name,
   796→          content: exec.content,
   797→          tokens: rawTotal,
   798→        });
   799→      }
   800→
   801→      const result: DuelLaneResult = {
   802→        lane,
   803→        model,
   804→        content: exec.content,
   805→        tokens: Math.round(rawTotal * provider.tokenMultiplier),
   806→        rawTokens: rawTotal,
   807→        cached: false,
   808→        servedBy: exec.servedBy,
   809→        keyLabel: key?.label ?? null,
   810→        error: null,
   811→      };
   812→      emit({ type: "lane_done", ...result });
   813→      return result;
   814→    } catch (e) {
   815→      // One lane failing must never kill the other — the duel
   816→      // degrades to a single-lane answer the user can still keep.
   817→      const result: DuelLaneResult = {
   818→        lane,
   819→        model,
   820→        content: "",
   821→        tokens: 0,
   822→        rawTokens: 0,
   823→        cached: false,
   824→        servedBy: null,
   825→        keyLabel: null,
   826→        error: (e as Error).message || "Generation failed",
   827→      };
   828→      emit({ type: "lane_done", ...result });
   829→      return result;
   830→    }
   831→  };
   832→
   833→  const [a, b] = await Promise.all([
   834→    runLane("a", modelA),
   835→    runLane("b", modelB),
   836→  ]);
   837→
   838→  if (a.error && b.error) {
   839→    emit({
   840→      type: "error",
   841→      message: `Both models failed — A (${modelA}): ${a.error} · B (${modelB}): ${b.error}`,
   842→    });
   843→    return { chatId, userMessageId, title, sources, lanes: { a, b }, bothFailed: true };
   844→  }
   845→
   846→  // No assistant message is persisted here — the client picks the
   847→  // winner and /api/chat/duel/pick stores it (with the runner-up).
   848→  emit({
   849→    type: "done",
   850→    chatId,
   851→    userMessageId,
   852→    assistantMessageId: null,
   853→    duel: { a, b },
   854→    // Top-level fields mirror lane A for simple/legacy consumers.
   855→    content: a.content,
   856→    cached: a.cached,
   857→    tokens: a.tokens,
   858→    rawTokens: a.rawTokens,
   859→    servedBy: a.servedBy,
   860→    keyLabel: a.keyLabel,
   861→    providerName: provider.name,
   862→    model: modelA,
   863→    sources,
   864→    searchFailed: params.search && sources.length === 0,
   865→  });
   866→  return { chatId, userMessageId, title, sources, lanes: { a, b }, bothFailed: false };
   867→}
