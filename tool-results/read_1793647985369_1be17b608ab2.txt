    74→/** The /-token under the caret, if any. The "/" must sit at the very
    75→ *  start of the input or right after whitespace — typing "/" mid-word
    76→ *  ("a/b") or inside a URL ("https://x") never triggers the popover. */
    77→function activeSlashToken(
    78→  value: string,
    79→  caret: number
    80→): { query: string; start: number } | null {
    81→  const upto = value.slice(0, caret);
    82→  const m = /(^|\s)\/([\w-]{0,20})$/.exec(upto);
    83→  if (!m) return null;
    84→  return { query: m[2].toLowerCase(), start: caret - m[2].length - 1 };
    85→}

    87→function formatChars(n: number): string {
    88→  if (n >= 1000) return `${(n / 1000).toFixed(1)}k chars`;
    89→  return `${n} chars`;
    90→}
