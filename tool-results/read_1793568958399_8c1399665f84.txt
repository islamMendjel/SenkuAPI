   381→  async sendMessage(content) {
   382→    const state = get();
   383→    if (state.streaming || !content.trim()) return;
   384→    // A duel still awaiting its pick? The primary lane becomes the
   385→    // canonical answer so the next turn's history has one voice.
   386→    const pending = [...state.messages].reverse().find((m) => m.duel && !m.duelPicked);
   387→    if (pending) {
   388→      await get().pickDuelWinner(pending.id, "a");
   389→    }
   390→    const s2 = get();
   391→    const history: ChatMessageData[] = s2.messages.map((m) => ({
   392→      role: m.role,
   393→      content: m.content,
   394→    }));
   395→    // Resolve the challenger at send time — the model list may have
   396→    // changed since duel was armed.
   397→    let challenger = s2.challenger;
   398→    if (challenger === s2.model || !s2.models.includes(challenger ?? "")) {
   399→      challenger = autoChallenger(s2.models, s2.model);
   400→    }
   401→    const duel =
   402→      s2.duelEnabled && challenger ? { challenger } : undefined;
   403→    await streamTurn(set, get, {
   404→      chatId: s2.temporary ? null : s2.currentChatId,
   405→      history,
   406→      content,
   407→      persistUser: true,
   408→      duel,
   409→    });
   410→  },

   412→  async editMessage(messageId, content) {
   413→    const state = get();
   414→    if (state.streaming || !content.trim()) return;
   415→    const idx = state.messages.findIndex((m) => m.id === messageId);
   416→    if (idx === -1) return;
   417→    if (state.temporary || !state.currentChatId) {
   418→      // Local-only timeline: truncate everything after the edited message.
   419→      const kept = state.messages
   420→        .slice(0, idx + 1)
   421→        .map((m) => (m.id === messageId ? { ...m, content: content.trim() } : m));
   422→      set({ messages: kept });
   423→      const history: ChatMessageData[] = kept.slice(0, -1).map((m) => ({
   424→        role: m.role,
   425→        content: m.content,
   426→      }));
   427→      await streamTurn(set, get, {
   428→        chatId: null,
   429→        history,
   430→        content: content.trim(),
   431→        persistUser: false,
   432→        temporary: true,
   433→      });
   434→      return;
   435→    }
   436→
   437→    const res = await fetch("/api/chat/truncate", {
   438→      method: "POST",
   439→      headers: { "Content-Type": "application/json" },
   440→      body: JSON.stringify({ messageId, content: content.trim() }),
   441→    });
   442→    const data = await res.json();
   443→    if (!Array.isArray(data.messages)) return;
   444→    const kept: UIMessage[] = data.messages.map((m: UIMessage) => ({
   445→      ...m,
   446→      sources: m.sources ?? null,
   447→    }));
   448→    set({ messages: kept });
   449→    const history: ChatMessageData[] = kept.slice(0, -1).map((m) => ({
   450→      role: m.role,
   451→      content: m.content,
   452→    }));
   453→    await streamTurn(set, get, {
   454→      chatId: state.currentChatId,
   455→      history,
   456→      content: content.trim(),
   457→      persistUser: false,
   458→    });
   459→  },

   461→  async retryLast(modelOverride) {
   462→    const state = get();
   463→    if (state.streaming) return;
   464→
   465→    // The most recent failed assistant reply.
   466→    let failedIdx = -1;
   467→    for (let i = state.messages.length - 1; i >= 0; i--) {
   468→      const m = state.messages[i];
   469→      if (m.role === "assistant" && m.error) {
   470→        failedIdx = i;
   471→        break;
   472→      }
   473→    }
   474→    if (failedIdx === -1) return;
   475→
   476→    // The user message that failed turn was answering.
   477→    let userIdx = -1;
   478→    for (let i = failedIdx - 1; i >= 0; i--) {
   479→      if (state.messages[i].role === "user") {
   480→        userIdx = i;
   481→        break;
   482→      }
   483→    }
   484→    if (userIdx === -1) return;
   485→    const userMsg = state.messages[userIdx];
   486→
   487→    const history: ChatMessageData[] = state.messages
   488→      .slice(0, userIdx)
   489→      .map((m) => ({ role: m.role, content: m.content }));
   490→
   491→    // Drop the failed bubble from the timeline.
   492→    set({ messages: state.messages.slice(0, failedIdx) });
   493→
   494→    // If the user message never received a server id, the turn died
   495→    // before persistence — persist it on retry. Otherwise it is
   496→    // already in the DB and must not be duplicated.
   497→    const persisted = !userMsg.id.startsWith("u-");
   498→    await streamTurn(set, get, {
   499→      chatId: state.temporary ? null : state.currentChatId,
   500→      history,
   501→      content: userMsg.content,
   502→      persistUser: !persisted,
   503→      appendUserBubble: false, // the original user bubble stays
   504→      temporary: state.temporary,
   505→      model: modelOverride,
   506→    });
   507→  },
   508→
   509→  async pickDuelWinner(messageId, laneId) {
   510→    const s = get();
   511→    const msg = s.messages.find((m) => m.id === messageId);
   512→    if (!msg?.duel || msg.duelPicked) return;
   513→    const winner = msg.duel[laneId];
   514→    const loser = msg.duel[laneId === "a" ? "b" : "a"];
   515→    if (!winner.content.trim()) return; // nothing to keep
   516→    const providerName = s.limiter?.providerName ?? null;
   517→
   518→    // Local commit first — the pick must feel instant.
   519→    set({
   520→      messages: s.messages.map((m) =>
   521→        m.id === messageId
   522→          ? {
   523→              ...m,
   524→              content: winner.content,
   525→              model: winner.model,
   526→              tokens: winner.tokens,
   527→              cached: winner.cached,
   528→              servedBy: winner.servedBy,
   529→              keyLabel: winner.keyLabel,
   530→              provider: providerName,
   531→              duel: null,
   532→              duelPicked: laneId,
   533→              runnerUp: loser.content.trim()
   534→                ? { content: loser.content, model: loser.model }
   535→                : null,
   536→              streaming: false,
   537→              phase: null,
   538→            }
   539→          : m
   540→      ),
   541→    });
   542→
   543→    // Persist for saved chats (temporary duels stay local-only).
   544→    if (!s.temporary && s.currentChatId) {
   545→      try {
   546→        const res = await fetch("/api/chat/duel/pick", {
   547→          method: "POST",
   548→          headers: { "Content-Type": "application/json" },
   549→          body: JSON.stringify({
   550→            chatId: s.currentChatId,
   551→            winner: {
   552→              content: winner.content,
   553→              model: winner.model,
   554→              tokens: winner.tokens,
   555→              cached: winner.cached,
   556→              servedBy: winner.servedBy,
   557→              keyLabel: winner.keyLabel,
   558→            },
   559→            runnerUp: loser.content.trim()
   560→              ? { content: loser.content, model: loser.model }
   561→              : null,
   562→          }),
   563→        });
   564→        const data = (await res.json()) as { id?: string; error?: string };
   565→        if (typeof data.id === "string" && data.id) {
   566→          // Reconcile with the server id so later edits/truncates hit
   567→          // the real message row.
   568→          set((st) => ({
   569→            messages: st.messages.map((m) =>
   570→              m.id === messageId ? { ...m, id: data.id as string } : m
   571→            ),
   572→          }));
   573→        }
   574→      } catch {
   575→        // keep the local pick — the answer stays in the timeline
   576→      }
   577→    }
   578→  },
   579→
   580→  stopStreaming: () => {
   581→    streamController?.abort();
   582→    streamController = null;
   583→  },
   584→}));

   592→interface TurnOptions {
   593→  chatId: string | null;
   594→  history: ChatMessageData[];
   595→  content: string;
   596→  persistUser: boolean;
   597→  temporary?: boolean;
   598→  /** Overrides the store's active model (error-card "switch model"). */
   599→  model?: string;
   600→  /** Force whether a new local user bubble is appended. Defaults to
   601→      the persistUser flag — retries keep the original bubble but may
   602→      still need server-side persistence, so the two are decoupled. */
   603→  appendUserBubble?: boolean;
   604→  /** Duel: race the active model against `duel.challenger`. */
   605→  duel?: { challenger: string };
   606→}
   607→
   608→async function streamTurn(
   609→  set: SetState,
   610→  get: GetState,
   611→  opts: TurnOptions
   612→): Promise<void> {
   613→  const state = get();
   614→  const temporary = opts.temporary ?? state.temporary;
   615→  const model = opts.model ?? state.model;
   616→  const userId = `u-${Date.now()}`;
   617→  const assistantId = `a-${Date.now()}`;
   618→  const search = state.searchEnabled;
   619→  const duel = opts.duel;
   620→  const makeLane = (id: "a" | "b", laneModel: string): DuelLane => ({
   621→    id,
   622→    model: laneModel,
   623→    content: "",
   624→    streaming: true,
   625→    error: null,
   626→    phase: search ? ("searching" as const) : ("thinking" as const),
   627→    tokens: 0,
   628→    cached: false,
   629→    servedBy: null,
   630→    keyLabel: null,
   631→  });
   632→  // Regenerate-after-edit turns keep the (already updated) user message
   633→  // in the timeline — only fresh sends append a new local user bubble.
   634→  // Retries decouple the two: the bubble stays, persistence is decided
   635→  // by whether the failed turn ever reached the server.
   636→  const appendUserBubble = opts.appendUserBubble ?? (opts.persistUser !== false);
   637→  // Set when this turn creates a brand-new persisted chat — the
   638→  // raw derived title then gets refined by a cheap model pass.
   639→  let createdChatId: string | null = null;
   640→
   641→  set({
   642→    streaming: true,
   643→    messages: [
   644→      ...state.messages,
   645→      ...(appendUserBubble
   646→        ? [
   647→            {
   648→              id: userId,
   649→              role: "user" as const,
   650→              content: opts.content,
   651→              sources: search ? [] : null,
   652→            },
   653→          ]
   654→        : []),
   655→      {
   656→        id: assistantId,
   657→        role: "assistant" as const,
   658→        content: "",
   659→        streaming: true,
   660→        model,
   661→        provider: state.limiter?.providerName ?? null,
   662→        phase: search ? "searching" : "thinking",
   663→        ...(duel
   664→          ? {
   665→              duel: {
   666→                a: makeLane("a", model),
   667→                b: makeLane("b", duel.challenger),
   668→              },
   669→              duelPicked: null,
   670→            }
   671→          : {}),
   672→      },
   673→    ],
   674→  });
   675→
   676→  const updateAssistant = (patch: Partial<UIMessage>) => {
   677→    set((s) => ({
   678→      messages: s.messages.map((m) =>
   679→        m.id === assistantId ? { ...m, ...patch } : m
   680→      ),
   681→    }));
   682→  };
   683→  // Duel-only: patch one racing lane of the assistant message.
   684→  const updateLane = (lane: "a" | "b", patch: Partial<DuelLane>) => {
   685→    set((s) => ({
   686→      messages: s.messages.map((m) =>
   687→        m.id === assistantId && m.duel
   688→          ? { ...m, duel: { ...m.duel, [lane]: { ...m.duel[lane], ...patch } } }
   689→          : m
   690→      ),
   691→    }));
   692→  };
   693→  const updateUser = (patch: Partial<UIMessage>) => {
   694→    set((s) => {
   695→      const idx = appendUserBubble
   696→        ? s.messages.findIndex((m) => m.id === currentUserMsgId)
   697→        : (() => {
   698→            for (let i = s.messages.length - 1; i >= 0; i--) {
   699→              if (s.messages[i].role === "user") return i;
   700→            }
   701→            return -1;
   702→          })();
   703→      if (idx === -1) return {};
   704→      return {
   705→        messages: s.messages.map((m, i) => (i === idx ? { ...m, ...patch } : m)),
   706→      };
   707→    });
   708→  };
   709→
   710→  const controller = new AbortController();
   711→  streamController = controller;
   712→  // Reconciled with the server-assigned id as soon as "meta" arrives so
   713→  // later edits hit /api/chat/truncate with the real message id.
   714→  let currentUserMsgId = userId;
   715→
   716→  try {
   717→    const res = await fetch("/api/chat/send", {
   718→      method: "POST",
   719→      headers: { "Content-Type": "application/json" },
   720→      signal: controller.signal,
   721→      body: JSON.stringify({
   722→        chatId: opts.chatId,
   723→        history: opts.history,
   724→        content: opts.content,
   725→        model,
   726→        search,
   727→        temporary,
   728→        persistUser: opts.persistUser,
   729→        ...(duel ? { duel: true, challengerModel: duel.challenger } : {}),
   730→      }),
   731→    });
   732→
   733→    if (!res.ok || !res.body) {
   734→      const errText = await res.text().catch(() => "");
   735→      throw new Error(errText || `Request failed (${res.status})`);
   736→    }
   737→
   738→    const reader = res.body.getReader();
   739→    const decoder = new TextDecoder();
   740→    let buffer = "";
   741→    let assistantContent = "";
   742→    // Duel lanes accumulate independently; untagged deltas (normal
   743→    // turns) fall through to assistantContent only.
   744→    const laneContents: Record<"a" | "b", string> = { a: "", b: "" };
   745→
   746→    while (true) {
   747→      const { done, value } = await reader.read();
   748→      if (done) break;
   749→      buffer += decoder.decode(value, { stream: true });
   750→      const parts = buffer.split("\n\n");
   751→      buffer = parts.pop() ?? "";
   752→      for (const part of parts) {
   753→        const line = part.trim();
   754→        if (!line.startsWith("data:")) continue;
   755→        let ev: Record<string, unknown>;
   756→        try {
   757→          ev = JSON.parse(line.slice(5).trim());
   758→        } catch {
   759→          continue;
   760→        }
   761→        if (ev.type === "meta") {
   762→          if (!temporary && typeof ev.chatId === "string" && ev.chatId) {
   763→            set({ currentChatId: ev.chatId });
   764→            if (opts.chatId == null) createdChatId = ev.chatId;
   765→          }
   766→          if (
   767→            typeof ev.userMessageId === "string" &&
   768→            ev.userMessageId &&
   769→            appendUserBubble
   770→          ) {
   771→            const serverId = ev.userMessageId;
   772→            // patch FIRST (message still carries the local id), then move
   773→            // the cursor so later patches (sources, …) target the new id
   774→            updateUser({ id: serverId });
   775→            currentUserMsgId = serverId;
   776→          }
   777→        } else if (ev.type === "duel_start") {
   778→          // Reconcile lane models with the server's normalization.
   779→          if (duel) {
   780→            if (typeof ev.a === "string" && ev.a) updateLane("a", { model: ev.a });
   781→            if (typeof ev.b === "string" && ev.b) updateLane("b", { model: ev.b });
   782→          }
   783→        } else if (ev.type === "search_status") {
   784→          const status = String(ev.status ?? "");
   785→          if (status === "searching") {
   786→            updateAssistant({ phase: "searching" });
   787→            if (duel) {
   788→              updateLane("a", { phase: "searching" });
   789→              updateLane("b", { phase: "searching" });
   790→            }
   791→          } else if (status === "ok") {
   792→            updateAssistant({ phase: "thinking", searchFailed: false });
   793→            if (duel) {
   794→              updateLane("a", { phase: "thinking" });
   795→              updateLane("b", { phase: "thinking" });
   796→            }
   797→          } else if (status === "failed") {
   798→            updateAssistant({ phase: "thinking", searchFailed: true });
   799→            if (duel) {
   800→              updateLane("a", { phase: "thinking" });
   801→              updateLane("b", { phase: "thinking" });
   802→            }
   803→          }
   804→        } else if (ev.type === "sources") {
   805→          const sources = (ev.sources ?? []) as SearchSource[];
   806→          if (sources.length) updateUser({ sources });
   807→        } else if (ev.type === "restart") {
   808→          // A provider died mid-stream; the cloud re-generates that
   809→          // lane from scratch. Discard only the affected lane's
   810→          // partial text (untagged = normal single-lane turn).
   811→          const lane = ev.lane === "b" ? "b" : "a";
   812→          if (duel) {
   813→            laneContents[lane] = "";
   814→            updateLane(lane, { content: "", phase: "thinking" });
   815→          } else {
   816→            assistantContent = "";
   817→            updateAssistant({ content: "", phase: "thinking" });
   818→          }
   819→        } else if (ev.type === "delta") {
   820→          const lane = ev.lane === "b" ? "b" : "a";
   821→          if (duel) {
   822→            laneContents[lane] += String(ev.text ?? "");
   823→            updateLane(lane, { content: laneContents[lane], phase: null });
   824→          } else {
   825→            assistantContent += String(ev.text ?? "");
   826→            updateAssistant({ content: assistantContent, phase: null });
   827→          }
   828→        } else if (ev.type === "lane_done") {
   829→          // Final per-lane stats from the gateway — server truth wins.
   830→          const lane = ev.lane === "b" ? "b" : "a";
   831→          if (duel) {
   832→            updateLane(lane, {
   833→              content: (ev.content as string) ?? laneContents[lane],
   834→              streaming: false,
   835→              phase: null,
   836→              tokens: (ev.tokens as number) ?? 0,
   837→              cached: ev.cached === true,
   838→              servedBy: (ev.servedBy as string) ?? null,
   839→              keyLabel: (ev.keyLabel as string) ?? null,
   840→              error: (ev.error as string) ?? null,
   841→            });
   842→          }
   843→        } else if (ev.type === "done") {
   844→          if (duel && ev.duel && typeof ev.duel === "object") {
   845→            const lanes = ev.duel as {
   846→              a: { content: string; error: string | null; tokens: number; cached: boolean; servedBy: string | null; keyLabel: string | null; model: string };
   847→              b: { content: string; error: string | null; tokens: number; cached: boolean; servedBy: string | null; keyLabel: string | null; model: string };
   848→            };
   849→            updateAssistant({
   850→              streaming: false,
   851→              phase: null,
   852→              searchFailed: ev.searchFailed === true,
   853→            });
   854→            updateLane("a", {
   855→              content: lanes.a.content,
   856→              model: lanes.a.model,
   857→              streaming: false,
   858→              phase: null,
   859→              tokens: lanes.a.tokens,
   860→              cached: lanes.a.cached,
   861→              servedBy: lanes.a.servedBy,
   862→              keyLabel: lanes.a.keyLabel,
   863→              error: lanes.a.error,
   864→            });
   865→            updateLane("b", {
   866→              content: lanes.b.content,
   867→              model: lanes.b.model,
   868→              streaming: false,
   869→              phase: null,
   870→              tokens: lanes.b.tokens,
   871→              cached: lanes.b.cached,
   872→              servedBy: lanes.b.servedBy,
   873→              keyLabel: lanes.b.keyLabel,
   874→              error: lanes.b.error,
   875→            });
   876→            // One lane died → the duel silently degrades: keep the
   877→            // surviving lane as the answer (no pick needed).
   878→            if (lanes.a.error && !lanes.b.error && lanes.b.content.trim()) {
   879→              void get().pickDuelWinner(assistantId, "b");
   880→            } else if (lanes.b.error && !lanes.a.error && lanes.a.content.trim()) {
   881→              void get().pickDuelWinner(assistantId, "a");
   882→            }
   883→            // Both lanes healthy → the message waits for the user's pick.
   884→          } else {
   885→            updateAssistant({
   886→              id: (ev.assistantMessageId as string) ?? assistantId,
   887→              content: (ev.content as string) ?? assistantContent,
   888→              streaming: false,
   889→              cached: Boolean(ev.cached),
   890→              tokens: (ev.tokens as number) ?? 0,
   891→              servedBy: (ev.servedBy as string) ?? null,
   892→              keyLabel: (ev.keyLabel as string) ?? null,
   893→              provider: (ev.providerName as string) ?? null,
   894→              model: (ev.model as string) ?? model,
   895→              phase: null,
   896→              searchFailed: ev.searchFailed === true,
   897→            });
   898→          }
   899→        } else if (ev.type === "error") {
   900→          updateAssistant({
   901→            streaming: false,
   902→            error: String(ev.message ?? "Generation failed"),
   903→            content: "",
   904→            phase: null,
   905→          });
   906→        }
   907→      }
   908→    }
   909→  } catch (e) {
   910→    const err = e as Error;
   911→    if (err.name === "AbortError") {
   912→      // User pressed stop: a stopped duel keeps both partial lanes —
   913→      // the user can still pick either one as the answer.
   914→      if (duel) {
   915→        updateAssistant({ streaming: false, phase: null });
   916→        updateLane("a", { streaming: false, phase: null });
   917→        updateLane("b", { streaming: false, phase: null });
   918→      } else {
   919→        updateAssistant({ streaming: false, phase: null });
   920→      }
   921→    } else {
   922→      updateAssistant({
   923→        streaming: false,
   924→        error: err.message,
   925→        content: "",
   926→        phase: null,
   927→      });
   928→    }
   929→  } finally {
   930→    streamController = null;
   931→    set({ streaming: false });
   932→    if (!temporary) {
   933→      // Await the refresh so the raw derived title is in the list
   934→      // BEFORE the refined one patches it (no clobber race).
   935→      await get().refreshChats();
   936→      if (createdChatId) void refineTitle(set, createdChatId);
   937→    }
   938→    void get().refreshUsage();
   939→  }
   940→}
   941→
   942→/**
   943→ * Replace a brand-new chat's raw derived title ("who r u?") with a
   944→ * proper one from a cheap cloud model pass: "Who Are You?". Fire and
   945→ * forget — failure keeps the derived title.
   946→ */
   947→async function refineTitle(set: SetState, chatId: string): Promise<void> {
   948→  try {
   949→    const res = await fetch("/api/chat/title", {
   950→      method: "POST",
   951→      headers: { "Content-Type": "application/json" },
   952→      body: JSON.stringify({ chatId }),
   953→    });
   954→    if (!res.ok) return;
   955→    const data = (await res.json()) as { title?: string };
   956→    const title = typeof data.title === "string" ? data.title.trim() : "";
   957→    if (!title) return;
   958→    set((s) => ({
   959→      chats: s.chats.map((c) => (c.id === chatId ? { ...c, title } : c)),
   960→    }));
   961→  } catch {
   962→    // keep the derived title
   963→  }
   964→}
