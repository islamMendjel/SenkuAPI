    30→import type { EffortLevel } from "@/lib/senku/types";

    52→  },
    53→  {
    54→    level: "medium",
    55→    label: "Medium",
    56→    desc: "Your selected model · balanced (default)",
    57→    icon: Sparkles,
    58→    pill: "",
    59→  },
    60→  {
    61→    level: "deep",
    62→    label: "Deep",
    63→    desc: "Strongest engine · thinking · deep research with Search",

    66→  },

    68→
    69→function formatElapsed(ms: number): string {
    70→  const total = Math.floor(ms / 1000);
    71→  return `${Math.floor(total / 60)}:${String(total % 60).padStart(2, "0")}`;
    72→}
    73→
    74→export function InputBar() {
    75→  const {
    76→    searchEnabled,
    77→    toggleSearch,
    78→    temporary,
    79→    model,
    80→    models,
    81→    setModel,
    82→    syncModels,
    83→    streaming,
    84→    sendMessage,
    85→    stopStreaming,
    86→    limiter,
    87→    duelEnabled,
    88→    toggleDuel,
    89→    challenger,
    90→    setChallenger,
    91→    effort,
    92→    setEffort,
    93→  } = useSenku();
    94→  const { toast } = useToast();
    95→
    96→  const [value, setValue] = useState("");
    97→  const [attachOpen, setAttachOpen] = useState(false);
    98→  const [modelOpen, setModelOpen] = useState(false);
    99→  const [challengerOpen, setChallengerOpen] = useState(false);
   100→  const [effortOpen, setEffortOpen] = useState(false);
   101→  const [syncing, setSyncing] = useState(false);
   102→  const [transcribing, setTranscribing] = useState(false);
   103→  const textareaRef = useRef<HTMLTextAreaElement>(null);
   104→  const fileRef = useRef<HTMLInputElement>(null);
   105→  const recorder = useVoiceRecorder();
   106→  const waveBarsRef = useRef<(HTMLDivElement | null)[]>([]);
   107→  const waveLevelsRef = useRef<number[]>(new Array(WAVE_BARS).fill(4));
   108→
   109→  const providerName = limiter?.providerName ?? "";
   110→  const canSend = value.trim().length > 0 && !streaming;
   111→  const isRecording = recorder.state === "recording";
   112→  const isProcessing = recorder.state === "processing" || transcribing;
   113→  // A duel needs a second model to race against.
   114→  const duelAvailable = models.filter((m) => m !== model).length > 0;
   115→
   116→  // auto-grow textarea
   117→  useEffect(() => {
   118→    const el = textareaRef.current;
   119→    if (!el) return;
   120→    el.style.height = "0px";
   121→    el.style.height = Math.min(el.scrollHeight, 208) + "px";
   122→  }, [value]);
   123→
   124→  // surface recorder permission/device errors
   125→  useEffect(() => {
   126→    if (recorder.error) {
   127→      toast({ description: recorder.error, variant: "destructive" });
   128→    }
   129→  }, [recorder.error, toast]);
   130→
   131→  // live waveform: push the smoothed mic level through a rolling window
   132→  // and paint bar heights directly on the DOM (no re-render per frame)
   133→  useEffect(() => {
   134→    if (recorder.state !== "recording") return;
   135→    let raf = 0;
   136→    const loop = () => {
   137→      const level = recorder.levelRef.current;
   138→      const next = waveLevelsRef.current
   139→        .slice(1)
   140→        .concat(4 + level * level * 42 + Math.random() * 2);
   141→      waveLevelsRef.current = next;
   142→      for (let i = 0; i < WAVE_BARS; i++) {
   143→        const el = waveBarsRef.current[i];
   144→        if (el) el.style.height = `${Math.max(4, next[i]).toFixed(1)}px`;
   145→      }
   146→      raf = requestAnimationFrame(loop);
   147→    };
   148→    raf = requestAnimationFrame(loop);
   149→    return () => cancelAnimationFrame(raf);
   150→  }, [recorder.state, recorder.levelRef]);
   151→
   152→  const submit = () => {
   153→    if (!canSend) return;
   154→    const content = value.trim();
   155→    setValue("");
   156→    void sendMessage(content);
   157→  };
   158→
   159→  // stop → transcribe (server ASR) → land the text in the input
   160→  const finishDictation = async () => {
   161→    recorder.stop();
   162→    setTranscribing(true);
   163→    try {
   164→      const wav = await recorder.takeResult();
   165→      if (!wav) {
   166→        toast({ description: "Didn't catch that — try speaking a bit louder." });
   167→        return;
   168→      }
   169→      const res = await fetch("/api/voice/transcribe", {
   170→        method: "POST",
   171→        headers: { "Content-Type": "application/json" },
   172→        body: JSON.stringify({ audio: wav }),
   173→      });
   174→      const data = await res.json();
   175→      if (!res.ok) throw new Error(data.error || "Transcription failed");
   176→      const text = String(data.text ?? "").trim();
   177→      if (!text) {
   178→        toast({ description: "Didn't catch that — try speaking a bit louder." });
   179→        return;
   180→      }
   181→      setValue((v) => (v ? `${v} ${text}` : text));
   182→      requestAnimationFrame(() => {
   183→        const el = textareaRef.current;
   184→        if (el) {
   185→          el.focus();
   186→          el.setSelectionRange(el.value.length, el.value.length);
   187→        }
   188→      });
   189→    } catch (e) {
   190→      toast({
   191→        description: (e as Error).message || "Transcription failed.",
   192→        variant: "destructive",
   193→      });
   194→    } finally {
   195→      setTranscribing(false);
   196→    }
   197→  };
   198→
   199→  const onMicClick = () => {
   200→    if (recorder.state === "recording") {
   201→      void finishDictation();
   202→    } else if (recorder.state === "idle") {
   203→      void recorder.start();
   204→    }
   205→  };
   206→
   207→  const onUpload = () => {
   208→    setAttachOpen(false);
   209→    fileRef.current?.click();
   210→  };
   211→
   212→  const onFilePicked = (e: React.ChangeEvent<HTMLInputElement>) => {
   213→    const files = Array.from(e.target.files ?? []);
   214→    if (files.length) {
   215→      setValue((v) =>
   216→        (v ? v + "\n" : "") + files.map((f) => `[Attached: ${f.name}]`).join("\n")
   217→      );
   218→      toast({
   219→        description: `${files.length} file${
   220→          files.length === 1 ? "" : "s"
   221→        } attached — names included in the prompt.`,
   222→      });
   223→    }
   224→    e.target.value = "";
   225→  };
   226→
   227→  const activeMeta = modelMeta(model);
   228→
   229→  return (
   230→    <div className="relative mx-auto w-full max-w-3xl shrink-0 px-4">
   231→      <input
   232→        ref={fileRef}
   233→        type="file"
   234→        multiple
   235→        hidden
   236→        onChange={onFilePicked}
   237→        aria-hidden
   238→      />
   239→
   240→      <div className="rounded-[26px] border border-zinc-700/70 bg-[#262627] shadow-[0_8px_32px_rgba(0,0,0,0.45)] transition-colors focus-within:border-zinc-500">
   241→        {isRecording ? (
   242→          /* live waveform strip replaces the textarea while dictating */
   243→          <div className="senku-rec-in flex h-[52px] items-center gap-3 px-4" aria-live="polite">
   244→            <span className="flex shrink-0 items-center gap-2 tabular-nums text-[13px] text-red-400">
   245→              <span className="senku-rec-dot inline-block h-2 w-2 rounded-full bg-red-400" />
   246→              {formatElapsed(recorder.elapsedMs)}
   247→            </span>
   248→            <div className="flex h-10 min-w-0 flex-1 items-center justify-center gap-[3px]" aria-hidden>
   249→              {Array.from({ length: WAVE_BARS }).map((_, i) => (
   250→                <div
   251→                  key={i}
   252→                  ref={(el) => {
   253→                    waveBarsRef.current[i] = el;
   254→                  }}
   255→                  className="senku-wave-bar"
   256→                  style={{ height: 4 }}
   257→                />
   258→              ))}
   259→            </div>
   260→            <span className="hidden shrink-0 text-[11px] text-zinc-500 sm:block">
   261→              tap ◼ to stop
   262→            </span>
   263→          </div>
   264→        ) : isProcessing ? (
   265→          <div className="flex h-[52px] items-center px-4" aria-live="polite">
   266→            <span className="senku-thinking-label text-[14px] font-medium">
   267→              Transcribing…
   268→            </span>
   269→          </div>
   270→        ) : (
   271→          <textarea
   272→            ref={textareaRef}
   273→            value={value}
   274→            onChange={(e) => setValue(e.target.value)}
   275→            onKeyDown={(e) => {
   276→              if (e.key === "Enter" && !e.shiftKey) {
   277→                e.preventDefault();
   278→                submit();
   279→              }
   280→            }}
   281→            rows={1}
   282→            placeholder="Message SenkuAPI..."
   283→            aria-label="Message SenkuAPI"
   284→            className="senku-scroll max-h-52 w-full resize-none bg-transparent px-4 pb-1 pt-3.5 text-[15px] leading-relaxed text-zinc-100 outline-none placeholder:text-zinc-500"
   285→          />
   286→        )}
   287→
   288→        <div className="flex items-center justify-between gap-2 px-2.5 pb-2.5 pt-0.5">
   289→          {/* left controls — horizontally scrollable on narrow screens
   290→              so Search + Duel + challenger never wrap or overflow */}
   291→          <div className="senku-scroll flex min-w-0 items-center gap-1.5 overflow-x-auto">
   292→            <Popover open={attachOpen} onOpenChange={setAttachOpen}>
   293→              <PopoverTrigger asChild>
   294→                <button
   295→                  aria-label="Add attachments"
   296→                  className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full text-zinc-400 transition-colors hover:bg-white/10 hover:text-zinc-200"
   297→                >
   298→                  <Plus className="h-[18px] w-[18px]" />
   299→                </button>
   300→              </PopoverTrigger>
   301→              <PopoverContent
   302→                side="top"
   303→                align="start"
   304→                className="w-64 rounded-xl border-zinc-800 bg-[#262627] p-1.5 text-zinc-200 shadow-2xl"
   305→              >
   306→                <button
   307→                  onClick={onUpload}
   308→                  className="flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-[13px] hover:bg-white/5"
   309→                >
   310→                  <Paperclip className="h-4 w-4 text-zinc-400" />
   311→                  Upload from computer
   312→                </button>
   313→                <button
   314→                  onClick={() => {
   315→                    toggleSearch();
   316→                    setAttachOpen(false);
   317→                  }}
   318→                  className="flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-[13px] hover:bg-white/5"
   319→                >
   320→                  <Globe className="h-4 w-4 text-zinc-400" />
   321→                  Search the web
   322→                  <span
   323→                    className={`ml-auto rounded-full px-1.5 py-0.5 text-[10px] font-medium ${
   324→                      searchEnabled
   325→                        ? "bg-emerald-400/15 text-emerald-300"
   326→                        : "bg-zinc-700/60 text-zinc-400"
   327→                    }`}
   328→                  >
   329→                    {searchEnabled ? "On" : "Off"}
   330→                  </span>
   331→                </button>
   332→                {duelAvailable && (
   333→                  <button
   334→                    onClick={() => {
   335→                      if (!duelEnabled) {
   336→                        toast({
   337→                          description:
   338→                            "Duel armed — next message races two models; you pick the keeper.",
   339→                        });
   340→                      }
   341→                      toggleDuel();
   342→                      setAttachOpen(false);
   343→                    }}
   344→                    className="flex w-full items-center gap-2.5 rounded-lg px-2.5 py-2 text-[13px] hover:bg-white/5"
   345→                  >
   346→                    <Swords className="h-4 w-4 text-zinc-400" />
   347→                    Duel two models
   348→                    <span
   349→                      className={`ml-auto rounded-full px-1.5 py-0.5 text-[10px] font-medium ${
   350→                        duelEnabled
   351→                          ? "bg-emerald-400/15 text-emerald-300"
   352→                          : "bg-zinc-700/60 text-zinc-400"
   353→                      }`}
   354→                    >
   355→                      {duelEnabled ? "On" : "Off"}
   356→                    </span>
   357→                  </button>
   358→                )}
   359→              </PopoverContent>
   360→            </Popover>
   361→
   362→            <button
   363→              onClick={toggleSearch}
   364→              aria-pressed={searchEnabled}
   365→              className={`flex h-8 shrink-0 items-center gap-1.5 rounded-full border px-2.5 text-[13px] transition-colors ${
   366→                searchEnabled
   367→                  ? "border-zinc-400 bg-white/10 text-white"
   368→                  : "border-zinc-600 text-zinc-300 hover:bg-white/5"
   369→              }`}
   370→              title="Toggle web search — grounded answers with cited sources (max 3 snippets, 300 chars each)"
   371→            >
   372→              <Globe className="h-3.5 w-3.5" />
   373→              Search
   374→            </button>
   375→
   376→            {/* Effort dial — fast / medium / deep. Each level can
   377→                route to a different pooled engine; the popover shows
   378→                exactly which engine serves each level. */}
   379→            <Popover open={effortOpen} onOpenChange={setEffortOpen}>
   380→              <PopoverTrigger asChild>
   381→                <button
   382→                  aria-label={`Effort level: ${effort}. Change how hard the gateway works`}
   383→                  aria-expanded={effortOpen}
   384→                  className={`flex h-8 shrink-0 items-center gap-1.5 rounded-full border px-2.5 text-[13px] transition-colors ${
   385→                    effort === "medium"
   386→                      ? "border-zinc-600 text-zinc-300 hover:bg-white/5"
   387→                      : EFFORTS.find((e) => e.level === effort)?.pill ?? ""
   388→                  }`}
   389→                  title="Effort — Fast: fastest engine, brief · Medium: your model · Deep: strongest engine, thinking + deep research"
   390→                >
   391→                  {(() => {
   392→                    const E = EFFORTS.find((e) => e.level === effort) ?? EFFORTS[1];
   393→                    return <E.icon className="h-3.5 w-3.5" />;
   394→                  })()}
   395→                  {EFFORTS.find((e) => e.level === effort)?.label ?? "Medium"}
   396→                  <ChevronDown className="h-3 w-3 shrink-0 text-zinc-500" />
   397→                </button>
   398→              </PopoverTrigger>
   399→              <PopoverContent
   400→                side="top"
   401→                align="start"
   402→                className="w-80 rounded-xl border-zinc-800 bg-[#212121] p-0 shadow-2xl"
   403→              >
   404→                <div className="border-b border-zinc-800 px-3.5 py-2.5">
   405→                  <span className="text-[11px] font-semibold tracking-[0.12em] text-zinc-400">
   406→                    EFFORT — HOW HARD THIS TURN WORKS
   407→                  </span>
   408→                </div>
   409→                <div className="py-1">
   410→                  {EFFORTS.map(({ level, label, desc, icon: Icon }) => {
   411→                    const routed = effortModel(models, model, level);
   412→                    const routedMeta = modelMeta(routed);
   413→                    return (
   414→                      <button
   415→                        key={level}
   416→                        onClick={() => {
   417→                          setEffort(level);
   418→                          setEffortOpen(false);
   419→                        }}
   420→                        className={`flex w-full items-center gap-2.5 rounded-lg px-3 py-2 text-left transition-colors hover:bg-white/5 ${
   421→                          level === effort ? "text-white" : "text-zinc-300"
   422→                        }`}
   423→                      >
   424→                        <Icon
   425→                          className={`h-4 w-4 shrink-0 ${
   426→                            level === effort
   427→                              ? level === "fast"
   428→                                ? "text-yellow-300"
   429→                                : level === "deep"
   430→                                  ? "text-emerald-300"
   431→                                  : "text-zinc-200"
   432→                              : "text-zinc-500"
   433→                          }`}
   434→                        />
   435→                        <span className="min-w-0 flex-1">
   436→                          <span className="block text-[13px] font-medium">{label}</span>
   437→                          <span className="block truncate text-[11px] text-zinc-500">
   438→                            {desc}
   439→                          </span>
   440→                        </span>
   441→                        <span
   442→                          className="max-w-[96px] shrink-0 truncate text-[11px] text-zinc-400"
   443→                          title={routed}
   444→                        >
   445→                          {routed === model ? routedMeta.name : `→ ${routedMeta.name}`}
   446→                        </span>
   447→                        {level === effort && (
   448→                          <Check className="h-3.5 w-3.5 shrink-0 text-emerald-400" />
   449→                        )}
   450→                      </button>
   451→                    );
   452→                  })}
   453→                </div>
   454→                <p className="border-t border-zinc-800 px-3.5 py-2 text-[10.5px] leading-relaxed text-zinc-600">
   455→                  Deep turns run multi-round audited search when Search is on —
   456→                  plan → retrieve → cross-check — and always show their sources.
   457→                </p>
   458→              </PopoverContent>
   459→            </Popover>
   460→
   461→            {/* Duel — race two models on one question, keep the winner.
   462→                Only a pooled gateway can offer this in-chat. */}
   463→            <button
   464→              onClick={() => {
   465→                if (!duelEnabled && duelAvailable) {
   466→                  toast({
   467→                    description:
   468→                      "Duel armed — next message races two models; you pick the keeper.",
   469→                  });
   470→                }
   471→                toggleDuel();
   472→              }}
   473→              disabled={!duelAvailable}
   474→              aria-pressed={duelEnabled}
   475→              className={`flex h-8 shrink-0 items-center gap-1.5 rounded-full border px-2.5 text-[13px] transition-colors disabled:cursor-not-allowed disabled:opacity-40 ${
   476→                duelEnabled
   477→                  ? "border-amber-300/70 bg-amber-300/10 text-amber-200"
   478→                  : "border-zinc-600 text-zinc-300 hover:bg-white/5"
   479→              }`}
   480→              title={duelAvailable ? "Duel — send to two models at once, keep the better answer (bills both)" : "Add a second model to the active provider to duel"}
   481→            >
   482→              <Swords className="h-3.5 w-3.5" />
   483→              Duel
   484→            </button>
   485→
   486→            {/* Challenger picker — shown while duel is armed */}
   487→            {duelEnabled && challenger && (
   488→              <Popover open={challengerOpen} onOpenChange={setChallengerOpen}>
   489→                <PopoverTrigger asChild>
   490→                  <button
   491→                    aria-label="Pick the dueling challenger model"
   492→                    title={`Challenger — currently ${modelMeta(challenger).name}`}
   493→                    className="flex h-8 max-w-[92px] shrink-0 items-center gap-1 rounded-full border border-zinc-700 bg-zinc-800/40 px-2.5 text-[12px] text-zinc-300 transition-colors hover:bg-white/5 sm:max-w-[170px]"
   494→                  >
   495→                    <span className="text-[10px] font-semibold uppercase tracking-wider text-amber-300/80">
   496→                      vs
   497→                    </span>
   498→                    <span className="truncate font-medium">
   499→                      {modelMeta(challenger).name}
   500→                    </span>

   504→                <PopoverContent

   506→                  align="start"

   509→                  <div className="border-b border-zinc-800 px-3.5 py-2.5">

   511→                      CHALLENGER MODEL
   512→                    </span>

   514→                  <div className="senku-scroll max-h-72 overflow-y-auto py-1">
   515→                    {models
   516→                      .filter((m) => m !== model)
   517→                      .map((m) => {
   518→                        const meta = modelMeta(m);

   526→                            title={m}
   527→                            className={`flex w-full items-center justify-between gap-2 rounded-lg px-3 py-2 text-left text-[13px] transition-colors hover:bg-white/5 ${

   530→                          >

   543→                          </button>
