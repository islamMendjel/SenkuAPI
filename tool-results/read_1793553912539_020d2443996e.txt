     1→#!/usr/bin/env python3
     2→"""
     3→test_senku.py — end-to-end verification suite for SenkuAPI.
     4→
     5→Pure Python 3 stdlib (urllib.request + json — no third-party packages).
     6→Runs against the LIVE dev server (default http://localhost:3000, override
     7→with the SENKU_BASE_URL env var):
     8→
     9→    python3 tests/test_senku.py        (equivalently: bun run test:senku)
    10→
    11→Exit code 0 only if every check passes.
    12→
    13→NOTE ON CACHE PROBES: SenkuAPI caches exact-match responses for 24 h, so
    14→each probe message carries a per-run nonce. This keeps the suite
    15→re-runnable (the "first" request of every cache test is always a genuine
    16→cache miss) while preserving the semantics under test: the two requests
    17→inside a cache test share an IDENTICAL body, byte for byte.
    18→"""

    28→BASE = os.environ.get("SENKU_BASE_URL", "http://localhost:3000")
    29→TIMEOUT = 90  # seconds — cloud calls can be slow
    30→NONCE = "{}-{}".format(int(time.time()), uuid.uuid4().hex[:6])

    32→RESULTS = []  # (test_no, label, ok, detail)

    39→def check(test_no, label, ok, detail=""):
    40→    """Record + print one PASS/FAIL line; returns the boolean."""
    41→    ok = bool(ok)
    42→    RESULTS.append((test_no, label, ok, detail))
    43→    mark = "PASS" if ok else "FAIL"
    44→    line = "  [{}] Test {}: {}".format(mark, test_no, label)
    45→    if detail:
    46→        line += "  — {}".format(detail)
    47→    print(line, flush=True)
    48→    return ok

    51→def api(method, path, body=None, timeout=TIMEOUT):
    52→    """JSON request helper. Returns (status, parsed_body, content_type)."""
    53→    url = BASE + path
    54→    data = None
    55→    headers = {"Accept": "application/json"}
    56→    if body is not None:
    57→        data = json.dumps(body).encode("utf-8")
    58→        headers["Content-Type"] = "application/json"
    59→    req = urllib.request.Request(url, data=data, method=method, headers=headers)
    60→    try:
    61→        with urllib.request.urlopen(req, timeout=timeout) as resp:
    62→            status, ctype = resp.status, resp.headers.get("Content-Type", "")
    63→            raw = resp.read()
    64→    except urllib.error.HTTPError as e:
    65→        status, ctype = e.code, e.headers.get("Content-Type", "")
    66→        raw = e.read()
    67→    except urllib.error.URLError as e:
    68→        raise RuntimeError("cannot reach {} {}: {}".format(method, url, e))
    69→    try:
    70→        parsed = json.loads(raw.decode("utf-8"))
    71→    except (ValueError, UnicodeDecodeError):
    72→        parsed = None
    73→    return status, parsed, ctype

    76→def sse_post(path, body, timeout=TIMEOUT):
    77→    """POST and consume a text/event-stream response.
    78→
    79→    Reads the raw stream, splits frames on blank lines ("\\n\\n") and
    80→    parses every line starting with "data: ". Returns
    81→    (content_type, [events]).
    82→    """
    83→    url = BASE + path
    84→    data = json.dumps(body).encode("utf-8")
    85→    req = urllib.request.Request(
    86→        url, data=data, method="POST",
    87→        headers={"Content-Type": "application/json", "Accept": "text/event-stream"},
    88→    )
    89→    try:
    90→        with urllib.request.urlopen(req, timeout=timeout) as resp:
    91→            ctype = resp.headers.get("Content-Type", "")
    92→            raw = resp.read()  # stream stays open until the server closes it
    93→    except urllib.error.HTTPError as e:
    94→        # Non-2xx: surface the error body as a single error-ish event.
    95→        ctype = e.headers.get("Content-Type", "")
    96→        raw = e.read()
    97→    except urllib.error.URLError as e:
    98→        raise RuntimeError("cannot reach POST {}: {}".format(url, e))
    99→
   100→    text = raw.decode("utf-8", errors="replace")
   101→    events = []
   102→    for frame in text.split("\n\n"):
   103→        for line in frame.split("\n"):
   104→            line = line.strip()
   105→            if line.startswith("data:"):
   106→                payload = line[len("data:"):].strip()
   107→                if payload:
   108→                    try:
   109→                        events.append(json.loads(payload))
   110→                    except ValueError:
   111→                        pass
   112→    return ctype, events

   518→def run_test(test_no, fn, *args):
   519→    """Run one test function; a crash counts as a FAIL for that test."""
   520→    try:
   521→        fn(*args)
   522→    except Exception as e:  # noqa: BLE001 — report, don't die
   523→        check(test_no, "{} completed without exception".format(fn.__name__), False,
   524→              "{}: {}".format(type(e).__name__, e))

   527→def summarize():
   528→    by_test = {}
   529→    for test_no, label, ok, detail in RESULTS:
   530→        agg = by_test.setdefault(test_no, {"pass": 0, "fail": 0, "fails": []})
   531→        if ok:
   532→            agg["pass"] += 1
   533→        else:
   534→            agg["fail"] += 1
   535→            agg["fails"].append("{}{}".format(label, " ({})".format(detail) if detail else ""))
   536→
   537→    print("\n" + "=" * 68)
   538→    print("SUMMARY")
   539→    print("=" * 68)
   540→    all_ok = True
   541→    for test_no in sorted(by_test):
   542→        agg = by_test[test_no]
   543→        status = "PASS" if agg["fail"] == 0 else "FAIL"
   544→        print("  Test {:<3}: {}  ({}/{} checks)".format(test_no, status, agg["pass"],
   545→                                                        agg["pass"] + agg["fail"]))
   546→        for f in agg["fails"]:
   547→            print("           ✗ {}".format(f))
   548→        if agg["fail"]:
   549→            all_ok = False
   550→    total_pass = sum(1 for _, _, ok, _ in RESULTS if ok)
   551→    print("-" * 68)
   552→    print("TOTAL: {}/{} checks passed — {}".format(
   553→        total_pass, len(RESULTS), "ALL TESTS PASSED" if all_ok else "SOME TESTS FAILED"))
   554→    return all_ok
