    55→function CopyButton({ text }: { text: string }) {

    81→function SourcesBlock({ sources }: { sources: NonNullable<UIMessage["sources"]> }) {
    82→  // >3 results can only come from the deep-research pipeline
    83→  // (medium search is capped at 3) — say so, and scroll.
    84→  const deep = sources.length > 3;
    85→  return (
    86→    <div className="mt-3 rounded-xl border border-zinc-800 bg-[#1d1d20] p-3">
    87→      <div className="mb-2 flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-zinc-500">
    88→        <RefreshCcw className="h-3 w-3" />
    89→        {deep
    90→          ? `Deep research · ${sources.length} sources`
    91→          : `Searched the web · ${sources.length} result${
    92→              sources.length === 1 ? "" : "s"
    93→            }`}
    94→      </div>
    95→      <div
    96→        className={`space-y-2.5${deep ? " senku-scroll max-h-96 overflow-y-auto pr-1" : ""}`}
    97→      >
    98→        {sources.map((s, i) => {
    99→          let host = s.url;
   100→          try {
   101→            host = new URL(s.url).hostname.replace(/^www\./, "");
   102→          } catch {
   103→            /* keep raw */
   104→          }
   105→          return (
   106→            <div key={i} className="min-w-0">
   107→              <a
   108→                href={s.url}

   110→                rel="noopener noreferrer"

   113→                {s.title || host}

   254→/** One quiet provenance line: "GLM 5.3 · APInex pool · 1.9k tokens · cache: hit". */
   255→function provenanceLine(m: UIMessage): string {
   256→  // Run 22 — forged images carry their own receipt: dimensions, file
   257→  // size, forge duration, and the fact that zero pool tokens were
   258→  // spent (the forge runs on the Z.ai image engine, not the pool).
   259→  if (m.image) {
   260→    const parts = ["Image Forge", `${m.image.width}×${m.image.height}`];
   261→    if (m.image.bytes > 0) {
   262→      parts.push(
   263→        m.image.bytes >= 1024 * 1024
   264→          ? `${(m.image.bytes / (1024 * 1024)).toFixed(1)} MB`
   265→          : `${Math.max(1, Math.round(m.image.bytes / 1024))} KB`
   266→      );
   267→    }
   268→    if (m.image.ms > 0) parts.push(`${(m.image.ms / 1000).toFixed(1)}s`);
   269→    parts.push("0 pool tokens");
   270→    return parts.join(" · ");
   271→  }
   272→  const parts: string[] = [];
   273→  // The effort dial leaves its mark — fast/deep turns say so.
   274→  if (m.effort === "fast") parts.push("Fast mode");
   275→  if (m.effort === "deep") parts.push("Deep research");
   276→  if (m.model) parts.push(modelMeta(m.model).name);
   277→  if (m.provider) {
   278→    parts.push(
   279→      m.servedBy === "zai-cloud"
   280→        ? `${m.provider} · cloud failover`
   281→        : `${m.provider} pool`
   282→    );
   283→  }
   284→  if (typeof m.tokens === "number" && m.tokens > 0) {
   285→    // Estimated counts (upstream sent no usage stats) say so — an
   286→    // estimate is never presented as exact usage (Run 17).
   287→    parts.push(
   288→      m.tokensEstimated
   289→        ? `≈${formatCompact(m.tokens)} tokens (estimated)`
   290→        : `${formatCompact(m.tokens)} tokens`
   291→    );
   292→  }
   293→  parts.push(m.cached ? "cache: hit" : "cache: miss");
   294→  return parts.join(" · ");
   295→}

   297→/** Body without the trailing label line (display only). */
   298→function withoutConfidenceLine(content: string): string {
   299→  if (!confidenceOf(content)) return content;
   300→  const lines = content.trimEnd().split("\n");
   301→  return lines.slice(0, -1).join("\n").trimEnd();
   302→}

   331→  lane,
   332→  canPick,

   343→  const display = confidence ? withoutConfidenceLine(lane.content) : lane.content;
   344→  const keepable = canPick && !lane.error && lane.content.trim().length > 0;
   345→
   346→  return (
   347→    <div
   348→      className={`flex min-w-0 flex-col rounded-xl border p-3 ${
   349→        lane.error ? "border-red-500/25 bg-red-500/5" : "border-zinc-800 bg-[#1d1d20]"

   351→      aria-label={`Duel lane ${lane.id.toUpperCase()} — ${meta.name}`}
   352→    >
   353→      <div className="mb-2 flex items-center gap-2">
   354→        <span
   355→          className="flex h-5 w-5 shrink-0 items-center justify-center rounded-full border border-zinc-700 bg-zinc-800/60 text-[10px] font-bold uppercase text-zinc-400"
   356→          aria-hidden
   357→        >
   358→          {lane.id}
   359→        </span>
   360→        <span
   361→          className="min-w-0 truncate text-[12.5px] font-semibold text-zinc-200"
   362→          title={lane.model}
   363→        >
   364→          {meta.name}
   365→        </span>
   366→        {lane.cached && !lane.error && (
   367→          <span
   368→            className="flex shrink-0 items-center gap-1 text-emerald-400/70"
   369→            title="Served from the 24h response cache — zero tokens spent"

   373→        )}
   374→        <span className="ml-auto shrink-0">
   375→          <button
   376→            onClick={onKeep}
   377→            disabled={!keepable}
   378→            className={`flex h-7 items-center gap-1.5 rounded-full border px-3 text-[12px] font-medium transition-colors ${

   418→/** A duel turn awaiting its pick: two models racing side by side
   419→    (stacked on mobile), each with its own Keep button. */
   420→function DuelMessage({ message }: { message: UIMessage }) {
   421→  const { pickDuelWinner } = useSenku();
   422→  const { toast } = useToast();
   423→  const duel = message.duel;
   424→  if (!duel) return null;
   425→  // Picks unlock once the turn finishes (or is stopped) — picking
   426→  // mid-stream would race the lane_done/done reconciliation.
   427→  const canPick = !message.streaming;
   428→
   429→  const keep = (lane: "a" | "b") => {
   430→    const meta = modelMeta(duel[lane].model);
   431→    void pickDuelWinner(message.id, lane).then(() => {
   432→      toast({
   433→        description: `Kept ${meta.name} — the runner-up stays under the answer.`,
   434→      });
   435→    });
   436→  };
   437→
   438→  return (
   439→    <div>
   440→      <div className="mb-2 flex items-center gap-1.5 text-[11px] font-medium uppercase tracking-wider text-amber-300/70">
   441→        <Swords className="h-3 w-3" aria-hidden />
   442→        Duel — pick the keeper
   443→      </div>
   444→      <div className="grid grid-cols-1 items-start gap-3 md:grid-cols-2">
   445→        <DuelLaneCard lane={duel.a} canPick={canPick} onKeep={() => keep("a")} />
   446→        <DuelLaneCard lane={duel.b} canPick={canPick} onKeep={() => keep("b")} />
   447→      </div>
   448→      <p className="mt-2 text-[11px] text-zinc-600">
   449→        Both answers are billed separately ·{" "}
   450→        {canPick
   451→          ? "pick one to keep it in the chat"
   452→          : "generating both answers…"}
   453→      </p>
   454→    </div>
   455→  );
   456→}

   493→/** Run 22 — a settled forged image: rounded card, hover-zoom affordance,
   494→ *  click → fullscreen lightbox (Esc closes), one-click download, prompt
   495→ *  caption. Served from the local file route with immutable caching. */
   496→function GeneratedImageCard({ image }: { image: ImageRef }) {
   497→  const [zoom, setZoom] = useState(false);
   498→  const url = `/api/image/file/${image.id}`;
   499→
   500→  // Esc closes the lightbox — the keyboard path is first-class.
   501→  useEffect(() => {
   502→    if (!zoom) return;
   503→    const onKey = (e: KeyboardEvent) => {
   504→      if (e.key === "Escape") setZoom(false);
   505→    };
   506→    window.addEventListener("keydown", onKey);
   507→    return () => window.removeEventListener("keydown", onKey);
   508→  }, [zoom]);
   509→
   510→  return (
   511→    <figure className="senku-fade-up mt-0.5 w-full max-w-[min(480px,100%)]">
   512→      <button
   513→        type="button"
   514→        onClick={() => setZoom(true)}
   515→        aria-label="Open image full size"
   516→        title="Open full size"
   517→        className="group relative block w-full overflow-hidden rounded-xl border border-zinc-800 bg-[#1d1d20] transition-colors hover:border-zinc-600"
   518→      >
   519→        {/* Plain img: the file route serves with immutable caching and
   520→            exact dimensions — next/image adds nothing here. */}
   521→        <img
   522→          src={url}
   523→          alt={`Generated image — ${image.prompt}`}
   524→          width={image.width}
   525→          height={image.height}

   870→export function ChatTimeline() {
   871→  const { messages } = useSenku();
   872→  const { toast } = useToast();
   873→  const bottomRef = useRef<HTMLDivElement>(null);
   874→  const scrollRef = useRef<HTMLDivElement>(null);
   875→  // Auto-scroll stickiness: following the newest word pauses the
   876→  // moment the user scrolls up to read and resumes when they return
   877→  // to the bottom (every StreamedMarkdown consults the same state).
   878→  const stickRef = useStickyScroll(scrollRef);
   879→
   880→  // ── Read-aloud (TTS) playback state ──
   881→  // One audio element for the whole timeline; only one message can
   882→  // speak at a time. A token guards against stale fetches after a stop.
   883→  const [speech, setSpeech] = useState<SpeechState>(null);
   884→  const audioRef = useRef<HTMLAudioElement | null>(null);
   885→  const audioUrlRef = useRef<string | null>(null);
   886→  const speechTokenRef = useRef(0);
   887→
   888→  const stopSpeech = () => {
   889→    speechTokenRef.current += 1;
   890→    if (audioRef.current) {
   891→      audioRef.current.pause();
   892→      audioRef.current.removeAttribute("src");
   893→    }
   894→    if (audioUrlRef.current) {
   895→      URL.revokeObjectURL(audioUrlRef.current);
   896→      audioUrlRef.current = null;
   897→    }
   898→    setSpeech(null);
   899→  };
   900→
   901→  const toggleSpeech = async (m: UIMessage) => {
   902→    if (speech?.id === m.id) {
   903→      stopSpeech();
   904→      return;
   905→    }
   906→    stopSpeech();
   907→    const token = ++speechTokenRef.current;
   908→    setSpeech({ id: m.id, phase: "loading" });
   909→    try {
   910→      const res = await fetch("/api/voice/speak", {
   911→        method: "POST",
   912→        headers: { "Content-Type": "application/json" },
   913→        body: JSON.stringify({ text: m.content }),
   914→      });
   915→      if (!res.ok) {
   916→        const data = await res.json().catch(() => ({}));
   917→        throw new Error(data.error || "Read-aloud is unavailable right now.");
   918→      }
   919→      if (token !== speechTokenRef.current) return; // user moved on
   920→      const blob = await res.blob();
   921→      const url = URL.createObjectURL(blob);
   922→      audioUrlRef.current = url;
   923→      const audio = (audioRef.current ??= new Audio());
   924→      audio.src = url;
   925→      audio.onended = () => {
   926→        if (token === speechTokenRef.current) stopSpeech();
   927→      };
   928→      audio.onerror = () => {
   929→        if (token === speechTokenRef.current) stopSpeech();
   930→      };
   931→      await audio.play();
   932→      if (token !== speechTokenRef.current) return;
   933→      setSpeech({ id: m.id, phase: "playing" });
   934→    } catch (e) {
   935→      if (token === speechTokenRef.current) {
   936→        setSpeech(null);
   937→        if ((e as Error).name !== "NotAllowedError") {
   938→          toast({
   939→            description: (e as Error).message || "Could not play the audio.",
   940→            variant: "destructive",
   941→          });
   942→        }
   943→      }
   944→    }
   945→  };
   946→
   947→  // stop playback when the timeline unmounts or the chat changes
   948→  useEffect(() => {
   949→    return () => {
   950→      speechTokenRef.current += 1;
   951→      audioRef.current?.pause();
   952→      if (audioUrlRef.current) URL.revokeObjectURL(audioUrlRef.current);
   953→    };
   954→  }, []);
   955→
   956→  // Scroll for NEW messages only — content growth is followed
   957→  // word-by-word by the typewriter gate — and never yank the view
   958→  // back down if the user has scrolled up.
   959→  const count = messages.length;
   960→  const lastId = messages[count - 1]?.id ?? "";
   961→  useEffect(() => {
   962→    if (!stickRef.current) return;
   963→    bottomRef.current?.scrollIntoView({ behavior: "smooth", block: "end" });
   964→  }, [count, lastId, stickRef]);
   965→
   966→  return (
   967→    <div ref={scrollRef} className="senku-scroll flex-1 overflow-y-auto">
   968→      <div className="mx-auto w-full max-w-3xl space-y-6 px-4 pb-6 pt-4">
   969→        {messages.map((m, i) =>
   970→          m.role === "user" ? (
   971→            <UserMessage key={m.id} message={m} />
   972→          ) : (
   973→            /* Slot-stable key: the server reconciles the assistant
   974→               message id at "done" — keying by id would remount the
   975→               message right then and dump the typewriter's remaining
   976→               reveal. The slot never shifts mid-turn (messages are
   977→               append-only while streaming). */
   978→            <AssistantMessage
   979→              key={`slot-${i}`}
   980→              message={m}
   981→              isLast={i === messages.length - 1}
   982→              speech={speech}
   983→              onToggleSpeech={(msg) => void toggleSpeech(msg)}
   984→            />
   985→          )
   986→        )}
   987→        <div ref={bottomRef} />
   988→      </div>
   989→    </div>
   990→  );
   991→}
   992→
