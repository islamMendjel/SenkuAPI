    40→export interface TurnEvent {
    41→  type:
    42→    | "meta"
    43→    | "search_status"

    45→    | "delta"
    46→    | "restart"

    53→
    54→export interface TurnParams {
    55→  chatId?: string | null;
    56→  history: ChatMessageData[]; // prior turns (already persisted)
    57→  content: string; // new user message
    58→  model: string;
    59→  search: boolean;
    60→  temporary: boolean; // temporary chats are never persisted
    61→  persistUser?: boolean; // false when regenerating after an edit (message already updated)
    62→  /** Effort dial. "medium" (default) = byte-identical legacy
    63→   *  behavior; "fast" routes to the fastest pooled engine with a
    64→   *  brevity directive; "deep" routes to the strongest pooled
    65→   *  engine with thinking + multi-round audited retrieval. */
    66→  effort?: EffortLevel;
    67→  /** Files the gateway ALREADY read (extraction happened at pick
    68→   *  time). Their text is appended to the request context — never
    69→   *  to the stored message content — and persisted on the user
    70→   *  message row so follow-up turns and reloads keep the context. */
    71→  attachments?: AttachmentRecord[];
    72→  /** One-shot response style (Run 17 slash commands): "code" →
    73→   *  code-focused directive, "concise" → ultra-brief directive.
    74→   *  Appended to the system prompt (and thus the cache key) — the
    75→   *  /code and /concise variants of one question cache separately. */
    76→  style?: TurnStyle;
    77→  /** Fires when the CLIENT disconnects (Stop button, tab close).
    78→   *  The upstream generation is cancelled at the source, the
    79→   *  partial reply is persisted, and the truncated answer NEVER
    80→   *  enters the response cache. */
    81→  signal?: AbortSignal;
    82→  emit: (ev: TurnEvent) => void;
    83→}

    85→export interface TurnResult {
    86→  chatId: string | null;
    87→  userMessageId: string | null;
    88→  assistantMessageId: string | null;
    89→  title: string | null;
    90→  sources: SearchSource[];
    91→  content: string;
    92→  cached: boolean;
    93→  tokens: number; // effective (multiplied) tokens billed — 0 when cached
    94→  rawTokens: number;
    95→  servedBy: "provider" | "zai-cloud" | "cache";
    96→  keyLabel: string | null;
    97→  providerName: string;
    98→  model: string;
    99→  /** which effort level served this turn */
   100→  effort: EffortLevel;
   101→  /** true when the token count is a chars/4 estimate because the
   102→   *  upstream did not report usage (Run 17 — the UI marks it) */
   103→  tokensEstimated: boolean;
   104→  /** true when search was requested but every engine failed */
   105→  searchFailed: boolean;
   106→  /** true when the client disconnected mid-generation — content is
   107→   *  the partial text that actually streamed */
   108→  aborted?: boolean;
   109→}
   110→
   111→export function deriveTitle(content: string): string {
   112→  const clean = content
   113→    .replace(/\s+/g, " ")
   114→    .replace(/[^\p{L}\p{N} ?!.,'’-]/gu, "")
   115→    .trim();
   116→  const words = clean.split(" ").filter(Boolean).slice(0, 6).join(" ");
   117→  return (words || "New chat").slice(0, 60);
   118→}
   119→
   120→function monthKey(d = new Date()): string {
   121→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;
   122→}
   123→
   124→/** History rows keep their attachment extracts — a follow-up
   125→ *  question ("what was the revenue figure?") must still see the
   126→ *  file from the earlier turn, exactly like the incumbents. */
   127→function expandHistory(history: ChatMessageData[]): ChatMessageData[] {
   128→  return history.map((m) =>
   129→    m.role === "user" && m.attachments?.length
   130→      ? { ...m, content: m.content + attachmentContextBlock(m.attachments) }
   131→      : m
   132→  );
   133→}

   135→export async function runChatTurn(params: TurnParams): Promise<TurnResult> {
   136→  const { emit } = params;
   137→  const provider = await getActiveProvider();
   138→  const providerModels = JSON.parse(provider.models || "[]") as string[];
   139→  const requestedModel =
   140→    params.model || providerModels[0] || "free/glm-5.3";
   141→  const effort: EffortLevel = params.effort ?? "medium";
   142→  // The effort dial routes across the POOL: fast → the fastest
   143→  // engine the provider offers, deep → the strongest. Medium keeps
   144→  // the user's selection untouched (legacy behavior).
   145→  const model = effortModel(providerModels, requestedModel, effort);
   146→  // Untrusted client payload — rebuilt field-by-field server-side.
   147→  const attachments = sanitizeAttachments(params.attachments);
   148→
   149→  // ── 1. Web search ──
   150→  // Medium/fast: strict discipline (≤3 snippets, ≤300 chars), one
   151→  // pass. Deep: multi-round iterative retrieval — plan queries,
   152→  // search the question verbatim, then the planned angles in
   153→  // parallel, merged URL-deduped (≤8 sources), with live stage
   154→  // labels streamed to the client.
   155→  let sources: SearchSource[] = [];
   156→  let searchEngine: string | undefined;
   157→  if (params.search) {
   158→    if (effort === "deep") {
   159→      emit({ type: "search_status", status: "searching", depth: "deep" });
   160→      const outcome = await deepSearchDetailed(params.content, (detail) =>
   161→        emit({ type: "search_status", status: "searching", detail, depth: "deep" })
   162→      );
   163→      sources = outcome.results;
   164→      searchEngine = "deep";
   165→    } else {
   166→      emit({ type: "search_status", status: "searching" });
   167→      const outcome = await webSearchDetailed(params.content);
   168→      sources = outcome.results;
   169→      searchEngine = outcome.source;
   170→    }
   171→    emit({ type: "sources", sources });
   172→    emit({
   173→      type: "search_status",
   174→      status: sources.length ? "ok" : "failed",
   175→      engine: searchEngine,
   176→      depth: effort === "deep" ? "deep" : undefined,
   177→    });
   178→  }
   179→
   180→  // Effort directives append AFTER the base contract. Medium adds
   181→  // NOTHING — the prompt stays byte-identical to pre-dial turns so
   182→  // the 24h response cache still hits. Fast/deep additions also
   183→  // namespace the cache key per effort level for free.
   184→  const effortAddon =
   185→    effort === "fast"
   186→      ? FAST_SYSTEM_ADDON
   187→      : effort === "deep"
   188→        ? sources.length
   189→          ? DEEP_RESEARCH_ADDON
   190→          : DEEP_THINKING_ADDON
   191→        : "";
   192→  // One-shot response style (/code, /concise) — appended AFTER the
   193→  // effort addon. Medium + no style stays byte-identical to legacy
   194→  // turns so the 24h response cache still hits for them.
   195→  const styleAddon =
   196→    params.style === "code"
   197→      ? CODE_STYLE_ADDON
   198→      : params.style === "concise"
   199→        ? CONCISE_STYLE_ADDON
   200→        : "";
   201→  // Run 28 — the user's persistent memory. Fetched once per turn,
   202→  // appended to the BASE prompt so both the live system prompt and
   203→  // the hour-stable cache-key prompt carry it (editing memory
   204→  // naturally invalidates stale cached answers). No memories →
   205→  // empty block → byte-identical prompts → old cache entries still hit.
   206→  const memBlock = memoryBlock(await getMemories());
   207→  const baseSystemPrompt =
   208→    (params.search && sources.length ? SEARCH_SYSTEM_DIRECTIVE : BASE_SYSTEM_PROMPT) +
   209→    (effortAddon ? `\n\n${effortAddon}` : "") +
   210→    (styleAddon ? `\n\n${styleAddon}` : "") +
   211→    (memBlock ? `\n\n${memBlock}` : "");
   212→  // The model always knows the real "now" (user's timezone).
   213→  const systemPrompt = `${baseSystemPrompt}\n\n${liveContextBlock()}`;
   214→
   215→  const finalUserContent =
   216→    params.content +
   217→    attachmentContextBlock(attachments) +
   218→    (sources.length ? searchContextBlock(sources) : "");
   219→
   220→  // ── 2. Exact-match cache (prompt + model + hour-stable system prompt) ──
   221→  // The clock block is hour-stable here so repeat questions within the
   222→  // same hour still hit the cache, while the model sees second-level time.
   223→  const cacheKey = computeCacheKey(
   224→    finalUserContent,
   225→    model,
   226→    `${baseSystemPrompt}\n\n${liveContextBlockHourStable()}`
   227→  );
   228→  const cached = await getCached(cacheKey);
   229→
   230→  // ── 3. Persistence: chat row + user message ──
   231→  let chatId = params.chatId ?? null;
   232→  let userMessageId: string | null = null;
   233→  let title: string | null = null;
   234→
   235→  if (!params.temporary && params.persistUser !== false) {
   236→    if (!chatId) {
   237→      title = deriveTitle(params.content);
   238→      const chat = await db.chat.create({
   239→        data: {
   240→          title,
   241→          model,
   242→          provider: provider.name,
   243→          isTemporary: false,
   244→        },
   245→      });
   246→      chatId = chat.id;
   247→    }
   248→    const userMsg = await db.message.create({
   249→      data: {
   250→        chatId,
   251→        role: "user",
   252→        content: params.content,
   253→        sources: sources.length ? JSON.stringify(sources) : null,
   254→        attachments: attachments.length ? JSON.stringify(attachments) : null,
   255→      },
   256→    });
   257→    userMessageId = userMsg.id;
   258→  } else if (!params.temporary && params.chatId) {
   259→    chatId = params.chatId;
   260→  }
   261→
   262→  emit({
   263→    type: "meta",
   264→    chatId,
   265→    userMessageId,
   266→    title,
   267→    provider: provider.name,
   268→    model,
   269→    effort,
   270→  });
   271→
   272→  // ── 4a. Cache hit → 0 tokens spent ──
   273→  if (cached) {
   274→    // Contract enforcement applies to cached search answers too
   275→    // (older cache entries may predate the confidence-label rule).
   276→    if (sources.length && !confidenceOf(cached.content)) {
   277→      cached.content = await appendConfidenceLabel(cached.content);
   278→    }
   279→    // Cached answers type out live too — the answer never appears
   280→    // in a single shot, even when it cost 0 tokens.
   281→    await revealWords(
   282→      cached.content,
   283→      (text) => emit({ type: "delta", text }),
   284→      6,
   285→      () => params.signal?.aborted === true
   286→    );
   287→    let assistantMessageId: string | null = null;
   288→    if (!params.temporary && chatId) {
   289→      const msg = await db.message.create({
   290→        data: {
   291→          chatId,
   292→          role: "assistant",
   293→          content: cached.content,
   294→          model,
   295→          provider: provider.name,
   296→          keyLabel: "cache",
   297→          tokens: 0,
   298→          cached: true,
   299→        },
   300→      });
   301→      assistantMessageId = msg.id;
   302→      await db.chat.update({
   303→        where: { id: chatId },
   304→        data: { updatedAt: new Date(), model, provider: provider.name },
   305→      });
   306→    }
   307→    // Audit trail: cached hit, zero tokens.
   308→    await db.usageEvent.create({
   309→      data: {
   310→        keyId: null,
   311→        providerId: provider.id,
   312→        chatId,
   313→        model,
   314→        promptTokens: 0,
   315→        completionTokens: 0,
   316→        totalTokens: 0,
   317→        multiplier: provider.tokenMultiplier,
   318→        cached: true,
   319→      },
   320→    });
   321→    // Run 36 correctness fix: this object is now typed as TurnResult
   322→    // and carries `title`, mirroring the non-cached path below — the
   323→    // missing title made every cached turn's result read undefined
   324→    // where the contract (and the main path) promised the value.
   325→    const done: TurnResult = {
   326→      chatId,
   327→      userMessageId,
   328→      assistantMessageId,
   329→      title,
   330→      sources,
   331→      content: cached.content,
   332→      cached: true,
   333→      tokens: 0,
   334→      rawTokens: 0,
   335→      // Cache hits bill zero tokens — nothing was estimated.
   336→      tokensEstimated: false,
   337→      servedBy: "cache",
   338→      keyLabel: "cache",
   339→      providerName: provider.name,
   340→      model,
   341→      effort,
   342→      searchFailed: params.search && sources.length === 0,
   343→    };
   344→    emit({ type: "done", ...done });
   345→    return done;
   346→  }
   347→
   348→  // ── 4b + 5. Health-aware key dispatch, then cloud completion ──
   349→  // The dispatcher round-robins pooled keys, quarantines dead ones
   350→  // (auth failures), retries the next healthy key within the same
   351→  // turn, and only fails over to the Z.ai cloud when every key
   352→  // failed. Hard pool errors (no keys / limiter exhaustion) keep
   353→  // their original coded messages.
   354→  //
   355→  // `signal` is the CLIENT's connection: the moment it drops, the
   356→  // upstream fetch is aborted at the source — the pooled keys stop
   357→  // paying for tokens nobody will ever see.
   358→  const messages: ChatMessageData[] = [
   359→    { role: "system", content: systemPrompt },
   360→    ...expandHistory(params.history).slice(-20),
   361→    { role: "user", content: finalUserContent },
   362→  ];
   363→
   364→  const routed = await dispatchCompletion({
   365→    provider,
   366→    model,
   367→    messages,
   368→    onDelta: (text) => emit({ type: "delta", text }),
   369→    // Mid-stream provider death → the cloud re-emits from scratch;
   370→    // "restart" tells the client to clear the partial text first.
   371→    onRestart: () => emit({ type: "restart" }),
   372→    // Deep effort → the cloud failover THINKS before answering.
   373→    thinking: effort === "deep",
   374→    signal: params.signal,
   375→  });
   376→  const exec = routed.exec;
   377→  // The key that actually served — null when the Z.ai cloud did.
   378→  const key = routed.key;
   379→  // True when the CLIENT disconnected mid-generation: the partial
   380→  // text is final for this turn.
   381→  const aborted = routed.aborted === true;
   382→
   383→  // ── Grounded-answer contract enforcement ──
   384→  // On search turns the confidence label is mandatory. The directive
   385→  // asks for it, but model compliance is probabilistic — when the
   386→  // answer lacks the label, one cheap corrective pass appends it so
   387→  // the contract holds deterministically (cache fills afterwards,
   388→  // so corrected answers are reused corrected). Aborted turns skip
   389→  // it: no corrective cloud spend on an answer nobody is watching.
   390→  if (!aborted && sources.length && !confidenceOf(exec.content)) {
   391→    exec.content = await appendConfidenceLabel(exec.content);
   392→  }
   393→
   394→  // ── 6. Usage accounting (multiplier applies to effective totals) ──
   395→  // Only the key that ACTUALLY served the response is billed — a
   396→  // dead key that failed over to the cloud never gets charged.
   397→  const rawTotal =
   398→    exec.totalTokens ||
   399→    exec.promptTokens + exec.completionTokens ||
   400→    estimateTokens(finalUserContent) + estimateTokens(exec.content);
   401→  // Honesty marker: without upstream usage stats the counts are
   402→  // chars/4 estimates — the UI must never present them as exact.
   403→  const tokensEstimated = exec.usageReported !== true && rawTotal > 0;
   404→
   405→  await db.usageEvent.create({
   406→    data: {
   407→      keyId: key?.id ?? null,
   408→      providerId: provider.id,
   409→      chatId,
   410→      model,
   411→      promptTokens: exec.promptTokens,
   412→      completionTokens: exec.completionTokens,
   413→      totalTokens: rawTotal,
   414→      multiplier: provider.tokenMultiplier,
   415→      cached: false,
   416→    },
   417→  });
   418→
   419→  // Roll the serving key's monthly accumulator forward (monthKey
   420→  // flip safe). Skipped entirely on cloud failover — no key served.
   421→  if (key) {
   422→    const now = new Date();
   423→    const mk = monthKey(now);
   424→    const monthAgg = await db.usageEvent.aggregate({
   425→      _sum: { totalTokens: true },
   426→      where: {
   427→        keyId: key.id,
   428→        createdAt: { gte: new Date(now.getFullYear(), now.getMonth(), 1) },
   429→      },
   430→    });
   431→    await db.apiKey.update({
   432→      where: { id: key.id },
   433→      data: {
   434→        lastUsedAt: now,
   435→        monthKey: mk,
   436→        tokensUsedThisMonth: monthAgg._sum.totalTokens ?? rawTotal,
   437→      },
   438→    });
   439→  }
   440→
   441→  // ── 7. Cache fill (24h TTL) ──
   442→  // A turn the user ABORTED never fills the cache: a truncated
   443→  // answer must never be served to a future identical question.
   444→  if (!aborted) {
   445→    await putCached({
   446→      cacheKey,
   447→      model,
   448→      provider: provider.name,
   449→      content: exec.content,
   450→      tokens: rawTotal,
   451→    });
   452→  }
   453→
   454→  // ── 8. Persist assistant message ──
   455→  // keyLabel is only recorded when a pooled key actually served the
   456→  // response — a cloud-failover answer never claims a key it didn't use.
   457→  const effectiveTokens = Math.round(rawTotal * provider.tokenMultiplier);
   458→  const servedKeyLabel = key?.label ?? null;
   459→  let assistantMessageId: string | null = null;
   460→  // The partial of an aborted turn is persisted too — the timeline
   461→  // keeps exactly what streamed before the user pressed stop (and
   462→  // an empty partial creates no ghost message).
   463→  if (!params.temporary && chatId && exec.content.trim()) {
   464→    const msg = await db.message.create({
   465→      data: {
   466→        chatId,
   467→        role: "assistant",
   468→        content: exec.content,
   469→        model,
   470→        provider: provider.name,
   471→        keyLabel: servedKeyLabel,
   472→        tokens: effectiveTokens,
   473→        tokensEstimated,
   474→        cached: false,
   475→        effort,
   476→      },
   477→    });
   478→    assistantMessageId = msg.id;
   479→    await db.chat.update({
   480→      where: { id: chatId },
   481→      data: { updatedAt: new Date(), model, provider: provider.name },
   482→    });
   483→  }
   484→
   485→  const done: TurnResult = {
   486→    chatId,
   487→    userMessageId,
   488→    assistantMessageId,
   489→    title,
   490→    sources,
   491→    content: exec.content,
   492→    cached: false,
   493→    tokens: effectiveTokens,
   494→    rawTokens: rawTotal,
   495→    servedBy: exec.servedBy,
   496→    keyLabel: servedKeyLabel,
   497→    providerName: provider.name,
   498→    model,
   499→    effort,
   500→    tokensEstimated,
   501→    searchFailed: params.search && sources.length === 0,
   502→    ...(aborted ? { aborted: true } : {}),
   503→  };
   504→  emit({ type: "done", ...done });
   505→  if (aborted) {
   506→    console.log(
   507→      `[gateway] turn cancelled by client — persisted ${exec.content.length} chars, ` +
   508→        `${effectiveTokens} tokens billed, cache skipped`
   509→    );
   510→  }
   511→  return done;
   512→}

   514→// ── Duel turn ─────────────────────────────────────────────────
   515→// TWO models race in parallel on one question: shared web search,
   516→// shared chat + user-message persistence, per-lane cache lookup,
   517→// health-aware dispatch, grounded-contract enforcement, usage
   518→// accounting and cache fill. Both lanes are billed honestly for
   519→// what they actually served. The WINNER is not persisted here —
   520→// the client calls POST /api/chat/duel/pick once the user chooses;
   521→// that endpoint stores the winner as the assistant message and the
   522→// loser as its runner-up reference.
   523→
   524→export interface DuelLaneResult {
   525→  lane: "a" | "b";
   526→  model: string;
   527→  content: string;
   528→  tokens: number; // effective (multiplied) tokens billed — 0 when cached
   529→  rawTokens: number;
   530→  cached: boolean;
   531→  servedBy: "provider" | "zai-cloud" | "cache" | null; // null on error
   532→  keyLabel: string | null;
   533→  error: string | null;
   534→}
   535→
   536→export interface DuelParams {
   537→  chatId?: string | null;
   538→  history: ChatMessageData[];
   539→  content: string;
   540→  model: string; // lane A — the user's selected model
   541→  challenger: string; // lane B
   542→  search: boolean;
   543→  temporary: boolean;
   544→  persistUser?: boolean;
   545→  /** Effort affects the duel's SHARED retrieval (deep → multi-round
   546→   *  search + thinking on the cloud path) but never the lane models
   547→   *  — a duel is explicitly about the two engines the user chose. */
   548→  effort?: EffortLevel;
   549→  /** Files the gateway read — shared by both lanes (the same
   550→   *  material, the same rules as a normal turn). */
   551→  attachments?: AttachmentRecord[];
   552→  /** Client disconnect cancels BOTH racing lanes at the source. */
   553→  signal?: AbortSignal;
   554→  emit: (ev: TurnEvent) => void;
   555→}
   556→
   557→export interface DuelTurnResult {
   558→  chatId: string | null;
   559→  userMessageId: string | null;
   560→  title: string | null;
   561→  sources: SearchSource[];
   562→  lanes: { a: DuelLaneResult; b: DuelLaneResult };
   563→  bothFailed: boolean;
   564→}
   565→
   566→export async function runDuelTurn(
   567→  params: DuelParams
   568→): Promise<DuelTurnResult> {
   569→  const { emit } = params;
   570→  const provider = await getActiveProvider();
   571→  const providerModels = JSON.parse(provider.models || "[]") as string[];
   572→  const modelA = params.model || providerModels[0] || "free/glm-5.3";
   573→  const modelB = params.challenger;
   574→  const effort: EffortLevel = params.effort ?? "medium";
   575→  const attachments = sanitizeAttachments(params.attachments);
   576→
   577→  if (!modelB || modelB === modelA) {
   578→    throw new Error("Duel needs two different models");
   579→  }
   580→  if (
   581→    providerModels.length &&
   582→    (!providerModels.includes(modelA) || !providerModels.includes(modelB))
   583→  ) {
   584→    throw new Error("Both duel models must be available on the active provider");
   585→  }
   586→
   587→  // ── 1. Shared web search — both lanes reason from the SAME retrieval ──
   588→  let sources: SearchSource[] = [];
   589→  if (params.search) {
   590→    if (effort === "deep") {
   591→      emit({ type: "search_status", status: "searching", depth: "deep" });
   592→      const outcome = await deepSearchDetailed(params.content, (detail) =>
   593→        emit({ type: "search_status", status: "searching", detail, depth: "deep" })
   594→      );
   595→      sources = outcome.results;
   596→    } else {
   597→      emit({ type: "search_status", status: "searching" });
   598→      const outcome = await webSearchDetailed(params.content);
   599→      sources = outcome.results;
   600→    }
   601→    emit({ type: "sources", sources });
   602→    emit({
   603→      type: "search_status",
   604→      status: sources.length ? "ok" : "failed",
   605→      depth: effort === "deep" ? "deep" : undefined,
   606→    });
   607→  }
   608→
   609→  // Run 28 — duels run with the same persistent memory as normal
   610→  // turns (appended to the base prompt → both the live prompt and
   611→  // each lane's cache key include it; empty memory = byte-identical).
   612→  const memBlock = memoryBlock(await getMemories());
   613→  const baseSystemPrompt =
   614→    (params.search && sources.length ? SEARCH_SYSTEM_DIRECTIVE : BASE_SYSTEM_PROMPT) +
   615→    (memBlock ? `\n\n${memBlock}` : "");
   616→  const systemPrompt = `${baseSystemPrompt}\n\n${liveContextBlock()}`;
   617→  const finalUserContent =
   618→    params.content +
   619→    attachmentContextBlock(attachments) +
   620→    (sources.length ? searchContextBlock(sources) : "");
   621→
   622→  // ── 2. Shared persistence: chat row + user message (same as a
   623→  //      normal turn — the winner is appended later, at pick time) ──
   624→  let chatId = params.chatId ?? null;
   625→  let userMessageId: string | null = null;
   626→  let title: string | null = null;
   627→
   628→  if (!params.temporary && params.persistUser !== false) {
   629→    if (!chatId) {
   630→      title = deriveTitle(params.content);
   631→      const chat = await db.chat.create({
   632→        data: {
   633→          title,
   634→          model: modelA,
   635→          provider: provider.name,
   636→          isTemporary: false,
   637→        },
   638→      });
   639→      chatId = chat.id;
   640→    }
   641→    const userMsg = await db.message.create({
   642→      data: {
   643→        chatId,
   644→        role: "user",
   645→        content: params.content,
   646→        sources: sources.length ? JSON.stringify(sources) : null,
   647→        attachments: attachments.length ? JSON.stringify(attachments) : null,
   648→      },
   649→    });
   650→    userMessageId = userMsg.id;
   651→  } else if (!params.temporary && params.chatId) {
   652→    chatId = params.chatId;
   653→  }
   654→
   655→  emit({
   656→    type: "meta",
   657→    chatId,
   658→    userMessageId,
   659→    title,
   660→    provider: provider.name,
   661→    model: modelA,
   662→    effort,
   663→  });
   664→  emit({ type: "duel_start", a: modelA, b: modelB });
   665→
   666→  const messages: ChatMessageData[] = [
   667→    { role: "system", content: systemPrompt },
   668→    ...expandHistory(params.history).slice(-20),
   669→    { role: "user", content: finalUserContent },
   670→  ];
   671→
   672→  // ── 3. The race: two independent lanes over the shared context ──
   673→  const runLane = async (
   674→    lane: "a" | "b",
   675→    model: string
   676→  ): Promise<DuelLaneResult> => {
   677→    // Per-lane exact-match cache (the cache key includes the model).
   678→    const cacheKey = computeCacheKey(
   679→      finalUserContent,
   680→      model,
   681→      `${baseSystemPrompt}\n\n${liveContextBlockHourStable()}`
   682→    );
   683→    const cached = await getCached(cacheKey);
   684→    if (cached) {
   685→      // Contract enforcement applies to cached search answers too.
   686→      if (sources.length && !confidenceOf(cached.content)) {
   687→        cached.content = await appendConfidenceLabel(cached.content);
   688→      }
   689→      // Cached lane answers type out live too (word by word),
   690→      // exactly like a fresh lane.
   691→      await revealWords(
   692→        cached.content,
   693→        (text) => emit({ type: "delta", lane, text }),
   694→        6,
   695→        () => params.signal?.aborted === true
   696→      );
   697→      await db.usageEvent.create({
   698→        data: {
   699→          keyId: null,
   700→          providerId: provider.id,
   701→          chatId,
   702→          model,
   703→          promptTokens: 0,
   704→          completionTokens: 0,
   705→          totalTokens: 0,
   706→          multiplier: provider.tokenMultiplier,
   707→          cached: true,
   708→        },
   709→      });
   710→      const result: DuelLaneResult = {
   711→        lane,
   712→        model,
   713→        content: cached.content,
   714→        tokens: 0,
   715→        rawTokens: 0,
   716→        cached: true,
   717→        servedBy: "cache",
   718→        keyLabel: "cache",
   719→        error: null,
   720→      };
   721→      emit({ type: "lane_done", ...result });
   722→      return result;
   723→    }
   724→
   725→    try {
   726→      const routed = await dispatchCompletion({
   727→        provider,
   728→        model,
   729→        messages,
   730→        onDelta: (text) => emit({ type: "delta", lane, text }),
   731→        // Mid-stream death in THIS lane → the cloud re-emits this
   732→        // lane from scratch; clear only this lane's partial text.
   733→        onRestart: () => emit({ type: "restart", lane }),
   734→        // Deep effort → both lanes' cloud failover thinks too.
   735→        thinking: effort === "deep",
   736→        // Client disconnect cancels this lane's upstream too.
   737→        signal: params.signal,
   738→      });
   739→      const exec = routed.exec;
   740→      const key = routed.key;
   741→      const aborted = routed.aborted === true;
   742→
   743→      if (!aborted && sources.length && !confidenceOf(exec.content)) {
   744→        exec.content = await appendConfidenceLabel(exec.content);
   745→      }
   746→
   747→      // Per-lane billing — identical rules to a normal turn: only
   748→      // the key that actually served is charged; cloud lanes bill
   749→      // no key.
   750→      const rawTotal =
   751→        exec.totalTokens ||
   752→        exec.promptTokens + exec.completionTokens ||
   753→        estimateTokens(finalUserContent) + estimateTokens(exec.content);
   754→      await db.usageEvent.create({
   755→        data: {
   756→          keyId: key?.id ?? null,
   757→          providerId: provider.id,
   758→          chatId,
   759→          model,
   760→          promptTokens: exec.promptTokens,
   761→          completionTokens: exec.completionTokens,
   762→          totalTokens: rawTotal,
   763→          multiplier: provider.tokenMultiplier,
   764→          cached: false,
   765→        },
   766→      });
   767→      if (key) {
   768→        const now = new Date();
   769→        const mk = monthKey(now);
   770→        const monthAgg = await db.usageEvent.aggregate({
   771→          _sum: { totalTokens: true },
   772→          where: {
   773→            keyId: key.id,
   774→            createdAt: { gte: new Date(now.getFullYear(), now.getMonth(), 1) },
   775→          },
   776→        });
   777→        await db.apiKey.update({
   778→          where: { id: key.id },
   779→          data: {
   780→            lastUsedAt: now,
   781→            monthKey: mk,
   782→            tokensUsedThisMonth: monthAgg._sum.totalTokens ?? rawTotal,
   783→          },
   784→        });
   785→      }
   786→
   787→      // An aborted lane never fills the cache (truncated text must
   788→      // never be served to a future identical question).
   789→      if (!aborted) {
   790→        await putCached({
   791→          cacheKey,
   792→          model,
   793→          provider: provider.name,
   794→          content: exec.content,
   795→          tokens: rawTotal,
   796→        });
   797→      }
   798→
   799→      const result: DuelLaneResult = {
   800→        lane,
   801→        model,
   802→        content: exec.content,
   803→        tokens: Math.round(rawTotal * provider.tokenMultiplier),
   804→        rawTokens: rawTotal,
   805→        cached: false,
   806→        servedBy: exec.servedBy,
   807→        keyLabel: key?.label ?? null,
   808→        error: null,
   809→      };
   810→      emit({ type: "lane_done", ...result });
   811→      return result;
   812→    } catch (e) {
   813→      // One lane failing must never kill the other — the duel
   814→      // degrades to a single-lane answer the user can still keep.
   815→      const result: DuelLaneResult = {
   816→        lane,
   817→        model,
   818→        content: "",
   819→        tokens: 0,
   820→        rawTokens: 0,
   821→        cached: false,
   822→        servedBy: null,
   823→        keyLabel: null,
   824→        error: (e as Error).message || "Generation failed",
   825→      };
   826→      emit({ type: "lane_done", ...result });
   827→      return result;
   828→    }
   829→  };
   830→
   831→  const [a, b] = await Promise.all([
   832→    runLane("a", modelA),
   833→    runLane("b", modelB),
   834→  ]);
   835→
   836→  if (a.error && b.error) {
   837→    emit({
   838→      type: "error",
   839→      message: `Both models failed — A (${modelA}): ${a.error} · B (${modelB}): ${b.error}`,
   840→    });
   841→    return { chatId, userMessageId, title, sources, lanes: { a, b }, bothFailed: true };
   842→  }
   843→
   844→  // No assistant message is persisted here — the client picks the
   845→  // winner and /api/chat/duel/pick stores it (with the runner-up).
   846→  emit({
   847→    type: "done",
   848→    chatId,
   849→    userMessageId,
   850→    assistantMessageId: null,
   851→    duel: { a, b },
   852→    // Top-level fields mirror lane A for simple/legacy consumers.
   853→    content: a.content,
   854→    cached: a.cached,
   855→    tokens: a.tokens,
   856→    rawTokens: a.rawTokens,
   857→    servedBy: a.servedBy,
   858→    keyLabel: a.keyLabel,
   859→    providerName: provider.name,
   860→    model: modelA,
   861→    sources,
   862→    searchFailed: params.search && sources.length === 0,
   863→  });
   864→  return { chatId, userMessageId, title, sources, lanes: { a, b }, bothFailed: false };
   865→}
