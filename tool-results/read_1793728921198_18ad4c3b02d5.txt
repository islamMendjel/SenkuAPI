    82→/** The /-token under the caret, if any. The "/" must sit at the very
    83→ *  start of the input or right after whitespace — typing "/" mid-word
    84→ *  ("a/b") or inside a URL ("https://x") never triggers the popover. */
    85→function activeSlashToken(
    86→  value: string,
    87→  caret: number
    88→): { query: string; start: number } | null {
    89→  const upto = value.slice(0, caret);
    90→  const m = /(^|\s)\/([\w-]{0,20})$/.exec(upto);
    91→  if (!m) return null;
    92→  return { query: m[2].toLowerCase(), start: caret - m[2].length - 1 };
    93→}

    95→function formatChars(n: number): string {
    96→  if (n >= 1000) return `${(n / 1000).toFixed(1)}k chars`;
    97→  return `${n} chars`;
