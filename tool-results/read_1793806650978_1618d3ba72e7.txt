    35→function cooldownLeft(coolingUntil: string | null | undefined, now: number) {
    36→  if (!coolingUntil) return null;
    37→  const ms = new Date(coolingUntil).getTime() - now;
    38→  return ms > 0 ? ms : null;
    39→}
    40→
    41→function fmtCountdown(ms: number) {
    42→  const s = Math.ceil(ms / 1000);
    43→  if (s >= 3600) return `${Math.floor(s / 3600)}h${Math.floor((s % 3600) / 60)}m`;
    44→  return `${Math.floor(s / 60)}:${String(s % 60).padStart(2, "0")}`;
    45→}

    47→// Run 41 — THE LIVING POOL: one honest phrase per key, built from
    48→// the prober's own records. Spoken by tooltips and screen readers
    49→// alike — the state, the evidence (latency / error), and the
    50→// per-boot serve ledger.
    51→function livePhrase(k: KeyView): string {
    52→  const live = k.live;
    53→  if (!live) return "";
    54→  const s =
    55→    live.state === "serving"
    56→      ? `probe-verified live${live.lastLatencyMs != null ? ` · ${live.lastLatencyMs}ms probe` : ""}`
    57→      : live.state === "checkin"
    58→        ? "needs its daily dashboard check-in (402) — it rejoins the pool automatically once checked in"
    59→        : live.state === "cooling"
    60→          ? live.lastProbeStatus === 401 || live.lastProbeStatus === 403
    61→            ? "credential rejected (suspended or rotated) — replace the key; auto re-probed every 10 min"
    62→            : "breaker cooldown"
    63→          : live.state === "down"
    64→            ? `not responding${live.lastError ? ` (${live.lastError})` : ""}`
    65→            : "awaiting first probe";
    66→  const serves =
    67→    live.serves > 0
    68→      ? ` · ${live.serves} serve${live.serves === 1 ? "" : "s"} this boot`
    69→      : "";
    70→  return ` · ${s}${serves}`;
    71→}

   345→              }.`
   346→          )
   347→          .join(" ")}
   348→      </span>
   349→    </div>
   350→  );
   351→}

   355→function ProviderCard({
   356→  provider,
   357→  onSwitch,
   358→  now,
   359→}: {
   360→  provider: ProviderView;
   361→  onSwitch: (id: string) => Promise<void>;
   362→  now: number;
   363→}) {
   364→  const { addKey, deleteKey, toggleKey, probeKey } = useHermes();
   365→  const { toast } = useToast();
   366→  const [switching, setSwitching] = useState(false);
   367→  const [label, setLabel] = useState("");
   368→  const [key, setKey] = useState("");
   369→  const [adding, setAdding] = useState(false);
   370→
   371→  const doSwitch = async () => {
   372→    setSwitching(true);
   373→    try {
   374→      await onSwitch(provider.id);
   375→      toast({
   376→        description: `${provider.name} is now the active provider.`,
   377→      });
   378→    } catch (e) {
   379→      toast({ description: (e as Error).message, variant: "destructive" });
   380→    } finally {
   381→      setSwitching(false);
   382→    }
   383→  };
   384→
   385→  const doAddKey = async () => {
   386→    if (!key.trim()) return;
   387→    setAdding(true);
   388→    try {
   389→      await addKey(provider.id, label.trim(), key.trim());
   390→      setLabel("");
   391→      setKey("");
   392→      toast({ description: "Key added to the pool." });
   393→    } catch (e) {
   394→      toast({ description: (e as Error).message, variant: "destructive" });
   395→    } finally {
   396→      setAdding(false);
   397→    }
   398→  };
   399→
   400→  return (
   401→    <div
   402→      className={`min-w-0 rounded-xl border p-3.5 transition-colors ${
   403→        provider.isActive
   404→          ? "border-emerald-500/40 bg-emerald-500/[0.04]"
   405→          : "border-zinc-800 bg-[#1d1d20]"
   406→      }`}
   407→    >
   408→      <div className="flex flex-wrap items-center gap-2">
   409→        <span className="flex items-center gap-2 text-[14px] font-semibold text-white">
   410→          {provider.isActive && (
   411→            <span className="hermes-pulse h-2 w-2 rounded-full bg-emerald-400" />
   412→          )}
   413→          {provider.name}
   414→        </span>
   415→        <span className="rounded-full bg-zinc-800/80 px-2 py-0.5 text-[10.5px] font-medium text-zinc-300">
   416→          {provider.hasTokenLimit
   417→            ? `${formatCompact(provider.tokenLimitPerKey)} tokens / key / mo`
   418→            : "Unlimited"}
   419→        </span>
   420→        <span className="rounded-full bg-zinc-800/80 px-2 py-0.5 text-[10.5px] font-medium text-zinc-300">
   421→          ×{provider.tokenMultiplier.toFixed(1)} multiplier
   422→        </span>
   423→        <span className="ml-auto">
   424→          {provider.isActive ? (
   425→            <span className="flex items-center gap-1 rounded-full border border-emerald-400/40 bg-emerald-400/10 px-3 py-1 text-[12px] font-medium text-emerald-300">
   426→              <ShieldCheck className="h-3.5 w-3.5" />
   427→              Active
   428→            </span>
   429→          ) : (
   430→            <button
   431→              onClick={doSwitch}
   432→              disabled={switching}
   433→              className="flex items-center gap-1.5 rounded-full bg-white px-3.5 py-1 text-[12px] font-medium text-black transition-colors hover:bg-zinc-200 disabled:opacity-60"
   434→            >
   435→              {switching ? (
   436→                <Loader2 className="h-3.5 w-3.5 animate-spin" />
   437→              ) : (
   438→                <Check className="h-3.5 w-3.5" />
   439→              )}
   440→              Switch
   441→            </button>
   442→          )}
   443→        </span>
   444→      </div>
   445→
   446→      <p className="mt-1.5 truncate font-mono text-[11px] text-zinc-500">
   447→        {provider.baseUrl ?? "no base URL"} · {provider.models.length} models ·{" "}
   448→        {formatFull(provider.totalTokens)} lifetime tokens
   449→      </p>
   450→
   451→      {/* Run 35 — the live ring: only the ACTIVE provider rotates. */}
   452→      {provider.isActive && <Orrery provider={provider} now={now} />}
   453→
   454→      {/* keys */}
   455→      <div className="mt-3 space-y-1.5">
   456→        {provider.keys.map((k) => (
   457→          <KeyRow
   458→            key={k.id}
   459→            k={k}
   460→            now={now}
   461→            onDelete={deleteKey}
   462→            onToggle={toggleKey}
   463→            onProbe={probeKey}
   464→          />
   465→        ))}
   466→
   467→        <div className="flex gap-2">
   468→          <Input
   469→            value={label}
   470→            onChange={(e) => setLabel(e.target.value)}
   471→            placeholder="Label (optional)"
   472→            className="h-8 w-28 shrink-0 rounded-lg border-zinc-800 bg-[#18181b] text-[12px] text-zinc-200 placeholder:text-zinc-600 focus-visible:ring-zinc-600 sm:w-32"
   473→          />
   474→          <Input
   475→            value={key}
   476→            onChange={(e) => setKey(e.target.value)}
   477→            placeholder="Paste API key…"
   478→            className="h-8 min-w-0 flex-1 rounded-lg border-zinc-800 bg-[#18181b] font-mono text-[12px] text-zinc-200 placeholder:text-zinc-600 focus-visible:ring-zinc-600"
   479→            onKeyDown={(e) => {
   480→              if (e.key === "Enter") void doAddKey();
   481→            }}
   482→          />
   483→          <button
   484→            onClick={doAddKey}
   485→            disabled={adding || !key.trim()}
   486→            aria-label="Add key"
   487→            className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg border border-zinc-700 text-zinc-300 transition-colors hover:bg-white/5 disabled:opacity-40"
   488→          >
   489→            {adding ? (
   490→              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   491→            ) : (
   492→              <Plus className="h-4 w-4" />
   493→            )}
   494→          </button>
   495→        </div>
   496→      </div>
   497→    </div>
   498→  );
   499→}

   503→  now,
   504→  onDelete,

   506→  onProbe,
   507→}: {
   508→  k: KeyView;
   509→  now: number;
   510→  onDelete: (id: string) => Promise<void>;
   511→  onToggle: (id: string, isActive: boolean) => Promise<void>;
   512→  onProbe: (id: string) => Promise<ProbeVerdict>;
   513→}) {
   514→  const { toast } = useToast();
   515→  const [probing, setProbing] = useState(false);
   516→  const cooling = cooldownLeft(k.coolingUntil, now);
   517→  const blocked = k.blockedTier ?? null;
   518→  const live = k.live ?? null;
   519→  // Run 41 — the row speaks the key's live truth, with a tone per
   520→  // verdict: amber = an owner ACTION (daily check-in), rose =
   521→  // cooling / not responding, emerald = probe-verified serving.
   522→  let state: string | null = null;
   523→  let stateCls = "";
   524→  if (!k.isActive) {
   525→    state = "off";
   526→  } else if (live?.state === "checkin") {
   527→    state = "needs daily check-in — self-heals once checked in";
   528→    stateCls = " text-amber-300/90";
   529→  } else if (cooling !== null) {
   530→    state =
   531→      live?.lastProbeStatus === 401 || live?.lastProbeStatus === 403
   532→        ? "credential rejected — replace the key"
   533→        : `cooling ${fmtCountdown(cooling)}`;
   534→    stateCls = " text-rose-300/80";
   535→  } else if (live?.state === "down") {
   536→    state = "not responding";
   537→    stateCls = " text-rose-300/80";
   538→  } else if (blocked) {
   539→    state = TIER_TAG[blocked];
   540→    stateCls = " text-amber-200/70";
   541→  } else if (live?.state === "serving") {
   542→    state = `serving${
   543→      live.lastLatencyMs != null ? ` · ${live.lastLatencyMs}ms probe` : ""
   544→    }${
   545→      live.serves > 0
   546→        ? ` · ${live.serves} serve${live.serves === 1 ? "" : "s"} this boot`
   547→        : ""
   548→    }`;
   549→    stateCls = " text-emerald-300/80";
   550→  } else if (live?.state === "unproven") {
   551→    state = "probing…";
   552→    stateCls = " text-zinc-500";
   553→  }
   554→
   555→  // Run 42 — the probe speaks its verdict out loud: live (with the
   556→  // latency it just measured), still check-in-blocked, credential
   557→  // rejected, or whatever the upstream actually said. No silent
   558→  // failures — every click lands somewhere the owner can act on.
   559→  const handleProbe = async () => {
   560→    setProbing(true);
   561→    try {
   562→      const r = await onProbe(k.id);
   563→      if (r.ok) {
   564→        toast({
   565→          description: `${k.label} verified live · ${r.latencyMs}ms probe`,
   566→        });
   567→      } else if (r.status === 402) {
   568→        toast({
   569→          description:
   570→            "Still demands its daily check-in (402). Complete it on the dashboard, then probe again.",
   571→          variant: "destructive",
   572→        });
   573→      } else if (r.status === 401 || r.status === 403) {
   574→        toast({
   575→          description: "Credential rejected (401/403) — replace the key.",
   576→          variant: "destructive",
   577→        });
   578→      } else {
   579→        toast({
   580→          description: `Probe failed · ${r.error || `status ${r.status}`}`,
   581→          variant: "destructive",
   582→        });
   583→      }
   584→    } catch (e) {
   585→      toast({ description: (e as Error).message, variant: "destructive" });
   586→    } finally {
   587→      setProbing(false);
   588→    }
   589→  };
   590→  return (
   591→    <div className="flex items-center gap-2.5 rounded-lg border border-zinc-800/80 bg-[#18181b] px-2.5 py-2">
   592→      <div className="min-w-0 flex-1">
   593→        <p className="truncate font-mono text-[12px] text-zinc-300">
   594→          {k.masked}
   595→        </p>
   596→        <p className="text-[10.5px] text-zinc-600">
   597→          {k.label} · {formatFull(k.tokensUsedThisMonth)} tokens this month
   598→          {k.lastUsedAt
   599→            ? ` · used ${new Date(k.lastUsedAt).toLocaleDateString()}`
   600→            : " · never used"}
   601→          {state && (
   602→            <span className={stateCls}> · {state}</span>
   603→          )}
   604→        </p>
   605→      </div>
   606→      {/* Run 42 — self-service healing: one liveness probe NOW.
   607→          After a dashboard check-in or a key swap, the owner clicks
   608→          and watches the seat flip emerald — no waiting for the
   609→          prober's tick or the breaker's cooldown. Amber-ring when
   610→          the key awaits its check-in: that's the "I checked in —
   611→          re-verify now" button. */}
   612→      <button
   613→        onClick={() => void handleProbe()}
   614→        disabled={probing || !k.isActive}
   615→        aria-label={`Probe ${k.label} now`}
   616→        title="Probe now — verify this key is live instantly"
   617→        className={`flex h-7 w-7 shrink-0 items-center justify-center rounded-md transition-colors disabled:opacity-40 ${
   618→          k.live?.state === "checkin"
   619→            ? "border border-amber-400/40 text-amber-300 hover:bg-amber-400/10"
   620→            : "text-zinc-500 hover:bg-white/5 hover:text-zinc-300"
   621→        }`}
   622→      >
   623→        {probing ? (
   624→          <Loader2 className="h-3.5 w-3.5 animate-spin" />
   625→        ) : (
   626→          <Zap className="h-3.5 w-3.5" />
   627→        )}
   628→      </button>
   629→      <Switch
   630→        checked={k.isActive}
   631→        onCheckedChange={(v) => void onToggle(k.id, v)}
   632→        aria-label={`Toggle key ${k.label}`}
   633→      />
   634→      <button
   635→        onClick={() => void onDelete(k.id)}
   636→        aria-label={`Delete key ${k.label}`}
   637→        className="flex h-7 w-7 items-center justify-center rounded-md text-zinc-500 transition-colors hover:bg-red-400/10 hover:text-red-400"
   638→      >
   639→        <Trash2 className="h-3.5 w-3.5" />
   640→      </button>
   641→    </div>
   642→  );
   643→}

   647→export function ProvidersDialog({
   648→  open,
   649→  onOpenChange,
   650→}: {
   651→  open: boolean;
   652→  onOpenChange: (v: boolean) => void;
   653→}) {
   654→  const { providers, switchProvider, refreshUsage } = useHermes();
   655→  // Live ring: while the dialog is open, poll the store's own
   656→  // /api/usage refresh so the orrery advances in real time as the
   657→  // gateway serves turns. User mutations (switch/add/toggle/delete)
   658→  // already write through the same store — one source of truth,
   659→  // zero staleness windows.
   660→  const [now, setNow] = useState(() => Date.now());
   661→
   662→  useEffect(() => {
   663→    if (!open) return;
   664→    let alive = true;
   665→    const tick = () => setNow(Date.now());
   666→    const poll = () => {
   667→      if (alive) void refreshUsage();
   668→    };
   669→    poll();
   670→    const poller = setInterval(poll, 4000);
   671→    const ticker = setInterval(tick, 1000);
   672→    return () => {
   673→      alive = false;
   674→      clearInterval(poller);
   675→      clearInterval(ticker);
   676→    };
   677→  }, [open, refreshUsage]);
   678→
   679→  return (
   680→    <Dialog open={open} onOpenChange={onOpenChange}>
   681→      <DialogContent className="max-h-[85vh] overflow-y-auto rounded-2xl border-zinc-800 bg-[#212121] p-5 text-zinc-200 sm:max-w-lg">
   682→        <DialogHeader>
   683→          <DialogTitle className="text-[15px] font-semibold text-white">
   684→            Manage Providers
   685→          </DialogTitle>
   686→          <DialogDescription className="text-[12.5px] leading-snug text-zinc-500">
   687→            Exactly one provider is active at a time. Switching activates the
   688→            selected provider and deactivates all others — rotation then runs
   689→            strictly on its keys, and spend-balanced drift repair keeps every
   690→            monthly pool even.
   691→          </DialogDescription>
   692→        </DialogHeader>
   693→        <div className="min-w-0 space-y-3 pt-1">
   694→          {/* Run 35 — the ACTIVE provider leads: its live ring is
   695→              the first thing the eye lands on when the dialog opens. */}
   696→          {[...providers]
   697→            .sort((a, b) => Number(b.isActive) - Number(a.isActive))
   698→            .map((p) => (
   699→              <ProviderCard
   700→                key={p.id}
   701→                provider={p}
   702→                onSwitch={switchProvider}
   703→                now={now}
   704→              />
   705→            ))}
   706→        </div>
   707→      </DialogContent>
   708→    </Dialog>
   709→  );
   710→}
