   434→/** Fires exactly when the usage day rolls over (01:00 Africa/Algiers),
   435→ *  so the daily token counter visibly resets to 0 at that moment —
   436→ *  no polling loop, one timer armed from the server's own schedule. */
   437→let rolloverTimer: ReturnType<typeof setTimeout> | null = null;
   438→function scheduleDailyRolloverRefresh(dailyResetsAt?: string) {
   439→  if (rolloverTimer) {
   440→    clearTimeout(rolloverTimer);
   441→    rolloverTimer = null;
   442→  }
   443→  if (typeof dailyResetsAt !== "string") return;
   444→  const at = new Date(dailyResetsAt).getTime();
   445→  if (!Number.isFinite(at)) return;
   446→  const delay = Math.max(at - Date.now() + 1000, 30_000); // never tighter than 30s
   447→  rolloverTimer = setTimeout(() => {
   448→    rolloverTimer = null;
   449→    void useHermes.getState().refreshUsage();
   450→  }, delay);
   451→}

   506→  async addAttachments(files) {
   507→    const existing = get().pendingAttachments;
   508→    const room = 5 - existing.length;
   509→    if (room <= 0) {
   510→      return; // caller toasts about the cap
   511→    }
   512→    const accepted = files.slice(0, room);
   513→    const skipped = files.length - accepted.length;
   514→    const staged: PendingAttachment[] = accepted.map((f) => ({
   515→      id: `p-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   516→      name: f.name,
   517→      kindGuess: f.type.startsWith("image/")
   518→        ? "image"
   519→        : f.type === "application/pdf" || f.name.toLowerCase().endsWith(".pdf")
   520→          ? "pdf"
   521→          : f.type.startsWith("text/") || f.name.includes(".")
   522→            ? "text"
   523→            : "unknown",
   524→      status: "extracting",
   525→    }));
   526→    set({ pendingAttachments: [...existing, ...staged] });
   527→    if (skipped > 0) return; // cap message handled by the caller
   528→
   529→    try {
   530→      const form = new FormData();
   531→      for (const f of accepted) form.append("files", f);
   532→      const res = await fetch("/api/attachments/extract", {
   533→        method: "POST",
   534→        body: form,
   535→      });

   540→      };

   545→        pendingAttachments: s.pendingAttachments.map((p) => {

   547→          if (rec) {
   548→            return {
   549→              ...p,
   550→              status: "ready" as const,
   551→              kindGuess: rec.kind,
   552→              record: rec,
   553→            };
   554→          }
   555→          const err = errors.find((e) => e.name === p.name);
   556→          if (err) {
   557→            return { ...p, status: "error" as const, error: err.error };

   561→            ...p,

   563→            error: "Could not read this file.",
   564→          };
   565→        }),
   566→      }));
   567→    } catch (e) {
   568→      // Whole upload failed — every staged file errors out.
   569→      const msg = (e as Error).message || "Extraction failed";

   571→        pendingAttachments: s.pendingAttachments.map((p) =>
   572→          staged.some((q) => q.id === p.id)

   826→  stagePastedSnippet(text) {
   827→    const t = text ?? "";
   828→    // Hard sanity ceiling — anything past this is not a "snippet".
   829→    if (t.length > 1_000_000) {
   830→      return {
   831→        ok: false,
   832→        error: `Snippet is ${(t.length / 1_000_000).toFixed(1)}M characters — too large to attach. Save it as a file and attach that instead.`,
   833→      };
   834→    }
   835→    const existing = get().pendingAttachments;
   836→    if (existing.length >= 5) {
   837→      return {
   838→        ok: false,
   839→        error: "Up to 5 attachments per message — remove one first.",
   840→      };
   841→    }
   842→    const lines = countLines(t);
   843→    // Built ENTIRELY client-side: no upload, no server call — the
   844→    // exact pasted text leaves only when the user sends the message.
   845→    const record: AttachmentRecord = {
   846→      name: `Pasted snippet · ${lines} lines`,
   847→      mime: "text/plain",
   848→      kind: "text",
   849→      method: "text-layer",
   850→      chars: t.length,
   851→      totalChars: t.length,
   852→      truncated: false,
   853→      extract: t,
   854→    };
   855→    set({
   856→      pendingAttachments: [
   857→        ...existing,
   858→        {
   859→          id: `paste-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   860→          name: record.name,
   861→          kindGuess: "text",
   862→          status: "ready",
   863→          record,
   864→        },
   865→      ],
   866→    });
   867→    return { ok: true };
   868→  },
